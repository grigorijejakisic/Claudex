**src/shared/types.ts** — Implement the complete type system from Architecture Section 3.1:

```typescript
// RuntimeCapabilities — declared once per adapter at init
interface RuntimeCapabilities {
  hasFullMessageHistory: boolean;
  hasNativeContextUsage: boolean;
  hasTranscriptAccess: boolean;
  supportsSystemInjection: boolean;
  supportsAsyncEnrichment: boolean;
  hasLocalEmbeddings: boolean;
  supportsTurnEndEvent: boolean;
}

// RuntimeEvent — host-neutral event envelope
interface RuntimeEvent {
  kind: 'session_init' | 'before_prompt' | 'after_tool' | 'after_turn' | 'before_compact' | 'session_end';
  sessionId: string;
  cwd: string;
  timestamp: number;  // Unix epoch ms
  payload: EventPayload;
}

// EventPayload — discriminated union by kind
type EventPayload =
  | SessionInitPayload
  | BeforePromptPayload
  | AfterToolPayload
  | AfterTurnPayload
  | BeforeCompactPayload
  | SessionEndPayload;

// Each payload interface as specified in Architecture Section 3.1
// SessionInitPayload, BeforePromptPayload (with optional tokenUsage, messageHistory),
// AfterToolPayload, AfterTurnPayload, BeforeCompactPayload, SessionEndPayload

// Shared types
interface TokenUsage { inputTokens, outputTokens, contextWindowTokens, utilization }
interface InjectPayload { content: string, tokenEstimate: number, sources: string[] }

// ClaudexCore interface
interface ClaudexCore {
  readonly capabilities: RuntimeCapabilities;
  handleEvent(event: RuntimeEvent): Promise<InjectPayload | void>;
  close(): void;
}

// Message type (for OpenClaw messageHistory)
interface Message { role: 'user' | 'assistant' | 'system'; content: string; }
```

Export ALL types and interfaces. Include JSDoc comments referencing architecture sections.
