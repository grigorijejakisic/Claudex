/**
 * Phase 4.1 feedback → critical_rules promotion bridge.
 *
 * Architecture B (CONTEXT.md lock): when 3 instances of equivalent feedback_*
 * lessons accumulate in a project, INSERT a row into critical_rules with
 * source='system-promoted'. The existing Critical Reminders Tier scorer
 * (assembleCriticalReminders) consumes the new row automatically — no scorer
 * changes for the basic path.
 *
 * Equivalence (4.1 lock): simple normalized text equality — lowercase, strip
 * punctuation, trim whitespace. Embedding-based similarity is deferred to 6.5.
 *
 * Scope: project-scoped only. Cross-project escalation deferred to 6.5.
 *
 * Density threshold: 3 (Claude's discretion to tune; documented in CONTEXT.md).
 */

import type { Database } from 'better-sqlite3';
import { cachedPrepare } from '../core/stmt-cache.js';
import { listLessonsForProject } from './lesson-reader.js';

const PROMOTION_DENSITY = 3;

// Drift risk inference — same regex set as critical-reminders.ts:146 for
// consistency. Reuse via duplication (not import) to avoid circular module
// graph between angel/feedback-promoter and intelligence/critical-reminders.
const SAFETY_KEYWORDS = /\b(never|deadlock|scope|verify|safety)\b/i;
const METHOD_KEYWORDS = /\b(systematic|analysis|method|read|before)\b/i;

/**
 * Normalize lesson body for equivalence comparison.
 *
 * Rule (CONTEXT.md): lowercase, strip punctuation, trim whitespace.
 * We use a conservative punctuation strip that preserves alphanumerics,
 * spaces, hyphens, and underscores — which are content-bearing — but
 * removes commas, periods, exclamation marks, question marks, brackets,
 * parens, and quotes.
 */
export function normalizeRuleText(text: string): string {
  return text
    .toLowerCase()
    .replace(/[.,!?;:'"\[\]\{\}\(\)]+/g, '')
    .replace(/\s+/g, ' ')
    .trim();
}

/**
 * Infer drift_risk for a system-promoted critical_rule.
 *
 * Rule (CONTEXT.md): working-method default; safety if SAFETY_KEYWORDS hit;
 * style fallback if METHOD_KEYWORDS hit and SAFETY didn't.
 *
 * Note: the priority is safety > working-method > style, but the spec says
 * working-method default. We implement: safety wins; otherwise default
 * working-method; style is a separate explicit signal not used here.
 */
function inferDriftRisk(text: string): 'safety' | 'working-method' | 'style' {
  if (SAFETY_KEYWORDS.test(text)) return 'safety';
  if (METHOD_KEYWORDS.test(text)) return 'working-method';
  // No keyword match: still prefer working-method default per CONTEXT.md
  return 'working-method';
}

/**
 * Run promotion for one project. Returns the number of rules promoted.
 *
 * Idempotent: if a system-promoted critical_rule already matches the
 * normalized text for a candidate, the insert is skipped. Existing
 * critical_rules unique index `(project, rule_text)` prevents true
 * duplicates by exact rule_text — we additionally guard via normalized
 * text comparison to handle punctuation-different rephrasing.
 */
export function promoteFeedbackToCriticalRules(db: Database, project: string): number {
  // Group all feedback_* lesson bodies by normalized text, count distinct files
  // (each lesson file is a distinct "instance" in the density count).
  const lessons = listLessonsForProject(project).filter(l => l.frontmatter.type === 'feedback');
  if (lessons.length < PROMOTION_DENSITY) return 0;

  // Equivalence: normalize first non-blank line (the salience headline)
  // OR the full body — we use the salience headline for stability across
  // verbose vs terse phrasings of the same rule.
  const groups = new Map<string, string[]>(); // norm_text → [filename]
  for (const lesson of lessons) {
    const firstLine = lesson.body.split('\n').map(l => l.trim()).find(l => l.length > 0);
    if (!firstLine) continue;
    const cleaned = firstLine.replace(/^#+\s+/, '').replace(/^[-*]\s+/, '');
    const norm = normalizeRuleText(cleaned);
    if (norm.length === 0) continue;
    const arr = groups.get(norm) ?? [];
    arr.push(lesson.filename);
    groups.set(norm, arr);
  }

  let promoted = 0;
  for (const [normText, files] of groups.entries()) {
    if (files.length < PROMOTION_DENSITY) continue;

    // Skip if a system-promoted critical_rule already exists for this normalized text.
    // V17 path: the critical_rules view exposes rule_text via the underlying
    // artifact's `body` column; we query that.
    const exists = cachedPrepare(db,
      `SELECT 1 FROM critical_rules
       WHERE project = ?
         AND source = 'system-promoted'
         AND lower(replace(replace(replace(rule_text, '.', ''), ',', ''), '!', '')) = ?
       LIMIT 1`
    ).get(project, normText);
    if (exists) continue;

    // Reconstruct a presentable rule text from the canonical first-line of
    // the most recent lesson. (We keep the un-normalized casing/punctuation
    // for human readability; normalization is for equivalence only.)
    const repFile = files.sort().slice(-1)[0]; // latest by filename order
    const repLesson = lessons.find(l => l.filename === repFile)!;
    const repFirstLine = repLesson.body.split('\n').map(l => l.trim()).find(l => l.length > 0)!;
    const presentable = repFirstLine.replace(/^#+\s+/, '').replace(/^[-*]\s+/, '');

    const driftRisk = inferDriftRisk(presentable);

    try {
      cachedPrepare(db,
        `INSERT INTO critical_rules (project, rule_text, source, drift_risk, domain_tags, base_ttl)
         VALUES (?, ?, 'system-promoted', ?, NULL, 8)`
      ).run(project, presentable, driftRisk);
      promoted++;
    } catch {
      // Existing unique constraint may reject — non-fatal, count as 0.
    }
  }

  return promoted;
}

/**
 * Phase 4.1 multi-project marker sweep.
 *
 * For every system-promoted critical_rules row, count the number of distinct
 * projects whose system-promoted rules normalize to the same rule_text.
 * Store the count in `data.multi_project_count` (V17 JSON-in-data path —
 * RESEARCH.md decision to avoid view rewrite).
 *
 * The flat +2 scoring boost in assembleCriticalReminders consumes this field.
 *
 * Cadence: heartbeat every N ticks (Claude's discretion — we use every 5 ticks
 * to keep cross-project queries off the hot path. Heartbeat caller decides.)
 */

import type { Database } from 'better-sqlite3';
import { cachedPrepare } from '../core/stmt-cache.js';

/**
 * Update data.multi_project_count for all system-promoted critical_rules.
 *
 * Returns the number of rows updated. Idempotent — repeated calls produce
 * the same field values.
 *
 * Implementation: V17 maps critical_rules to artifact rows via the
 * critical_rules view; the underlying table is `artifact` with kind='critical_rule'.
 * We update via the view (writes pass through INSTEAD OF triggers), updating
 * data.multi_project_count via json_set. If the view does not support UPDATE
 * on data, fall back to direct UPDATE on artifact.
 */
export function updateMultiProjectMarkers(db: Database): number {
  // Fetch all system-promoted rules grouped by normalized text → count of projects
  const counts = cachedPrepare(db,
    `WITH norm AS (
       SELECT
         id, project, rule_text,
         lower(replace(replace(replace(replace(rule_text, '.', ''), ',', ''), '!', ''), '?', '')) AS norm_text
       FROM critical_rules
       WHERE source = 'system-promoted'
     )
     SELECT
       n.id, n.project,
       (SELECT COUNT(DISTINCT m.project) FROM norm m WHERE m.norm_text = n.norm_text) AS multi_count
     FROM norm n`
  ).all() as Array<{ id: number; project: string; multi_count: number }>;

  let updated = 0;

  for (const row of counts) {
    // Update via direct UPDATE on artifact table since data is JSON-in-data.
    // We write through the artifact table — V17 critical_rules view's UPDATE
    // triggers may not support data-column updates, but artifact does.
    //
    // Locate the artifact row by joining to the critical_rules view's id
    // column. The V17 legacy_id_map (or direct mapping in v17-compose) provides
    // the link. Simpler: use the rule_text + project as the key.
    try {
      cachedPrepare(db,
        `UPDATE artifact
         SET data = json_set(COALESCE(data, '{}'), '$.multi_project_count', ?)
         WHERE kind = 'critical_rule'
           AND project_id = ?
           AND body = (SELECT rule_text FROM critical_rules WHERE id = ?)
           AND json_extract(data, '$.source') = 'system-promoted'`
      ).run(row.multi_count, row.project, row.id);
      updated++;
    } catch {
      // Non-fatal: continue with the rest. If the artifact table direct
      // update is not the right path (e.g., the V17 mapping uses a different
      // column for the id), the planner needs to investigate the v17-compose
      // module — Claude's discretion in CONTEXT.md.
    }
  }

  return updated;
}
```

/**
 * Phase 4.1 shape vocabulary promotion sweep.
 *
 * Curates the bounded vocabulary that lesson `shape:` fields draw from.
 * Mechanism (CONTEXT.md lock): LLM proposes candidate values when extracting
 * shape from a lesson; candidates that recur in ≥3 distinct sessions are
 * promoted to canonical vocabulary.
 *
 * Cadence: heartbeat OR /endsession boundary (Claude's discretion). Plan 07
 * runs it on heartbeat for simplicity; the /endsession path can also call
 * recordShapeCandidate when proposing shape values during curation.
 *
 * Tables (Plan 01 migration V18):
 *   shape_vocabulary(field, value, promoted_at_epoch, promoted_session_count)
 *     PRIMARY KEY (field, value)
 *   shape_candidates(field, value, session_id, project, proposed_at_epoch)
 *     PRIMARY KEY (field, value, session_id)
 */

import type { Database } from 'better-sqlite3';
import { cachedPrepare } from '../core/stmt-cache.js';

const PROMOTION_DENSITY = 3;

/**
 * Record a candidate shape proposal. Idempotent on (field, value, session_id).
 */
export function recordShapeCandidate(
  db: Database,
  field: 'task_shape' | 'failure_mode' | 'solution_pattern',
  value: string,
  sessionId: string,
  project: string,
): void {
  const epochMs = Date.now();
  cachedPrepare(db,
    `INSERT OR IGNORE INTO shape_candidates (field, value, session_id, project, proposed_at_epoch)
     VALUES (?, ?, ?, ?, ?)`
  ).run(field, value, sessionId, project, epochMs);
}

/**
 * Promote candidates with ≥3 distinct sessions to canonical vocabulary.
 *
 * After promotion, the candidate rows are NOT deleted (they remain as
 * historical record of which sessions independently proposed the value).
 * The canonical row is the source of truth for vocabulary lookups.
 *
 * Returns the number of candidates promoted in this pass.
 */
export function promoteShapeVocabulary(db: Database): number {
  const candidates = cachedPrepare(db,
    `SELECT field, value, COUNT(DISTINCT session_id) AS sess_count
     FROM shape_candidates
     GROUP BY field, value
     HAVING sess_count >= ?`
  ).all(PROMOTION_DENSITY) as Array<{ field: string; value: string; sess_count: number }>;

  let promoted = 0;
  const epochMs = Date.now();
  for (const c of candidates) {
    // Skip if already canonical
    const exists = cachedPrepare(db,
      'SELECT 1 FROM shape_vocabulary WHERE field = ? AND value = ? LIMIT 1'
    ).get(c.field, c.value);
    if (exists) continue;

    cachedPrepare(db,
      `INSERT INTO shape_vocabulary (field, value, promoted_at_epoch, promoted_session_count)
       VALUES (?, ?, ?, ?)`
    ).run(c.field, c.value, epochMs, c.sess_count);
    promoted++;
  }
  return promoted;
}
```
