The scorer is a function that returns 0-100 score for a given MEMORY.md content string. The test asserts the scorer correctly distinguishes good vs bad files using crafted fixtures.

```ts
import { describe, it, expect } from 'vitest';
import * as fs from 'fs';
import * as path from 'path';
import { findUserTailStart } from '../../angel/memory-md-writer.js';

interface ContentQualityScore {
  total: number;                        // 0-100
  parsing_ok: boolean;                  // any parsing error → 0
  pointer_specificity: number;          // 0-100
  topic_quality: number;                // 0-100
  pointer_density: number;              // 0-100
  handoff_freshness: number;            // 0-100
  details: string[];
}

/**
 * Mechanically score MEMORY.md content per Phase 4.1 SC#3 rubric.
 *
 * Rules:
 * - Zero parsing bugs: top sentinel must be valid; line-anchored marker must
 *   be present; no duplicated headers. → parsing_ok=true, otherwise 0.
 * - ≥80% pointers project-specific: of the lines starting with `- [`, ≥80%
 *   reference filenames matching `feedback_*.md|project_*.md|process_*.md`
 *   (NOT session-IDs or arbitrary text).
 * - Topics not session-IDs: scan ## Lessons section pointer lines; flag any
 *   that contain "session-XXXXXXXX" or look like session-IDs in salience text.
 * - Pointer density ≥1/10 lines: managed section (above marker) should have
 *   at least 1 pointer line per 10 total lines.
 * - Handoff freshness: ## Handoff section either has body content or
 *   "No active handoff." (clean state); not stale text from a prior write.
 *
 * Returns a score 0-100. Total = average of the 4 component scores, gated
 * at 0 if parsing_ok=false.
 */
function scoreContentQuality(content: string): ContentQualityScore {
  const score: ContentQualityScore = {
    total: 0,
    parsing_ok: false,
    pointer_specificity: 0,
    topic_quality: 0,
    pointer_density: 0,
    handoff_freshness: 0,
    details: [],
  };

  const lines = content.replace(/\r\n/g, '\n').split('\n');
  // Parsing: top sentinel must be valid sha256
  if (!/^<!-- CLAUDEX-MANAGED:.* hash=[0-9a-f]{64} -->$/.test(lines[0])) {
    score.details.push('Missing or malformed top sentinel');
    return score;
  }

  // Marker must exist on its own line
  const markerOffset = findUserTailStart(content);
  if (markerOffset < 0) {
    score.details.push('Missing line-anchored <!-- USER EDITABLE --> marker');
    return score;
  }

  // No duplicated headers (count unique ## headers in managed section)
  const managedSection = content.slice(0, markerOffset);
  const managedHeaders = managedSection.split('\n').filter(l => /^## /.test(l));
  const uniqueHeaders = new Set(managedHeaders);
  if (uniqueHeaders.size !== managedHeaders.length) {
    score.details.push(`Duplicated managed-section headers: ${managedHeaders.length - uniqueHeaders.size} duplicates`);
    return score;
  }

  score.parsing_ok = true;

  // Pointer specificity: lines starting with `- [` should reference lesson files
  const pointerLines = managedSection.split('\n').filter(l => /^- \[/.test(l));
  if (pointerLines.length === 0) {
    score.pointer_specificity = 100; // no pointers = trivially specific
  } else {
    const specificCount = pointerLines.filter(l => /\((feedback|project|process)_[a-z0-9-]+\.md\)/.test(l)).length;
    score.pointer_specificity = Math.round((specificCount / pointerLines.length) * 100);
  }

  // Topic quality: pointer lines should NOT contain session-IDs (8-hex pattern)
  const sessionIdRe = /session-[a-f0-9]{8}/i;
  const cleanPointers = pointerLines.filter(l => !sessionIdRe.test(l)).length;
  score.topic_quality = pointerLines.length === 0 ? 100 : Math.round((cleanPointers / pointerLines.length) * 100);

  // Pointer density: ≥1 pointer per 10 lines in managed section
  const managedLineCount = managedSection.split('\n').length;
  if (managedLineCount === 0) {
    score.pointer_density = 0;
  } else {
    const ratio = pointerLines.length / Math.max(1, managedLineCount / 10);
    // ratio ≥ 1.0 → 100; scale down linearly to 0 at ratio=0.5
    score.pointer_density = Math.min(100, Math.max(0, Math.round((ratio - 0.5) / 0.5 * 100)));
  }

  // Handoff freshness: section exists and is either populated or has the
  // canonical empty-state line.
  const handoffMatch = managedSection.match(/## Handoff\n([\s\S]*?)(?=\n##|$)/);
  if (handoffMatch) {
    const body = handoffMatch[1].trim();
    if (body.length > 0) score.handoff_freshness = 100;
```
