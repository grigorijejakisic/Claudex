```ts
/**
 * Phase 4.1 process_* salience trigger evaluation.
 *
 * Five candidate triggers (CONTEXT.md lock):
 *   1. ≥2 corrections in a session
 *   2. User probe broke a false framing the agent had committed to
 *   3. Decision spanned ≥3 pivots before resolving
 *   4. A novel pattern fired that's not yet in the shape vocabulary
 *   5. Long-form design discussion >30 min, conversation-shaped not action-shaped
 *
 * Rule: 2-of-5 must fire. Max 1 process_* per session. Explicit override flag
 * (set in lesson-proposer.ts when user/agent says "this matters") bypasses the
 * 2-of-5 check but NOT the max-1-per-session rule.
 *
 * Each trigger is evaluated as a boolean. The returned set lists which fired
 * with optional metadata for explainability in the proposer's output.
 */

import type { Database } from 'better-sqlite3';
import { cachedPrepare } from '../core/stmt-cache.js';

export interface TriggerEvaluation {
  fired: boolean;
  detail?: string;
}

export interface ProcessTriggerSet {
  corrections: TriggerEvaluation;       // T1: ≥2 corrections
  framing_break: TriggerEvaluation;     // T2: user probe broke false framing
  pivots: TriggerEvaluation;            // T3: ≥3 pivots
  novel_pattern: TriggerEvaluation;     // T4: novel shape outside vocabulary
  long_form: TriggerEvaluation;         // T5: >30min conversation-shaped
  fireCount: number;                    // 0..5
}

/**
 * Evaluate the 5 triggers for a session. Reads telemetry from:
 *   - conversation_turns (turn count, duration, content)
 *   - session_events (correction_detected, topic_shift, etc.)
 *   - artifact (kind='lesson' from this session for shape novelty check)
 */
export function evaluateProcessTriggers(db: Database, sessionId: string): ProcessTriggerSet {
  const set: ProcessTriggerSet = {
    corrections: { fired: false },
    framing_break: { fired: false },
    pivots: { fired: false },
    novel_pattern: { fired: false },
    long_form: { fired: false },
    fireCount: 0,
  };

  // T1: ≥2 corrections in session
  // Use existing correction-detection signals from session_events.
  // Existing event_type values to look for: 'correction_detected',
  // 'directive_correction'. Read src/intelligence/correction-detection.ts and
  // src/intelligence/directive-detector.ts to confirm the exact event names
  // before authoring the WHERE clause.
  try {
    const row = cachedPrepare(db,
      `SELECT COUNT(*) AS cnt FROM session_events
       WHERE session_id = ?
         AND event_type IN ('correction_detected', 'directive_correction')`
    ).get(sessionId) as { cnt: number };
    if (row.cnt >= 2) {
      set.corrections.fired = true;
      set.corrections.detail = `${row.cnt} corrections detected`;
    }
  } catch { /* non-fatal */ }

  // T2: user probe broke false framing
  // Heuristic: A topic shift event of type 'major' on a user-text turn that
  // also has a correction signal nearby. Approximation; refine in plan
  // execution by reading topic-shift.ts and correction-detection.ts.
  try {
    const row = cachedPrepare(db,
      `SELECT 1 FROM session_events
       WHERE session_id = ?
         AND event_type = 'topic_shift'
         AND COALESCE(detail, '') LIKE '%major%'
       LIMIT 1`
    ).get(sessionId);
    if (row && set.corrections.fired) {
      // Co-occurrence with correction is the framing-break signal.
      set.framing_break.fired = true;
      set.framing_break.detail = 'topic shift co-occurred with correction';
    }
  } catch { /* non-fatal */ }

  // T3: decision spanned ≥3 pivots
  // Pivots = topic_shift events of any severity in the same session.
  try {
    const row = cachedPrepare(db,
      `SELECT COUNT(*) AS cnt FROM session_events
       WHERE session_id = ?
         AND event_type = 'topic_shift'`
    ).get(sessionId) as { cnt: number };
    if (row.cnt >= 3) {
      set.pivots.fired = true;
      set.pivots.detail = `${row.cnt} pivots`;
    }
  } catch { /* non-fatal */ }

  // T4: novel pattern outside the bounded vocabulary
  // Heuristic: a candidate shape recorded for THIS session that doesn't
  // appear in shape_vocabulary. This is a session-scoped check — if even one
  // shape candidate was novel, T4 fires.
  try {
    const row = cachedPrepare(db,
      `SELECT 1 FROM shape_candidates sc
       LEFT JOIN shape_vocabulary sv ON sv.field = sc.field AND sv.value = sc.value
       WHERE sc.session_id = ? AND sv.value IS NULL
       LIMIT 1`
    ).get(sessionId);
    if (row) {
      set.novel_pattern.fired = true;
      set.novel_pattern.detail = 'one or more shape candidates not in canonical vocabulary';
    }
  } catch { /* non-fatal */ }

  // T5: long-form design discussion (>30 min, conversation-shaped)
  // Heuristic: session duration > 30 min AND conversation_turns count is high
  // (≥20 turns) AND tool-use count is low relative to turn count (suggests
  // discussion-heavy, not action-heavy). Read sessions.created_at_epoch +
  // sessions.ended_at_epoch (or session_events MAX/MIN timestamp) for duration.
  try {
    const turnCntRow = cachedPrepare(db,
      `SELECT COUNT(*) AS cnt FROM conversation_turns WHERE session_id = ?`
    ).get(sessionId) as { cnt: number };
    const sessionRow = cachedPrepare(db,
      `SELECT created_at_epoch, ended_at_epoch FROM sessions WHERE session_id = ?`
    ).get(sessionId) as { created_at_epoch: number; ended_at_epoch: number | null } | undefined;
    if (sessionRow && sessionRow.ended_at_epoch && turnCntRow.cnt >= 20) {
      const durationMin = (sessionRow.ended_at_epoch - sessionRow.created_at_epoch) / 60;
      // Read tool_use events to gauge action-vs-discussion ratio. If tool_use
      // events / turn count is < 0.5, classify as conversation-shaped.
      const toolUseRow = cachedPrepare(db,
        `SELECT COUNT(*) AS cnt FROM session_events
         WHERE session_id = ? AND event_type = 'tool_use'`
      ).get(sessionId) as { cnt: number };
      const toolPerTurn = toolUseRow.cnt / Math.max(1, turnCntRow.cnt);
      if (durationMin > 30 && toolPerTurn < 0.5) {
        set.long_form.fired = true;
        set.long_form.detail = `${durationMin.toFixed(0)}min, ${turnCntRow.cnt} turns, tool_use/turn=${toolPerTurn.toFixed(2)}`;
      }
    }
  } catch { /* non-fatal */ }

  set.fireCount = [
    set.corrections, set.framing_break, set.pivots, set.novel_pattern, set.long_form
  ].filter(t => t.fired).length;

  return set;
}
```

```ts
/**
 * Phase 4.1 /endsession lesson proposer.
 *
 * Produces 0-3 LessonProposal candidates from session telemetry. The user
 * accepts (writeLesson is called), edits (user-provided content replaces the
 * draft body), or rejects (no write; declined record kept to prevent re-propose).
 *
 * Proposal cap: 3 (CONTEXT.md). For a routine session (no salience signal),
 * returns []. The /endsession skill must handle the empty case silently.
 *
 * Process_* proposal: gated by 2-of-5 trigger rule (process-trigger.ts).
 * Max 1 process_* per session.
 *
 * Feedback_* proposal: extracted from user-correction events with high signal.
 * If the session had a clear corrective directive, surface it as a feedback_*
 * candidate.
 *
 * Project_* proposal: factual decisions/root-causes — extracted from
 * decision-capture events or directive-detector output where the directive
 * is FACT-shaped not RULE-shaped.
 */

import type { Database } from 'better-sqlite3';
import { cachedPrepare } from '../core/stmt-cache.js';
import type { LessonType, TelemetryHandles } from './lesson-types.js';
import { evaluateProcessTriggers } from './process-trigger.js';

export interface LessonProposal {
  type: LessonType;
  proposed_slug: string;             // sanitized; user can override
  proposed_body: string;             // markdown narrative draft
  proposed_telemetry: TelemetryHandles;
  proposed_shape?: { task_shape?: string; failure_mode?: string; solution_pattern?: string };
  // Why we proposed it (for the user prompt UX)
  rationale: string;
  // Trigger metadata for process_* proposals
  triggers_fired?: string[];
}

const MAX_PROPOSALS_PER_SESSION = 3;
const PROCESS_TRIGGER_THRESHOLD = 2; // 2-of-5

/**
 * Build telemetry handles from session telemetry.
 *
 * Deterministic — given the same session events + turns, produces the same
 * telemetry. No LLM calls.
 */
function harvestTelemetry(db: Database, sessionId: string): TelemetryHandles {
  // tools_used: distinct tool_name values from tool_use events
  const tools = cachedPrepare(db,
    `SELECT DISTINCT entity AS tool_name FROM session_events
     WHERE session_id = ? AND event_type = 'tool_use' AND entity IS NOT NULL
     ORDER BY entity ASC LIMIT 50`
  ).all(sessionId).map((r: any) => r.tool_name);

  // files_touched: from tool_use events with file path detail
  // (heuristic — read existing tool_use detail format from session-events.ts)
  const files = cachedPrepare(db,
    `SELECT DISTINCT json_extract(detail, '$.file_path') AS fp FROM session_events
     WHERE session_id = ? AND event_type = 'tool_use'
       AND json_extract(detail, '$.file_path') IS NOT NULL
     LIMIT 50`
  ).all(sessionId).map((r: any) => r.fp).filter(Boolean);

  // errors_encountered: tool errors recorded as event_type='error' or
  // event_type='tool_error'
  const errors = cachedPrepare(db,
    `SELECT DISTINCT entity AS err FROM session_events
     WHERE session_id = ? AND event_type IN ('error', 'tool_error')
       AND entity IS NOT NULL
     LIMIT 20`
  ).all(sessionId).map((r: any) => r.err);

  // user_framing_tokens: distinctive content words from user turns.
  // Heuristic: extract tokens longer than 5 chars that appear ≥2 times across
  // user_text. Use a static stopword list.
  const userTexts = cachedPrepare(db,
    `SELECT user_text FROM conversation_turns WHERE session_id = ? AND user_text IS NOT NULL`
  ).all(sessionId).map((r: any) => r.user_text as string);
  const tokenCounts = new Map<string, number>();
  const STOPWORDS = new Set(['this', 'that', 'with', 'from', 'have', 'will', 'should', 'would', 'could', 'about', 'their', 'these', 'those', 'because', 'which', 'where']);
  for (const text of userTexts) {
    for (const tok of (text.toLowerCase().match(/[a-z][a-z-]{4,}/g) || [])) {
      if (STOPWORDS.has(tok)) continue;
      tokenCounts.set(tok, (tokenCounts.get(tok) || 0) + 1);
    }
  }
  const userFramingTokens = Array.from(tokenCounts.entries())
    .filter(([, count]) => count >= 2)
    .sort((a, b) => b[1] - a[1])
    .slice(0, 10)
    .map(([tok]) => tok);

  // session_arc: derived from topic_shift segmentation labels (if any)
  // Or fallback: simple stages based on turn distribution (read/edit/test/review).
  const arc = cachedPrepare(db,
    `SELECT DISTINCT json_extract(detail, '$.label') AS label FROM session_events
     WHERE session_id = ? AND event_type = 'topic_shift'
       AND json_extract(detail, '$.label') IS NOT NULL
     LIMIT 10`
  ).all(sessionId).map((r: any) => r.label).filter(Boolean);

  // duration_min: ended_at - created_at
  const sessionRow = cachedPrepare(db,
    `SELECT created_at_epoch, ended_at_epoch FROM sessions WHERE session_id = ?`
  ).get(sessionId) as { created_at_epoch: number; ended_at_epoch: number | null } | undefined;
  const durationMin = sessionRow && sessionRow.ended_at_epoch
    ? Math.round((sessionRow.ended_at_epoch - sessionRow.created_at_epoch) / 60)
    : 0;

  // correction_count: from session_events
  const correctionRow = cachedPrepare(db,
    `SELECT COUNT(*) AS cnt FROM session_events
     WHERE session_id = ? AND event_type IN ('correction_detected', 'directive_correction')`
  ).get(sessionId) as { cnt: number };

  return {
    tools_used: tools as string[],
    files_touched: files as string[],
    errors_encountered: errors as string[],
    user_framing_tokens: userFramingTokens,
    session_arc: arc as string[],
    duration_min: durationMin,
    correction_count: correctionRow.cnt,
  };
}

/**
 * Slug from a body line: lowercase, hyphenate, alphanumeric+hyphen only,
 * max 60 chars. Fallback to 'lesson-<epoch_ms>' if line yields empty slug.
 */
function bodyToSlug(line: string): string {
  const slug = line.toLowerCase()
    .replace(/[^a-z0-9\s-]/g, '')
    .replace(/\s+/g, '-')
    .replace(/-+/g, '-')
    .replace(/^-|-$/g, '')
    .slice(0, 60);
  if (slug.length === 0) return `lesson-${Date.now()}`;
  return slug;
}

/**
 * Propose 0-3 lesson candidates for the given session.
 *
 * Returns [] for routine sessions (no correction events, no novel patterns,
 * triggers below threshold). Caller (the /endsession skill) shows nothing
 * to the user in that case.
 */
export function proposeLessonsForSession(db: Database, sessionId: string): LessonProposal[] {
  const proposals: LessonProposal[] = [];
  const telemetry = harvestTelemetry(db, sessionId);

  // ---- Feedback_* candidates from correction events ----
  // Extract directive text from directive-detector events. Each correction
  // becomes one candidate (cap at 2 to leave room for project_*/process_*).
  let feedbackCount = 0;
  try {
    const corrections = cachedPrepare(db,
      `SELECT detail FROM session_events
       WHERE session_id = ?
         AND event_type IN ('directive_correction', 'correction_detected')
         AND detail IS NOT NULL
       ORDER BY id ASC
       LIMIT 5`
    ).all(sessionId) as Array<{ detail: string }>;
    for (const row of corrections) {
      if (feedbackCount >= 2) break;
      let directiveText: string | null = null;
      try {
        const parsed = JSON.parse(row.detail);
        directiveText = parsed.directive_text ?? parsed.text ?? parsed.message ?? null;
      } catch { directiveText = row.detail; }
      if (!directiveText || directiveText.length < 5) continue;

      const firstLine = directiveText.split('\n')[0].trim();
      const slug = bodyToSlug(firstLine);
      proposals.push({
        type: 'feedback',
        proposed_slug: slug,
        proposed_body: `# ${firstLine}\n\nUser-corrected behavior captured during session ${sessionId}.\n`,
        proposed_telemetry: telemetry,
        rationale: 'User correction detected — feedback rule candidate',
      });
      feedbackCount++;
    }
  } catch { /* non-fatal */ }

  // ---- Project_* candidates from decision-capture events ----
  // Decision events that are fact-shaped (not directive-shaped) become
  // project_* proposals. Cap at 1 to leave room for process_* if it fires.
  if (proposals.length < MAX_PROPOSALS_PER_SESSION - 1) {
    try {
      const decisions = cachedPrepare(db,
        `SELECT detail FROM session_events
         WHERE session_id = ?
           AND event_type = 'decision_captured'
           AND detail IS NOT NULL
         ORDER BY id DESC
         LIMIT 1`
      ).get(sessionId) as { detail: string } | undefined;
      if (decisions) {
        let summary: string | null = null;
        try {
          const parsed = JSON.parse(decisions.detail);
          summary = parsed.summary ?? parsed.decision ?? null;
        } catch { summary = decisions.detail; }
        if (summary && summary.length >= 5) {
          const firstLine = summary.split('\n')[0].trim();
          proposals.push({
            type: 'project',
            proposed_slug: bodyToSlug(firstLine),
            proposed_body: `# ${firstLine}\n\nFact captured during session ${sessionId}.\n`,
            proposed_telemetry: telemetry,
            rationale: 'Decision/fact captured — project knowledge candidate',
          });
        }
      }
    } catch { /* non-fatal */ }
  }

  // ---- Process_* candidate (max 1, gated by 2-of-5) ----
  if (proposals.length < MAX_PROPOSALS_PER_SESSION) {
    const triggers = evaluateProcessTriggers(db, sessionId);
    if (triggers.fireCount >= PROCESS_TRIGGER_THRESHOLD) {
      const fired: string[] = [];
      if (triggers.corrections.fired) fired.push(`corrections (${triggers.corrections.detail})`);
      if (triggers.framing_break.fired) fired.push(`framing_break (${triggers.framing_break.detail})`);
      if (triggers.pivots.fired) fired.push(`pivots (${triggers.pivots.detail})`);
      if (triggers.novel_pattern.fired) fired.push(`novel_pattern (${triggers.novel_pattern.detail})`);
      if (triggers.long_form.fired) fired.push(`long_form (${triggers.long_form.detail})`);

      proposals.push({
        type: 'process',
        proposed_slug: `session-${sessionId.slice(0, 8)}-trajectory`,
        proposed_body: `# Decision trajectory for session ${sessionId.slice(0, 8)}\n\nTriggers fired:\n${fired.map(s => `- ${s}`).join('\n')}\n\n## Trajectory\n\n(populate with pivots, rejected alternatives, anti-patterns surfaced)\n`,
        proposed_telemetry: { ...telemetry, triggered_by: fired.map(s => s.split(' ')[0]) },
        rationale: `Process trajectory: ${triggers.fireCount}/5 salience triggers fired`,
        triggers_fired: fired.map(s => s.split(' ')[0]),
      });
    }
  }

  return proposals.slice(0, MAX_PROPOSALS_PER_SESSION);
}
```
