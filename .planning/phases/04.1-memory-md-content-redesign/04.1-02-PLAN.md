```ts
import { describe, it, expect, beforeEach } from 'vitest';
import Database from 'better-sqlite3';
import { initializeSchema, runMigrations } from '../../core/migrations.js';
import { chunkSessionTranscript } from '../../angel/transcript-chunker.js';

describe('transcript-chunker timestamp precision', () => {
  let db: Database.Database;
  const sessionId = 'test-session-ts';
  const project = 'test-project-ts';

  beforeEach(() => {
    db = new Database(':memory:');
    initializeSchema(db);
    runMigrations(db);
  });

  it('writes ms-precision created_at_epoch (13-digit) on new chunks', async () => {
    // Insert at least one session row + enough conversation_turns to force a segment
    db.prepare('INSERT INTO sessions (session_id, project, status) VALUES (?, ?, ?)').run(sessionId, project, 'completed');

    const nowSec = Math.floor(Date.now() / 1000);
    const stmt = db.prepare(
      'INSERT INTO conversation_turns (session_id, turn_number, user_text, assistant_text, timestamp_epoch) VALUES (?, ?, ?, ?, ?)'
    );
    for (let i = 1; i <= 10; i++) {
      stmt.run(sessionId, i, `user msg ${i}`, `assistant msg ${i}`, nowSec + i);
    }

    const result = await chunkSessionTranscript(db, sessionId, project);
    expect(result.inserted).toBeGreaterThan(0);

    const rows = db.prepare(
      `SELECT created_at_epoch FROM artifact WHERE kind='transcript_chunk' AND session_id = ?`
    ).all(sessionId) as Array<{ created_at_epoch: number }>;

    expect(rows.length).toBeGreaterThan(0);
    for (const row of rows) {
      // ms-precision lower bound: any time after 2001-09-09 is ≥ 1e12
      expect(row.created_at_epoch).toBeGreaterThanOrEqual(1e12);
      // upper sanity: no further than 1 second past a Date.now()-derived ms
      expect(row.created_at_epoch).toBeLessThanOrEqual(Date.now() + 1000);
    }
  });

  it('matches expected upscale factor (1000) for known input timestamps', async () => {
    db.prepare('INSERT INTO sessions (session_id, project, status) VALUES (?, ?, ?)').run(sessionId, project, 'completed');

    const fixedSec = 1745923400; // a known seconds-precision value (~ 2025)
    const stmt = db.prepare(
      'INSERT INTO conversation_turns (session_id, turn_number, user_text, assistant_text, timestamp_epoch) VALUES (?, ?, ?, ?, ?)'
    );
    for (let i = 1; i <= 10; i++) {
      stmt.run(sessionId, i, `u${i}`, `a${i}`, fixedSec + i);
    }

    const result = await chunkSessionTranscript(db, sessionId, project);
    expect(result.inserted).toBeGreaterThan(0);

    const rows = db.prepare(
      `SELECT MAX(created_at_epoch) AS maxEpoch FROM artifact WHERE kind='transcript_chunk' AND session_id = ?`
    ).get(sessionId) as { maxEpoch: number };

    expect(rows.maxEpoch).toBe((fixedSec + 10) * 1000);
  });
});
```
