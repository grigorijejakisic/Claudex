Vitest test, single `describe` "Phase 5.5 Vesna — curation feedback loop lifecycle". Setup uses `:memory:` DB + tmpdir for lesson files (mock `os.homedir()` or pass an injected basedir to `resolveLessonPath` per Plan 04 Task 3 testability seam).

```ts
describe('Phase 5.5 Vesna — curation feedback loop lifecycle', () => {
  // Simulate a 30-day timeline. Day 0 = T0; nowEpochMs at day N = T0 + N*86_400_000.
  const T0 = 1_700_000_000_000;  // arbitrary anchor
  const day = (n: number) => T0 + n * 86_400_000;

  let db: Database;
  let basedir: string;

  beforeEach(() => {
    db = new Database(':memory:');
    runMigrations(db);
    basedir = fs.mkdtempSync(path.join(os.tmpdir(), 'p55-vesna-'));
    __resetGatesForTests();
  });

  afterEach(() => {
    db.close();
    fs.rmSync(basedir, { recursive: true, force: true });
  });

  it('promotes a productive pointer (steady recalls + helpful=true)', () => {
    // Pointer A: lesson written at day 0 (created_at_epoch = T0).
    // 4 recalls spread across days 1, 3, 5, 7. Helpful=true on day 7.
    const pathA = seedLesson(basedir, {
      project: 'proj-a',
      filename: 'feedback_use-cache.md',
      createdAt: day(0),
      tier: 'foreground',
      lastFired: 0,  // never fired before
    });
    const pidA = ensurePointerId(db, 'proj-a', 'feedback_use-cache.md', 'lesson');
    insertRecall(db, pidA, 'sess-1', day(1));
    insertRecall(db, pidA, 'sess-2', day(3));
    insertRecall(db, pidA, 'sess-3', day(5));
    insertRecall(db, pidA, 'sess-4', day(7));
    markPointersHelpful(db, 'sess-4', [pidA]);

    // Day 8: weekly promote sweep runs.
    const promoted = sweepPromotePointers(db, day(8));
    expect(promoted).toBe(1);

    const parsed = parseLessonFile(pathA)!;
    expect(parsed.frontmatter.last_fired_at_epoch).toBe(day(8));
  });

  it('archives a never-touched pointer (0 recalls in 90d, helpful=null)', () => {
    // Pointer B: lesson written at day -91 (long ago). Never recalled.
    const pathB = seedLesson(basedir, {
      project: 'proj-a',
      filename: 'process_old-pattern.md',
      createdAt: day(-91),
      tier: 'foreground',
      lastFired: 0,
    });
    ensurePointerId(db, 'proj-a', 'process_old-pattern.md', 'lesson');

    // Day 0: daily archive sweep runs.
    const archived = sweepArchivePointers(db, day(0));
    expect(archived).toBe(1);

    const parsed = parseLessonFile(pathB)!;
    expect(parsed.frontmatter.tier).toBe('background');
  });

  it('does not archive an active pointer (recent recall within 90d)', () => {
    // Pointer C: lesson written at day -100, but had 1 recall on day -10.
    const pathC = seedLesson(basedir, {
      project: 'proj-a',
      filename: 'feedback_active.md',
      createdAt: day(-100),
      tier: 'foreground',
      lastFired: 0,
    });
    const pidC = ensurePointerId(db, 'proj-a', 'feedback_active.md', 'lesson');
    insertRecall(db, pidC, 'sess-x', day(-10));

    const archived = sweepArchivePointers(db, day(0));
    expect(archived).toBe(0);
    expect(parseLessonFile(pathC)!.frontmatter.tier).toBe('foreground');
  });

  it('does not promote an under-recalled pointer (recall_count < 3)', () => {
    // Pointer D: 2 recalls + helpful=1. Should NOT promote.
    const pathD = seedLesson(basedir, {
      project: 'proj-a',
      filename: 'feedback_underused.md',
      createdAt: day(0),
      tier: 'foreground',
      lastFired: 0,
    });
    const pidD = ensurePointerId(db, 'proj-a', 'feedback_underused.md', 'lesson');
    insertRecall(db, pidD, 'sess-1', day(1));
    insertRecall(db, pidD, 'sess-2', day(3));
    markPointersHelpful(db, 'sess-2', [pidD]);

    const promoted = sweepPromotePointers(db, day(8));
    expect(promoted).toBe(0);
    expect(parseLessonFile(pathD)!.frontmatter.last_fired_at_epoch).toBe(0);
  });

  it('does not promote without any helpful=1 (3 recalls but all null)', () => {
    const pathE = seedLesson(basedir, {
      project: 'proj-a',
      filename: 'feedback_nohelp.md',
      createdAt: day(0),
      tier: 'foreground',
      lastFired: 0,
    });
    const pidE = ensurePointerId(db, 'proj-a', 'feedback_nohelp.md', 'lesson');
    insertRecall(db, pidE, 'sess-1', day(1));
    insertRecall(db, pidE, 'sess-2', day(3));
    insertRecall(db, pidE, 'sess-3', day(5));
    // No mark.

    const promoted = sweepPromotePointers(db, day(8));
    expect(promoted).toBe(0);
  });

  it('rehabilitation path: archived pointer can re-promote out of background', () => {
    // Pointer F: starts in background (was archived previously). Then earns
    // 3 recalls + helpful within 7 days. Promote sweep should re-foreground it.
    const pathF = seedLesson(basedir, {
      project: 'proj-a',
      filename: 'process_revived.md',
      createdAt: day(-100),
      tier: 'background',  // already archived
      lastFired: 0,
    });
    const pidF = ensurePointerId(db, 'proj-a', 'process_revived.md', 'lesson');
    insertRecall(db, pidF, 'sess-1', day(1));
    insertRecall(db, pidF, 'sess-2', day(3));
    insertRecall(db, pidF, 'sess-3', day(5));
    markPointersHelpful(db, 'sess-3', [pidF]);

    const promoted = sweepPromotePointers(db, day(8));
    expect(promoted).toBe(1);

    const parsed = parseLessonFile(pathF)!;
    expect(parsed.frontmatter.tier).toBe('foreground');
    expect(parsed.frontmatter.last_fired_at_epoch).toBe(day(8));
  });
});
```

**Test seam constraint:** `seedLesson` and `insertRecall` are local helpers. `seedLesson` calls `writeLesson` (Plan 01) with overridden basedir; `insertRecall` is a direct INSERT bypassing `recordPointerRecall` so we can backdate `retrieved_at_epoch_ms` to simulated days. This is acceptable — the production write path is exercised in `pointer-recall.test.ts` (Plan 01 Task 7); here we're testing sweep behavior over time, not the recording pipeline.

The basedir override needs `resolveLessonPath` (in `curation-sweep.ts`) to be testable. If the executor of Plan 04 chose to mock `os.homedir()` instead, this test follows that approach. If Plan 04 made `basedir` injectable, this test passes it through.
