**src/angel/boundary/composition-rule.ts:**

Pure function — no DB, no I/O. Mirror the formal aliveness rule in CONTEXT.md verbatim so future readers can compare line-for-line.

```typescript
import type { BoundaryThresholds } from './thresholds.js';

export interface SessionLivenessRow {
  session_id: string;
  project: string;
  pid: number | null;        // resolved by caller (boundary-detector.ts) before invocation
  pid_alive: boolean;        // caller pre-computes via isPidAlive()
  last_heartbeat_ts: number | null;
  last_jsonl_write_ts: number | null;
  has_clean_endsession: boolean; // caller pre-computes by reading cursor.last_close_event_id
}

export type CloseReason = 'clean_endsession' | 'idle_timeout' | 'jsonl_silent' | 'pid_dead';

export type Classification =
  | { state: 'alive' }
  | { state: 'dormant' }
  | { state: 'terminated'; close_reason: CloseReason };

/**
 * Apply the formal aliveness composition rule from CONTEXT.md.
 *
 * ALIVE       <=>  pid_alive
 *                ∧ (now - last_jsonl_write_ts < T_jsonl)
 *                ∧ (now - last_heartbeat_ts < T_heartbeat OR now - last_jsonl_write_ts < T_jsonl_short)
 *
 * DORMANT     <=>  ¬ALIVE ∧ (now - last_jsonl_write_ts < T_jsonl + T_grace)
 *
 * TERMINATED  <=>  ¬ALIVE ∧ ( jsonl_idle_full ∨ clean_endsession ∨ pid_dead_corroborated )
 *
 * close_reason precedence (when TERMINATED):
 *   1. clean_endsession  (already short-circuited by caller — caller never invokes this for has_clean_endsession=true sessions, but defensive branch returns it if reached)
 *   2. idle_timeout      (jsonl silent ≥ T_jsonl + T_grace)
 *   3. jsonl_silent      (PID dead AND jsonl ≥ T_jsonl_short AND heartbeat ≥ T_heartbeat)
 *   4. pid_dead          (PID dead AND heartbeat ≥ T_heartbeat AND jsonl ≥ T_jsonl_short — distinguished for telemetry)
 */
export function classifySession(
  now: number,
  row: SessionLivenessRow,
  t: BoundaryThresholds,
): Classification {
  if (row.has_clean_endsession) {
    return { state: 'terminated', close_reason: 'clean_endsession' };
  }

  const jsonlAge = row.last_jsonl_write_ts !== null ? now - row.last_jsonl_write_ts : Infinity;
  const heartbeatAge = row.last_heartbeat_ts !== null ? now - row.last_heartbeat_ts : Infinity;

  // ALIVE clause
  const jsonlFreshFull  = jsonlAge < t.tJsonl;
  const heartbeatFresh  = heartbeatAge < t.tHeartbeat;
  const jsonlFreshShort = jsonlAge < t.tJsonlShort;
  const aliveClause     = row.pid_alive && jsonlFreshFull && (heartbeatFresh || jsonlFreshShort);

  if (aliveClause) return { state: 'alive' };

  // ¬ALIVE branch — decide DORMANT vs TERMINATED
  const jsonlIdleFull       = jsonlAge >= (t.tJsonl + t.tGrace);
  const pidDeadCorroborated = !row.pid_alive && heartbeatAge >= t.tHeartbeat && jsonlAge >= t.tJsonlShort;

  if (jsonlIdleFull) {
    return { state: 'terminated', close_reason: 'idle_timeout' };
  }
  if (pidDeadCorroborated) {
    // Distinguish jsonl_silent vs pid_dead for telemetry per CONTEXT close_reason table.
    // The table explicitly notes implementations may collapse if downstream doesn't differentiate;
    // we keep them distinct so `episode_close_emitted` telemetry can A/B them.
    if (jsonlAge >= t.tJsonl) {
      return { state: 'terminated', close_reason: 'jsonl_silent' };
    }
    return { state: 'terminated', close_reason: 'pid_dead' };
  }

  // ¬ALIVE but within DORMANT window
  return { state: 'dormant' };
}
```

**src/angel/boundary/cursor.ts:**

Two main exports:
- `loadCursor(db, project, sessionId)` — `SELECT * FROM episode_boundary_cursor WHERE project=? AND session_id=?`. Returns null if absent. Read-only, no transaction.
- `commitBoundaryTick(db, payload)` — atomic write: cursor advance + (optional) close-marker emission, with heartbeat-compare guard inside the same transaction.

Payload type:
```typescript
interface BoundaryTickPayload {
  project: string;
  sessionId: string;
  jsonlOffset: number;
  lastEventTsEpoch: number;
  closeMarker?: {
    reason: 'idle_timeout' | 'jsonl_silent' | 'pid_dead';  // clean_endsession is hook-driven, never via this path
    detectionSnapshot: {                                     // values seen by classifier at decision time
      last_heartbeat_ts: number | null;

    metadata: {                                              // metadata for the env event row
      duration_seconds: number;
      event_count: number;
      pid_alive: boolean;
      last_heartbeat_ts: number | null;
      last_jsonl_write_ts: number | null;
    };
  };
}
```

Function body:

```typescript
const tx = db.transaction(() => {
  if (payload.closeMarker) {
    // HEARTBEAT-COMPARE-BEFORE-CLEANUP — re-read freshness inside tx
    const fresh = db.prepare(
      `SELECT last_heartbeat_ts, last_jsonl_write_ts FROM sessions WHERE session_id = ?`
    ).get(payload.sessionId) as { last_heartbeat_ts: number | null; last_jsonl_write_ts: number | null } | undefined;

    const snap = payload.closeMarker.detectionSnapshot;
    const heartbeatWentFresh = fresh && fresh.last_heartbeat_ts !== snap.last_heartbeat_ts;
    const jsonlWentFresh     = fresh && fresh.last_jsonl_write_ts !== snap.last_jsonl_write_ts;

    if (heartbeatWentFresh || jsonlWentFresh) {
      // Session Amnesia guard fired — abort close, write telemetry INSIDE tx
      // so it commits with the cursor advance (cursor still progresses).
      db.prepare(
        `INSERT INTO telemetry (session_id, event_kind, detail, adapter)
         VALUES (?, 'close_aborted_stale_check_failed', json_object(
           'project', ?,
           'detection_heartbeat_ts', ?,
           'detection_jsonl_write_ts', ?,
           'fresh_heartbeat_ts', ?,
           'fresh_jsonl_write_ts', ?,
           'close_reason_attempted', ?
         ), 'angel-boundary')`
      ).run(
        payload.sessionId,
        payload.project,
        snap.last_heartbeat_ts,
        snap.last_jsonl_write_ts,
        fresh.last_heartbeat_ts,
        fresh.last_jsonl_write_ts,
        payload.closeMarker.reason,
      );
      // Fall through to cursor advance — no close emission.
    } else {
      // Heartbeat-compare passed → emit episode_closed env event row + telemetry.
      const evRow = db.prepare(
        `INSERT INTO episodic_events
           (session_id, project, turn_number, type, source, content, provenance, parent_event_id, content_hash, metadata_json)
         VALUES (?, ?, NULL, 'environmental_event', 'angel-boundary', ?, 'environmental', NULL, ?,
                 json_object(
                   'episode_closed', json('true'),
                   'close_reason', ?,
                   'duration_seconds', ?,
                   'event_count', ?,
                   'pid_alive', ?,
                   'last_heartbeat_ts', ?,
                   'last_jsonl_write_ts', ?
                 ))
         RETURNING id`
      ).get(
        payload.sessionId,
        payload.project,
        `Episode closed: ${payload.closeMarker.reason}`,
        sha256(`Episode closed: ${payload.closeMarker.reason}:${payload.sessionId}:${payload.lastEventTsEpoch}`),
        payload.closeMarker.reason,
        payload.closeMarker.metadata.duration_seconds,
        payload.closeMarker.metadata.event_count,
        payload.closeMarker.metadata.pid_alive ? 1 : 0,
        payload.closeMarker.metadata.last_heartbeat_ts,
        payload.closeMarker.metadata.last_jsonl_write_ts,
      ) as { id: number } | undefined;
      const closeEventId = evRow?.id ?? null;

      // Mirror SessionEnd hook's transaction: also flip sessions.status='completed'
      // (CONTEXT Open Question #2 — answered YES in RESEARCH.md to avoid divergence
      // between episodic_events and sessions).
      db.prepare(
        `UPDATE sessions SET status = 'completed', ended_at_epoch = COALESCE(ended_at_epoch, ?) WHERE session_id = ?`
      ).run(payload.lastEventTsEpoch, payload.sessionId);

      // Cursor UPSERT — including last_close_event_id
      db.prepare(
        `INSERT INTO episode_boundary_cursor
           (project, session_id, last_processed_jsonl_offset, last_processed_event_ts_epoch, last_close_event_id)
         VALUES (?, ?, ?, ?, ?)
         ON CONFLICT(project, session_id) DO UPDATE SET
           last_processed_jsonl_offset   = excluded.last_processed_jsonl_offset,
           last_processed_event_ts_epoch = excluded.last_processed_event_ts_epoch,
           last_close_event_id           = excluded.last_close_event_id`
      ).run(payload.project, payload.sessionId, payload.jsonlOffset, payload.lastEventTsEpoch, closeEventId);

      // Telemetry — episode_close_emitted
      db.prepare(
        `INSERT INTO telemetry (session_id, event_kind, detail, adapter)
         VALUES (?, 'episode_close_emitted', json_object('close_reason',?,'close_event_id',?), 'angel-boundary')`
      ).run(payload.sessionId, payload.closeMarker.reason, closeEventId);

      return; // skip the no-close cursor-only branch below
    }
  }

  // No close marker (or aborted) → cursor advance only
  db.prepare(
    `INSERT INTO episode_boundary_cursor (project, session_id, last_processed_jsonl_offset, last_processed_event_ts_epoch)
     VALUES (?, ?, ?, ?)
     ON CONFLICT(project, session_id) DO UPDATE SET
       last_processed_jsonl_offset   = excluded.last_processed_jsonl_offset,
       last_processed_event_ts_epoch = excluded.last_processed_event_ts_epoch`
  ).run(payload.project, payload.sessionId, payload.jsonlOffset, payload.lastEventTsEpoch);
});
tx();
```

**src/angel/boundary/boundary-detector.ts:**

Single export: `runBoundaryTick(db, thresholds)` returning a `BoundaryTickResult` object summarizing what happened (count of closes emitted, count of aborts, count of reopens, count of cursor replays).

Body outline:

1. **Find candidate sessions:** SELECT from `sessions LEFT JOIN episode_boundary_cursor` where `status='active'` OR `last_close_event_id IS NULL`, ordered by oldest `last_jsonl_write_ts` first, `LIMIT 25` (bounded sweep, similar to existing `getIdleSessions` LIMIT 10 but bigger to handle close + reopen passes in same tick).
2. **For each candidate:**
   a. Resolve PID — for v5, use the `pid_file` mechanism (existing pattern from `src/angel/pid-file.ts`) OR read from `sessions.adapter` metadata if PID is recorded there. If no PID is known, treat as `pid_alive=false` (defensive).
   b. Pre-compute `has_clean_endsession`: `cursor.last_close_event_id IS NOT NULL` AND the corresponding `episodic_events` row's `metadata_json` close_reason is `clean_endsession`. Skip this session if so — it's already closed cleanly. **Don't continue, don't reopen-check on this tick.**
   c. If `last_close_event_id IS NOT NULL` (closed by a prior tick, NON-clean reason): **REOPEN BRANCH** — check if `sessions.last_jsonl_write_ts` is fresher than `cursor.last_processed_event_ts_epoch + thresholds.tReopen`:
      - Within `T_reopen` AND fresher than the closed-time → emit `re_opened` env event row (provenance=environmental, source=angel-boundary, type=environmental_event, metadata: `{re_opened:true, gap_seconds, prior_close_event_id}`), `UPDATE sessions SET status='active' WHERE session_id=?`, write `episode_reopen` telemetry, advance cursor with `last_close_event_id=NULL`.
      - Beyond `T_reopen` → write `episode_reopen_anomaly` telemetry with `gap_seconds`, do NOT change session status (stays completed), do NOT auto-allocate new session_id.
   d. Otherwise call `classifySession(now, row, thresholds)`:
      - `alive` → no-op (cursor doesn't need advance unless we processed events; for v5 substrate we don't tail JSONL content, so alive sessions are noise — skip).
      - `dormant` → no-op (telemetry could flood; skip).
      - `terminated` with reason ≠ `clean_endsession` → call `commitBoundaryTick` with the reason and detection snapshot.
3. **Cursor offset overflow recovery (Pitfall 3):** for any session whose `cursor.last_processed_jsonl_offset > fs.statSync(jsonlPath).size`, write `boundary_cursor_replay` telemetry with `detail.reason='offset_overflow'` and reset cursor offset to 0. Skip this check if file doesn't exist (session may have been deleted from disk).
4. **Bounded errors:** every per-session block in a try/catch — one bad session never blocks others. Record per-session failures via `episodic_write_failure` telemetry (mirrors `episodic-events.ts` pattern).
