```typescript
/**
 * Phase 6 boundary thresholds — env-configurable with CONTEXT.md locked defaults.
 *
 * All values are seconds. Sweep cadence reuses Angel's existing
 * `heartbeatIntervalMs` (DEFAULT_ANGEL_CONFIG in src/angel/types.ts:135).
 */

export interface BoundaryThresholds {
  tJsonl: number;         // 15 min default — dormant idle threshold
  tGrace: number;         // 15 min default — dormant→terminated grace
  tHeartbeat: number;     // 5 min default — heartbeat staleness (PID-dead corroborator)
  tJsonlShort: number;    // 5 min default — short-window JSONL corroborator
  tReopen: number;        // 60 min default — re-open vs anomaly window
  jsonlDebounceMs: number;// 200 ms default — chokidar awaitWriteFinish stabilityThreshold
}

export const LOCKED_DEFAULTS: BoundaryThresholds = {
  tJsonl: 15 * 60,
  tGrace: 15 * 60,
  tHeartbeat: 5 * 60,
  tJsonlShort: 5 * 60,
  tReopen: 60 * 60,
  jsonlDebounceMs: 200,
};

function readEnvInt(name: string, fallback: number): number {
  const raw = process.env[name];
  if (raw === undefined || raw === '') return fallback;
  const parsed = parseInt(raw, 10);
  return Number.isFinite(parsed) && parsed > 0 ? parsed : fallback;
}

export function loadThresholds(): BoundaryThresholds {
  return {
    tJsonl:          readEnvInt('CLAUDEX_EPISODE_T_JSONL_SECONDS',        LOCKED_DEFAULTS.tJsonl),
    tGrace:          readEnvInt('CLAUDEX_EPISODE_T_GRACE_SECONDS',        LOCKED_DEFAULTS.tGrace),
    tHeartbeat:      readEnvInt('CLAUDEX_EPISODE_T_HEARTBEAT_SECONDS',    LOCKED_DEFAULTS.tHeartbeat),
    tJsonlShort:     readEnvInt('CLAUDEX_EPISODE_T_JSONL_SHORT_SECONDS',  LOCKED_DEFAULTS.tJsonlShort),
    tReopen:         readEnvInt('CLAUDEX_EPISODE_REOPEN_WINDOW_SECONDS',  LOCKED_DEFAULTS.tReopen),
    jsonlDebounceMs: readEnvInt('CLAUDEX_EPISODE_JSONL_DEBOUNCE_MS',      LOCKED_DEFAULTS.jsonlDebounceMs),
  };
}
```

```typescript
/**
 * Phase 6 PID liveness probe — cross-platform via process.kill(pid, 0).
 *
 * Per CONTEXT.md (Claude's Discretion #1): uses the cross-platform builtin,
 * not tasklist / ps shell-out. EPERM-means-alive idiom — POSIX returns EPERM
 * when the process exists but we lack signal permission; that still means
 * it's alive.
 *
 * NEVER alone closes a session — composition rule in boundary-detector.ts
 * always corroborates with heartbeat AND JSONL signals (CONTEXT decision:
 * stale heartbeat alone NEVER closes a session).
 */
export function isPidAlive(pid: number | null | undefined): boolean {
  if (typeof pid !== 'number' || pid <= 0 || !Number.isInteger(pid)) return false;
  try {
    process.kill(pid, 0);
    return true;
  } catch (err) {
    const code = (err as NodeJS.ErrnoException)?.code;
    if (code === 'EPERM') return true;
    return false;
  }
}
```

**src/angel/boundary/jsonl-watcher.ts:**

Module exports:
- `startJsonlWatcher(db, opts?)` — returns a controller `{ close(): Promise<void>, healthCheck(): { healthy: boolean; consecutiveErrors: number } }`. The controller's job is to wrap `chokidar.FSWatcher` with retry + telemetry, NOT expose chokidar directly.
- `parseSessionIdFromPath(filePath)` — exported for test access; returns `{ project, sessionId } | null`. Path layout: `~/.claude/projects/{projectFlattened}/{session_id}.jsonl`.

Implementation contract:
1. Watch glob: `path.join(homedir(), '.claude', 'projects', '**', '*.jsonl')`. Use forward slashes in glob (chokidar normalizes for Windows).
2. Options: `{ ignoreInitial: true, ignored: (p) => /node_modules/.test(p), awaitWriteFinish: { stabilityThreshold: thresholds.jsonlDebounceMs, pollInterval: 50 }, persistent: true, ignorePermissionErrors: true }`.
3. Event handlers:
   - `add` and `change` → call `updateWriteTs(filePath)` which: parses session_id, reads `now = floor(Date.now()/1000)`, runs `cachedPrepare(db, "UPDATE sessions SET last_jsonl_write_ts = ? WHERE session_id = ?").run(now, sessionId)`. Wrap entire body in try/catch — DB errors must NEVER kill the watcher.
   - `error` → write a `jsonl_watcher_unreachable` telemetry row (`detail: { error_message, consecutive_errors, backoff_ms_next }`), increment `consecutiveErrors`, schedule re-watch with exponential backoff (1000, 2000, 4000, 8000, 16000, 30000ms cap). After 5+ consecutive errors, KEEP retrying at 30s — never give up. Reset `consecutiveErrors` to 0 on first successful event after recovery.
4. `close()` — call `watcher.close()`, clear any pending backoff timer, mark controller closed.

Important: do NOT export the raw `chokidar.FSWatcher` from this module. Callers see only the controller. This keeps Plan 03's integration in heartbeat tick from leaking chokidar internals into Angel.
