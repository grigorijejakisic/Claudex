```typescript
import { describe, test, expect, vi, afterEach, it } from 'vitest';
import { encode } from 'gpt-tokenizer';
import { createHash } from 'node:crypto';
import {
  makeColdStart, makeWarmStart, makeHandoffStart, makeGsdActiveStart,
  cleanupFixture, type CacheStabilityFixture,
} from './assembler-cache-fixtures.js';

const SCENARIOS = [
  { name: 'cold-start', build: makeColdStart },
  { name: 'warm-start-with-memory-md', build: makeWarmStart },
  { name: 'handoff-start', build: makeHandoffStart },
  { name: 'gsd-active-start', build: makeGsdActiveStart },
];

const sha = (s: string) => createHash('sha256').update(s).digest('hex');

const TOKEN_BUDGET = 500;

for (const sc of SCENARIOS) {
  describe(`cache-stability: ${sc.name}`, () => {
    let fx: CacheStabilityFixture;
    afterEach(() => { if (fx) cleanupFixture(fx); });

    // Layer 1 — TOK-01: token budget ≤500
    // Pre-deletion: this is expected to fail. Use it.skipIf so the suite stays green
    // while we measure progress. After Plan 05 closes Tier C, remove the skip.
    test('layer 1 — token budget ≤500 (cl100k_base)', () => {
      fx = sc.build();
      const out = fx.run();
      const tokens = encode(out.content).length;

      // Phase 5 in-progress gate. Plan 09 will remove the conditional and make this a hard assert.
      const PRE_DELETION_OBSERVED_OK_TO_OVERRUN = process.env.CLAUDEX_P5_TOKEN_GATE_STRICT !== '1';
      if (tokens > TOKEN_BUDGET && PRE_DELETION_OBSERVED_OK_TO_OVERRUN) {
        // Document, do not fail, until Tier C closes
        console.warn(`[cache-stability:${sc.name}] tokens=${tokens} (over ${TOKEN_BUDGET} — expected pre-deletion)`);
        expect.soft(tokens).toBeLessThanOrEqual(TOKEN_BUDGET);
        return;
      }
      expect(tokens).toBeLessThanOrEqual(TOKEN_BUDGET);
    });

    // Layer 2 — CACH-01: byte-identical across consecutive runs
    test('layer 2 — byte-identical across consecutive runs', () => {
      fx = sc.build();
      const a = fx.run();
      const b = fx.run();
      expect(sha(a.content)).toBe(sha(b.content));
    });

    // Layer 3 — CACH-02: invariant under volatile mutation
    test('layer 3 — invariant under clock/session-ID/host-env mutation', () => {
      fx = sc.build();
      const baseline = fx.run();
      const baselineHash = sha(baseline.content);

      // Mutate clock by +100s
      const originalNowEpoch = fx.nowEpoch;
      fx.nowEpoch = originalNowEpoch + 100;
      vi.useFakeTimers();
      vi.setSystemTime((originalNowEpoch + 100) * 1000);

      // Mutate session-ID
      fx.sessionId = '11111111-1111-1111-1111-111111111111';

      // Mutate host-env: flip path separator style
      fx.projectDir = fx.projectDir.replace(/\\\\/g, '/');

      const mutated = fx.run();
      vi.useRealTimers();

      expect(sha(mutated.content)).toBe(baselineHash);
    });
  });
}
```

Notes for the implementer:
- The `expect.soft` + early return in Layer 1 is the controlled overrun escape for pre-deletion runs. Once Tier C is deleted (Plan 05), set `CLAUDEX_P5_TOKEN_GATE_STRICT=1` in CI env; Plan 09 will flip the default to strict.
- Layer 3 mutates fx fields BETWEEN calls to `fx.run()` — verify `fx.run()` reads the live values, not a snapshot. If the fixture closes over initial values, refactor to read from `this.*` each invocation.
- Use `vi.setSystemTime` for the clock mutation. The CACH-03 hardening fix in Plan 01 made `assembler.ts:572,:657` read `params.nowEpoch ?? Date.now()/1000`; tests pass `nowEpoch` so they're unaffected by the system clock. The `vi.setSystemTime` is a defense-in-depth check: any code path that bypasses `params.nowEpoch` will still be caught.
- If Layer 2 or 3 FAILS in this plan against the current cascade, that's a bug — either CACH-03 hardening missed a site, or there's a non-deterministic source we haven't catalogued. Treat as a Plan 01 bug, not a Plan 02 bug. Open a `claim` signal and fix forward.
