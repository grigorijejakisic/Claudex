```typescript
// I1: initialUserMessage auto-prime — frontmatter-contract-gated, default-on.
// Per Phase 5 / INJ-06: fires only when handoff frontmatter status: active AND
// phase EXACTLY matches STATE.md Current Phase. No config flag, no mtime gate.
// EXACT phase match (no startsWith / no decimal-prefix matching) — Phase 7.5
// will revisit semantics when handoff format redesigns.
let initialMessage: string | undefined;
try {
  const sessionType = (input.type as string) ?? '';
  if (sessionType === 'startup' || sessionType === '') {
    const handoffPath = path.join(input.cwd, 'context', 'handoffs', 'ACTIVE.md');
    if (fs.existsSync(handoffPath)) {
      const handoffContent = normalizeText(fs.readFileSync(handoffPath, 'utf-8'));
      const fmMatch = handoffContent.match(/^---\s*\n([\s\S]*?)\n---/);
      if (fmMatch) {
        const fmBlock = fmMatch[1];
        const isActive = /^status:\s*active\b/im.test(fmBlock);
        const handoffPhaseMatch = fmBlock.match(/^phase:\s*["']?([^"'\n]+?)["']?\s*$/im);
        const handoffPhase = handoffPhaseMatch?.[1]?.trim();

        // Read STATE.md current phase via state-reader's parser
        const statePath = path.join(input.cwd, '.planning', 'STATE.md');
        let statePhase: string | undefined;
        if (fs.existsSync(statePath)) {
          const stateContent = normalizeText(fs.readFileSync(statePath, 'utf-8'));
          const m = stateContent.match(/Current Phase:\s*(\d+(?:\.\d+)?)/);
          statePhase = m?.[1];
        }

        // EXACT match (per team-lead Q3 verdict)
        if (isActive && handoffPhase && statePhase && handoffPhase === statePhase) {
          // Extract summary: prefer `summary:` frontmatter key; fall back to first
          // non-blank, non-H1 body line
          const summaryMatch = fmBlock.match(/^summary:\s*["']?(.+?)["']?\s*$/im);
          let summary = summaryMatch?.[1]?.trim();
          if (!summary) {
            const body = handoffContent.slice(fmMatch[0].length).trim();
            const firstSubstantive = body.split('\n')
              .map(l => l.trim())
              .find(l => l.length > 0 && !l.startsWith('#'));
            summary = firstSubstantive;
          }
          if (summary) {
            initialMessage = `Resume handoff: ${summary}. Full state at .planning/handoffs/ACTIVE.md.`;
          }
        }
      }
    }
  }
} catch { /* non-fatal — auto-prime is best-effort */ }
```

Notes:
- `normalizeText` is the helper added in Plan 01 Task 3 (`src/shared/text-utils.ts`). Import it at the top of the file: `import { normalizeText } from '../../shared/text-utils.js'`.
- The `STATE.md` regex matches "Current Phase: 4" or "Current Phase: 4.1" (decimal supported). It does NOT pick up "Phase: ..." lines that aren't preceded by "Current ". This intentionally avoids matching plan-level phase entries.
- The `handoffPhase === statePhase` exact-match means handoff `4.1` against STATE `4` is suppressed — this is the team-lead's locked decision. Suppressing the prime in this case is the SAFE failure mode (user types real query, no leakage from a stale handoff).
- The regex for `phase:` in frontmatter strips matching surrounding quotes (`"5"` or `'5'` → `5`).
- Removal: ALL of the existing block at lines 333-374 goes away — the `auto_prime` config check, the priority parsing, the fallback "A handoff is active" generic message. The new contract is binary: contract met → fire; not met → no fire.

```typescript
import { describe, test, expect, afterEach } from 'vitest';
import * as fs from 'node:fs';
import * as path from 'node:path';
import * as os from 'node:os';
import { runSessionStartHook } from '../../adapters/cc-hooks/session-start.js'; // or whatever the entry shape is

describe('SC#4 — One-turn handoff pickup', () => {
  let tmpDir: string;

  afterEach(() => {
    if (tmpDir) fs.rmSync(tmpDir, { recursive: true, force: true });
  });

  function setup(opts: { phase: string; status: string; summary?: string; bodyFirstLine?: string }) {
    tmpDir = fs.mkdtempSync(path.join(os.tmpdir(), 'p5-sc4-'));
    fs.mkdirSync(path.join(tmpDir, '.planning'), { recursive: true });
    fs.writeFileSync(
      path.join(tmpDir, '.planning', 'STATE.md'),
      `# Project State\n\n**Current Phase:** ${opts.phase}\n**Current Phase Name:** Test Phase\n`,
    );
    fs.mkdirSync(path.join(tmpDir, 'context', 'handoffs'), { recursive: true });
    const summaryLine = opts.summary ? `summary: ${opts.summary}\n` : '';
    const body = opts.bodyFirstLine ? `\n${opts.bodyFirstLine}\n` : '';
    fs.writeFileSync(
      path.join(tmpDir, 'context', 'handoffs', 'ACTIVE.md'),
      `---\nstatus: ${opts.status}\nphase: "${opts.phase}"\n${summaryLine}---\n${body}`,
    );
    return tmpDir;
  }

  test('SC#4 — fresh session with active handoff, matching phase, with summary, fires prime', () => {
    const cwd = setup({ phase: '5', status: 'active', summary: 'Resume Phase 5 wave 3' });
    const result = runSessionStartHook({ cwd, type: 'startup' });
    expect(result.hookSpecificOutput?.initialUserMessage).toBe(
      'Resume handoff: Resume Phase 5 wave 3. Full state at .planning/handoffs/ACTIVE.md.'
    );
  });

  test('SC#4 — agent first response begins by addressing handoff topic (semantic check)', () => {
    // This is a contract test for the SHAPE, not the model behavior.
    // The prime begins with "Resume handoff:" — that's the contract that
    // SC#4 (one-turn handoff pickup) measures. The actual agent behavior
    // is verified in the live-fire soak in Plan 09.
    const cwd = setup({ phase: '5', status: 'active', summary: 'Phase 5 Tier C ready' });
    const result = runSessionStartHook({ cwd, type: 'startup' });
    expect(result.hookSpecificOutput?.initialUserMessage).toMatch(/^Resume handoff: /);
    expect(result.hookSpecificOutput?.initialUserMessage).toContain('.planning/handoffs/ACTIVE.md');
  });

  test('SC#4 negative — exploratory glob/grep before user-facing action would not be primed', () => {
    // SC#4 says: "Cold-start session reads ACTIVE.md (Phase 7.5 format), agent first
    //           response addresses handoff topic. No exploratory glob/grep/Bash before
    //           first user-facing action."
    // This is enforced by the prime contract: the model receives the message AS the
    // first user message, so its first response IS bound to the handoff topic.
    // We verify the prime exists as the gate; the no-glob/grep behavior is a model
    // property tested in Plan 09's live-fire soak.

    const result = runSessionStartHook({ cwd, type: 'startup' });

    expect(result.hookSpecificOutput?.initialUserMessage?.length).toBeGreaterThan(0);
  });
});
```
