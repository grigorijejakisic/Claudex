// === LAYER UPS-3: Codebase Index (per-turn, query-gated) ===
// Moved from session-start in Phase 5 / Plan 06.
// Fires only when:
//   1. searchQuery is non-empty
//   2. result fits in remaining per-turn budget
const upsCodebaseQuery = params.searchQuery ?? null;
if (upsCodebaseQuery) {
  try {
    const parts: string[] = [];
    const relevant = findRelevantFiles(params.db, params.project, upsCodebaseQuery, 3);
    if (relevant.length > 0) {
      parts.push('**Relevant files:**');
      for (const f of relevant) {
        const relPath = shortenPath(f.file_path);
        const topSymbols = f.symbols.filter(s => s.exported).slice(0, 5).map(s => `${s.kind} ${s.name}`).join(', ');
        parts.push(`- \`${relPath}\`: ${topSymbols || '(no exports)'}`);
      }
    }
    if (parts.length > 0) {
      const codeSection = `## Codebase Context\n${parts.join('\n')}`;
      const cost = estimateTokens(codeSection);
      // Hard cap: 200 tokens for UPS codebase index (more aggressive than session-start's 800
      // because UPS budget is 1KB total)
      if (cost <= upsBudget && cost <= 200) {
        sections.push(codeSection);
        upsBudget -= cost;
        sources.push('codebase_index');
      }
    }
  } catch { /* non-fatal */ }
}
```

import { describe, test, expect } from 'vitest';
import { assembleRegularPrompt } from '../../assembly/assembler.js';
import { encode } from 'gpt-tokenizer';
// Reuse fixture builders or create UPS-specific ones — UPS payload depends on
// `searchQuery`, `sessionId`, `project`, recent critical reminders, etc.
import { makeUpsFixture } from './assembler-ups-fixtures.js';

const BUDGET_BYTES = 1024;
const BUDGET_TOKENS_APPROX = 256;

const SCENARIOS = [
  { name: 'no-query', query: '' },
  { name: 'short-query', query: 'fix the cache test' },
  { name: 'long-query-with-tech-terms', query: 'why does the assembler skip the materialization layer when isPostCompaction is true and the predicted context fixture is empty' },
  { name: 'sticky-critical-reminder-active', query: 'continue debugging', preset: 'with-critical-reminder' },
];

for (const sc of SCENARIOS) {
  describe(`UPS budget: ${sc.name}`, () => {
    test('per-turn payload ≤ 1024 bytes', () => {
      const fx = makeUpsFixture({ query: sc.query, preset: sc.preset });
      const out = assembleRegularPrompt(fx.params);
      const bytes = Buffer.byteLength(out.content, 'utf8');
      const tokens = encode(out.content).length;
      // Soft warning if tokens-based projection > budget; hard fail on bytes
      if (tokens > BUDGET_TOKENS_APPROX) {
        console.warn(`[ups-budget:${sc.name}] tokens=${tokens} (approx token-budget ${BUDGET_TOKENS_APPROX})`);
      }
      expect(bytes).toBeLessThanOrEqual(BUDGET_BYTES);
    });
  });
}
```
