```ts
export interface DeliberationSurfaceOptions {
  /** Total assembly token budget — surfaced spans cap at token_pct_cap × this */
  totalAssemblyBudgetTokens: number;
  /** When true, the formatter is invoked; when false, returns null with no other side effects (opt-in per assembly site) */
  enabled: boolean;
  /** Optional source-artifact labels per span — keyed by chunk_id; e.g., "Phase 2.1 KILL was decided" */
  artifactLabels?: Record<number, string>;
}

export interface DeliberationSurfaceResult {
  /** Rendered section text including the advisory header + per-span citations; null if no spans / disabled / over budget */
  text: string | null;
  /** Spans actually included after greedy-packing (the rest were dropped because the next span would have overflowed) */
  packed: RoutingSpan[];
  /** True if the bi-encoder-only smaller budget was applied this call */
  bi_encoder_budget_applied: boolean;
}
```

Implementation outline:

1. If `opts.enabled === false`, return `{ text: null, packed: [], bi_encoder_budget_applied: false }`.
2. If `routing.spans.length === 0`, return `{ text: null, packed: [], bi_encoder_budget_applied: false }` — no empty header per truth #5.
3. Resolve budget multiplier from config: `loadConfig().v6.routing` → reads `token_pct_cap` and `bi_encoder_budget_pct`. Compute:
   ```ts
   const cfg = loadConfig().v6.routing;
   const baseBudgetPct = cfg.token_pct_cap / 100; // 0.15
   const biEncoderMultiplier = cfg.bi_encoder_budget_pct / 100; // 0.50
   const effectivePct = routing.bi_encoder_only ? baseBudgetPct * biEncoderMultiplier : baseBudgetPct;
   const budgetTokens = Math.floor(opts.totalAssemblyBudgetTokens * effectivePct);
   ```
4. Greedy-pack spans (already sorted descending by `rank_score` per Plan 10-01 contract):
   - For each span in order, format the citation block (see step 5), estimate tokens via `estimateTokens` from `../shared/text-utils.js`.
   - If running total + this span's tokens ≤ budgetTokens AND this span's body fits whole (no mid-sentence truncation per truth #6) → include.
   - Otherwise → drop silently (the lower-ranked spans likely don't fit either; continue scanning to allow shorter low-rank spans to slot in).
   - Stop scanning after 12 spans (matches `max_k_per_query` from Plan 10-01 — defensive cap if caller bypassed).
5. Format each citation block:
   ```
   From session {sessionId} turn {turnIndex}{labelSuffix}: {body}
   ```
   Where `labelSuffix = artifactLabels[chunk_id] ? ", where ${artifactLabels[chunk_id]}" : ""`. This matches CONTEXT § specifics: `"From session X turn 47, where Phase 2.1 KILL was decided: …"`.
6. Compute the advisory narration line:
   ```
   ## Deliberation Surfaced — {N} spans from {M} sessions
   ```
   Where N = packed.length and M = unique session_ids in packed.
7. If `packed.length === 0` (every span overflowed budget), return `{ text: null, packed: [], bi_encoder_budget_applied }` — no empty header per truth #5.
8. Otherwise compose the final text:
   ```
   {advisory_line}\n\n{citations.join('\n\n')}
   ```
   Return `{ text, packed, bi_encoder_budget_applied: routing.bi_encoder_only }`.

**Non-throwing discipline:**

- No `try/catch` is needed around the pure logic; `loadConfig()` is already non-throwing per the existing `src/shared/config.ts` contract.

```ts
import { routeFromArtifacts, type RoutingArtifact } from '../retrieval/transcript-routing.js';
import { formatDeliberationSurface, type DeliberationSurfaceOptions } from './deliberation-surface.js';

/**
 * v6 Phase 10 — opt-in deliberation surfacing.
 *
 * Calls Plan 10-01's routing surface with the artifact references the assembly
 * already retrieved, then formats the surfaced spans via Plan 10-02's pure
 * formatter. Non-throwing — returns null on any failure (routing unreachable,
 * empty result, disabled).
 *
 * Opt-in per assembly site via opts.enabled (default false in callers — sites
 * that opt in must pass `enabled: true` explicitly).
 *
 * Budget cap (token_pct_cap × totalAssemblyBudgetTokens) is enforced inside
 * formatDeliberationSurface; bi-encoder-only paths get the reduced
 * bi_encoder_budget_pct multiplier per CONTEXT decision 3.
 *
 * See .claude/rules/assembly-budget.md cascade — this section sits at the L2
 * reference-layer position alongside packed-artifact summaries.
 */
export async function formatDeliberationSurfaceSection(
  db: Database,
  artifacts: RoutingArtifact[],
  opts: DeliberationSurfaceOptions & { caller_session_id?: string },
): Promise<string | null> {
  if (!opts.enabled) return null;
  if (!artifacts || artifacts.length === 0) return null;
  try {
    const routing = await routeFromArtifacts(db, artifacts, {
      caller_session_id: opts.caller_session_id,
    });
    const result = formatDeliberationSurface(routing, opts);
    return result.text;
  } catch {
    return null;
  }
}
```

The `Database` type is already imported in sections.ts (see line 27).
