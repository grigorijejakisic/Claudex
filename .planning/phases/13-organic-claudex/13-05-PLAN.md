```typescript
import { getLatestHighlights } from '../intelligence/session-highlights.js';
import type { Database } from 'better-sqlite3';

export type CueType = 
  | 'handoff_read' | 'decision_lock' | 'wait_for_direction'
  | 'script_encounter' | 'error_investigation' | 'package_install';

interface TriggerContext {
  filePath?: string;
  command?: string;
  packageName?: string;
  errorKeyword?: string;
}

// Coverage check functions per 13-CONTEXT.md locked spec.
// Each returns true if highlights COVER this trigger (suppress cue).
const COVERAGE_CHECKS: Record<CueType, (h: ReturnType<typeof getLatestHighlights>[0], ctx: TriggerContext) => boolean> = {
  handoff_read: (h, ctx) => {
    const stem = ctx.filePath ? ctx.filePath.replace(/.*[/\\]/, '').replace(/\.md$/, '') : '';
    return (h.open_questions ?? []).some(q => q.context.includes(stem)) ||
           (h.mental_model ?? '').includes(stem);
  },
  decision_lock: (h, ctx) => {
    const fp = ctx.filePath ?? '';
    return (h.tools_introduced ?? []).some(t => t.path === fp) ||
           (h.decisions_not_made ?? []).some(d => fp && d.gray_area.includes(fp));
  },
  wait_for_direction: (h, _ctx) => {
    // Suppress only if no open questions at all — waiting = under-oriented
    return (h.open_questions ?? []).length === 0;
  },
  script_encounter: (h, ctx) => {
    const fp = ctx.filePath ?? '';
    return (h.tools_introduced ?? []).some(t => t.path === fp || fp.endsWith(t.path));
  },
  error_investigation: (h, ctx) => {
    const kw = (ctx.errorKeyword ?? '').toLowerCase();
    return (h.open_questions ?? []).some(q => q.context.toLowerCase().includes(kw));
  },
  package_install: (h, ctx) => {
    const pkg = ctx.packageName ?? '';
    return (h.tools_introduced ?? []).some(t => t.path.includes(pkg)) ||
           (h.decisions_not_made ?? []).some(d => d.gray_area.includes(pkg));
  },
};

export function shouldFireCue(
  cueType: CueType,
  triggerContext: TriggerContext,
  project: string,
  db: Database,
): boolean {
  try {
    const latestHighlights = getLatestHighlights(db, project, 3);
    if (latestHighlights.length === 0) return true; // no highlights = always fire
    const checkFn = COVERAGE_CHECKS[cueType];
    const isCovered = latestHighlights.some(h => checkFn(h, triggerContext));
    return !isCovered;
  } catch {
    return true; // on error, fire (conservative)
  }
}
```

**`buildScriptEncounterCue`:**

```typescript
// script_encounter — triggers when agent reads a script/module with prior session history
const SCRIPT_PATH_PATTERN = /([/\\](scripts|cli|skills|bin)[/\\]|[/\\]src[/\\].*\.(ts|js|mjs|cjs|py|sh|ps1)$)/;

function isPriorSessionScript(db: Database, filePath: string, project: string): boolean {
  try {
    // Count prior-session reads/edits for this path
    const rows = cachedPrepare(db, `
      SELECT COUNT(DISTINCT session_id) as cnt
      FROM session_events
      WHERE project = ?
        AND (content LIKE ? OR content LIKE ?)
        AND event_type IN ('tool_use', 'file_read', 'file_edit')
    `).get(project, `%${filePath}%`, `%${path.basename(filePath)}%`) as { cnt: number } | undefined;
    return (rows?.cnt ?? 0) >= 3;
  } catch {
    return false;
  }
}

export async function buildScriptEncounterCue(
  db: Database,
  filePath: string,
  project: string,
  sessionId: string,
  config: { 'v6.cues.script_encounter.enabled'?: boolean; 'v6.cues.enabled'?: boolean },
): Promise<string | null> {
  if (config['v6.cues.enabled'] === false) return null;
  if (config['v6.cues.script_encounter.enabled'] === false) return null;
  if (!SCRIPT_PATH_PATTERN.test(filePath)) return null;
  if (!shouldFireCue('script_encounter', { filePath }, project, db)) return null;
  if (!isPriorSessionScript(db, filePath, project)) return null;

  // Avoid re-firing for same file within this session
  const sessionReadKey = `script_encounter:${sessionId}:${filePath}`;
  if (sessionEncounterCache.has(sessionReadKey)) return null;
  sessionEncounterCache.add(sessionReadKey);

  // Surface prior context
  const highlights = getLatestHighlights(db, project, 3);
  const prior = highlights.flatMap(h => h.tools_introduced ?? []).find(t => filePath.endsWith(t.path));
  const priorNote = prior ? ` Prior context: ${prior.purpose}` : '';

  return `[Claudex] This script has prior-session history in this project.${priorNote} Consider checking what you know about it before reading fresh.`;
}

// In-memory session-dedup cache (cleared between sessions)
const sessionEncounterCache = new Set<string>();
```

**`buildErrorInvestigationCue`:**

```typescript
const ERROR_INVESTIGATION_PATTERNS = [
  /\b(cat|tail|less|head)\s+[^|]*\.log\b/i,
  /\b(grep|rg)\s.*\b(error|stack|trace|exception|failed)\b/i,
  /\b(npm\s+test|bun\s+test|vitest|pytest)\b.*\b(--reporter|-v|--verbose)\b/i,
  /\b(journalctl|docker\s+logs)\b/i,
];

function extractErrorKeyword(command: string): string {
  const m = command.match(/\b(error|stack|trace|exception|failed|crash|timeout)\b/i);
  return m?.[1]?.toLowerCase() ?? 'error';
}

export async function buildErrorInvestigationCue(
  db: Database,
  command: string,
  project: string,
  config: { 'v6.cues.error_investigation.enabled'?: boolean; 'v6.cues.enabled'?: boolean },
): Promise<string | null> {
  if (config['v6.cues.enabled'] === false) return null;
  if (config['v6.cues.error_investigation.enabled'] === false) return null;
  if (!ERROR_INVESTIGATION_PATTERNS.some(p => p.test(command))) return null;

  const errorKeyword = extractErrorKeyword(command);
  if (!shouldFireCue('error_investigation', { command, errorKeyword }, project, db)) return null;

  const highlights = getLatestHighlights(db, project, 3);
  const relatedQuestions = highlights.flatMap(h => h.open_questions ?? [])
    .filter(q => q.context.toLowerCase().includes(errorKeyword))
    .slice(0, 2);

  if (relatedQuestions.length === 0) {
    // No specific match, but pattern triggered — generic cue
    return `[Claudex] Similar error/log investigation patterns may exist in prior sessions. Consider: claudex_search("${errorKeyword} error investigation")`;
  }

  const related = relatedQuestions.map(q => `"${q.question}"`).join(', ');
  return `[Claudex] Prior sessions have open questions related to "${errorKeyword}": ${related}. Check claudex_search for context before re-investigating.`;
}
```

**`buildPackageInstallCue`:**

```typescript
const PACKAGE_INSTALL_PATTERN = /^(npm|pnpm|yarn)\s+(install|add|i)\s+|^bun\s+(install|add)\s+|^pip\s+install\s+|^uv\s+(pip\s+)?install\s+|^cargo\s+add\s+|^go\s+get\s+/i;

function extractPackageName(command: string): string {
  const parts = command.trim().split(/\s+/);
  // Package name is typically the last substantive argument
  const nameArgs = parts.filter(p => !p.startsWith('-') && !/^(npm|pnpm|yarn|bun|pip|uv|cargo|go|install|add|get|i)$/.test(p));
  return nameArgs[0] ?? '';
}

export async function buildPackageInstallCue(
  db: Database,
  command: string,
  project: string,
  config: { 'v6.cues.package_install.enabled'?: boolean; 'v6.cues.enabled'?: boolean },
): Promise<string | null> {
  if (config['v6.cues.enabled'] === false) return null;
  if (config['v6.cues.package_install.enabled'] === false) return null;
  if (!PACKAGE_INSTALL_PATTERN.test(command)) return null;

  const packageName = extractPackageName(command);
  if (!packageName) return null;
  if (!shouldFireCue('package_install', { command, packageName }, project, db)) return null;

  const highlights = getLatestHighlights(db, project, 3);
  const rejectedDecision = highlights.flatMap(h => h.decisions_not_made ?? [])
    .find(d => d.gray_area.toLowerCase().includes(packageName.toLowerCase()));

  if (rejectedDecision) {
    return `[Claudex] Package "${packageName}" was previously deferred: "${rejectedDecision.gray_area}" — Reason: ${rejectedDecision.why_deferred}. Consider checking if that reason still applies.`;
  }

  const priorTool = highlights.flatMap(h => h.tools_introduced ?? [])
    .find(t => t.path.includes(packageName));

  if (priorTool) {
    return `[Claudex] Package "${packageName}" has prior context: ${priorTool.purpose}. Check before installing again.`;
  }

  return `[Claudex] Installing "${packageName}" — check claudex_search("${packageName}") for prior evaluation in this project.`;
}
```
