```typescript
/**
 * Phase 2.1 IDX-02 — descriptive spot-check audit (CONTEXT.md decision 3).
 *
 * Audit is descriptive, NOT gating. The 20 sampled pairs per tier are
 * stratified proportionally across the three corpus_origin tiers from
 * Plan 02.1-01 (`v4_backfill` / `phase1_organic_pre_phase2_close` /
 * `phase1_organic_post_phase2_close`). The auditor agent reviews each pair
 * and records valid/invalid judgment + reasoning; the final precision
 * figure (e.g. 17/20) becomes a measured condition reported in RESULTS.md.
 *
 * NO precision gate. The Wilson CI binding from Phase 2 CONTEXT.md item 5
 * is the only gating logic; audit precision is descriptive context.
 */

import type { Database } from 'better-sqlite3';
import type { LabeledPair } from './pair-labeling.js';
import { labelPairsByTier } from './pair-labeling.js';
import { type CorpusOrigin, type LabelerTier } from './types.js';

export interface AuditStratum {
  /** v4_backfill | phase1_organic_pre_phase2_close | phase1_organic_post_phase2_close | mixed */
  origin: CorpusOrigin | 'mixed';
  /** Total pairs in this stratum within the tier. */
  population: number;
  /** Allocated sample size (proportional to population, summing to 20 across single-origin strata). */
  allocation: number;
  /** Sampled LabeledPairs from this stratum. */
  sampled: LabeledPair[];
}

export interface AuditPlan {
  tier: LabelerTier;
  total_pairs: number;
  strata: AuditStratum[];
  sampled_total: number; // sum of allocations across single-origin strata; ≤ 20
  seed: number;
}

/**
 * Pure stratification math: given a tier's full pair list and the desired
 * sample size, compute proportional allocations across single-origin
 * strata + sample deterministically.
 *
 * Proportional allocation:
 *   alloc_O = round(target * (n_pairs_with_origin_O / total_single_origin_pairs))
 * If sum(alloc_O) != target due to rounding, distribute the +/-1 remainders
 * by descending fractional-remainder to land exactly at target.
 *
 * Mixed-origin pairs (origin_a != origin_b) are NOT counted toward the 20
 * stratified slots; they ARE reported in the artifact descriptively
 * (CONTEXT.md item 3a final paragraph + the planner choice in this plan).
 */
export function planAudit(
  pairs: LabeledPair[],
  tier: LabelerTier,
  opts?: { target?: number; seed?: number },
): AuditPlan;

/**
 * Render the markdown skeleton — one entry per sampled pair with raw event
 * content fetched from `episodic_events` via the DB. Auditor fills in
 * per-pair valid/invalid judgment inline.
 */
export function renderAuditMarkdown(
  plan: AuditPlan,
  rawContentByEventId: Map<number, { content: string; project: string; ts_epoch: number }>,
): string;

/**
 * Produce the JSON sidecar: full pair list + raw event content snapshots.
 * Operator/auditor agent reviews this alongside the markdown.
 */
export function renderAuditJson(plan: AuditPlan, rawContentByEventId: Map<number, { content: string; project: string; ts_epoch: number }>): string;

/**
 * End-to-end runner — call from CLI's `audit` subcommand:
 *   const events = loadCorpus(db);
 *   const strict = await runAudit(db, events, 'strict_3frame');
 *   const relaxed = await runAudit(db, events, 'relaxed_2frame');
 *   write strict.markdown to .planning/.../02.1-03-strict-audit.md (skeleton).
 *   write relaxed.markdown to .planning/.../02.1-03-relaxed-audit.md (skeleton).
 *   write JSON sidecars to .planning/.../02.1-03-{strict,relaxed}-audit.json.
 *
 * The agent then opens both markdown files, fills in per-pair judgments,
 * and re-saves; the final precision figure replaces the {{TBD}} placeholder
 * at the top.
 */
export async function runAudit(
  db: Database,
  events: IndexedEvent[],
  tier: LabelerTier,
  opts?: { target?: number; seed?: number; outputDir?: string },
): Promise<{ markdownPath: string; jsonPath: string; plan: AuditPlan }>;
```

**Pair stratum classification rule** (per the must_haves above):
- If `pair.origin_a === pair.origin_b`, the pair's stratum = `pair.origin_a`.
- If `pair.origin_a !== pair.origin_b`, the pair's stratum = `'mixed'`. Mixed-stratum pairs are reported descriptively in the artifact (count + percentage of tier's total) but NOT counted toward the 20 stratified slots.

**Allocation algorithm**:
1. Group pairs by stratum (the four buckets including 'mixed'). Let `totalSingleOrigin = sum(populations of v4_backfill, phase1_organic_pre_phase2_close, phase1_organic_post_phase2_close)`. Mixed pairs are excluded from the stratification denominator.
2. For each single-origin stratum O: `floatAlloc_O = (target * pop_O / totalSingleOrigin)`. Floor to integer + record fractional remainder.
3. If sum(floor) < target: distribute the missing slots (target - sum_floor) by descending fractional remainder.
4. If a stratum has population < its allocation, downsize allocation to population (no upsampling); redistribute the remainder to other strata by descending fractional remainder. (Edge case: if a stratum's population is 0, skip it; redistribute.)
5. Final invariant: sum(allocation across single-origin strata) == min(target, totalSingleOrigin).

**Sampling**: within each stratum, sort pairs by (a, b) for stability, then mulberry32(seed=opts?.seed ?? 4321) shuffle (Fisher–Yates), take the first `allocation_O`.

**Markdown shape**: see Task 3 below for the artifact template.

**Avoid:**
- Sampling without the seeded PRNG — non-determinism poisons re-audit reproducibility.
- Including mixed-stratum pairs in the 20-slot allocation — they're a separate bound-experience condition reported descriptively.
- Adding a precision threshold or pass/fail field to the AuditPlan — it's descriptive, not gating (CONTEXT.md decision 3b).
- Looking up live DB content during the planAudit step — keep planAudit pure (events list in, plan out); the DB content fetch happens in `runAudit` for the markdown/JSON renderers only.
