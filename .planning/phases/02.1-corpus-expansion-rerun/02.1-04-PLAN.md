```typescript
/**
 * Phase 2.1 — milestone-level event-sourced aggregator helpers.
 *
 * The aggregator file at `.planning/aggregates/multi-handle.json` is
 * **strictly append-only** (CONTEXT.md decision 4d binding). Each phase
 * appends exactly the entries it produced. Existing entries are NEVER
 * modified — the bound-experience invariant requires that history
 * remain readable as written.
 *
 * Schema rules:
 *   - Top-level: `{schema_version, question, bound_experiences: BoundExperience[]}`
 *   - Each entry: `{phase, labeler, date, n, verdict, conditions, metrics}`
 *   - `phase` is a string ID (e.g. '2', '2.1-strict', '2.1-relaxed')
 *   - `labeler` is the LabelerTier string (Phase 2's entry uses 'strict_3frame')
 *   - `date` is ISO-8601 (YYYY-MM-DD)
 *   - `verdict` is one of: 'GREEN_LIGHT' | 'SCOPE_DOWN' | 'KILL' | 'BLOCKED'
 *   - `conditions` is an open-shape object: corpus_origin counts,
 *     auto_labeler_precision per stratum (joined from Plan 02.1-03's audit),
 *     pair-set descriptions, sparsity flags
 *   - `metrics` is an open-shape object: deltas + Wilson CI per metric +
 *     intra_project_share + p99 ratios; copied from the per-phase JSON
 *
 * NO `winning`, `current_consensus`, `primary`, `combined` key at the
 * top level. Aggregate IS the array; consumers iterate.
 */

import * as fs from 'node:fs';
import * as path from 'node:path';

export type AggregatorVerdict = 'GREEN_LIGHT' | 'SCOPE_DOWN' | 'KILL' | 'BLOCKED';

export interface BoundExperience {
  phase: string;            // '2', '2.1-strict', '2.1-relaxed', ...
  labeler: string;          // 'strict_3frame', 'relaxed_2frame'
  date: string;             // ISO-8601 date
  n: number;                // held-out test set size; 0 if BLOCKED
  verdict: AggregatorVerdict;
  conditions: Record<string, unknown>;
  metrics: Record<string, unknown>;
}

export interface AggregatorFile {
  schema_version: 1;
  question: string;
  bound_experiences: BoundExperience[];
}

const AGGREGATOR_PATH = path.resolve(
  process.cwd(),
  '.planning',
  'aggregates',
  'multi-handle.json',
);

const QUESTION = 'does multi-handle retrieval (semantic + non-semantic) improve recall over semantic-only at our scale?';

function atomicWrite(filePath: string, contents: string): void {
  const tmp = `${filePath}.tmp.${process.pid}`;
  fs.mkdirSync(path.dirname(filePath), { recursive: true });
  fs.writeFileSync(tmp, contents);
  fs.renameSync(tmp, filePath);
}

/**
 * Load aggregator file. If file does not exist, return a freshly seeded
 * empty aggregator (caller will then seed Phase 2's entry + append 2.1's
 * entries).
 */
export function loadAggregator(filePath: string = AGGREGATOR_PATH): AggregatorFile;

/**
 * Build Phase 2's bound-experience entry from Phase 2's published
 * `02-results.json`. Pure function over JSON parse — no DB, no clock.
 *
 * **Important**: this function PRESERVES Phase 2's metric breakdown shape
 * as published. Phase 2 used the OLD two-tier corpus_origin scheme
 * (phase1_organic / v4_backfill). We do NOT re-classify Phase 2's
 * metrics under the new three-tier scheme — the published bound
 * experience is what was published; future re-runs of Phase 2 (operator
 * choice, deferred per CONTEXT.md `Deferred Ideas`) would generate a new
 * entry, not modify this one.
 */
export function buildPhase2Entry(phase2ResultsJson: unknown): BoundExperience;

/**
 * Build a Phase 2.1 bound-experience entry from a (tier, HarnessRunResult,
 * Verdict, audit precision summary) tuple. Pure function.
 */
export function buildPhase21Entry(
  tier: 'strict_3frame' | 'relaxed_2frame',
  harness: { /* HarnessRunResult shape */ },
  verdict: { kind: AggregatorVerdict; /* Verdict shape */ },
  auditSummary: { tier_total_precision: string; per_stratum: Record<string, string> } | null,
  date: string, // ISO date
): BoundExperience;

/**
 * Append entries idempotently:
 *   - If aggregator file does not exist, create it; seed with Phase 2's entry,
 *     then append the new entries.
 *   - If aggregator file exists, validate against schema, then for each new
 *     entry: if (phase, labeler) tuple already exists, SKIP (idempotency); if
 *     not, push to bound_experiences array.
 *   - Write atomically.
 *
 * Returns the count of newly-appended entries (0..N).
 *
 * **Append-only invariant** (CONTEXT.md decision 4d binding): existing
 * entries are NEVER modified. The function reads the existing array,
 * filters new entries by (phase, labeler) tuple uniqueness, pushes the
 * NEW ones to the tail, and writes. Existing entries are byte-identical
 * before and after.
 */
export function appendBoundExperiences(
  newEntries: BoundExperience[],
  opts?: { phase2ResultsJson?: unknown; filePath?: string },
): number;
```

Implementation notes:
- `loadAggregator` returns `{ schema_version: 1, question: QUESTION, bound_experiences: [] }` when the file doesn't exist (the caller decides whether to seed Phase 2).
- The seeding logic is centralized inside `appendBoundExperiences`: if file is empty AND `opts?.phase2ResultsJson` is provided, seed Phase 2's entry first, then append the new entries.
- The (phase, labeler) tuple uniqueness check is the idempotency mechanism. Re-running the runner produces zero new appends (idempotent) AS LONG AS the prior run's entries were complete.
- Atomic write pattern: write to `.tmp.PID`, rename. The function is safe against partial failures.

- Sorting `bound_experiences` after append — order is chronological-by-append, not phase-numerical. (Phase 2's entry is always first because it's seeded first; 2.1's two entries always come after.)
- Any UPDATE-in-place semantics anywhere — only append.
- Reading the aggregator from disk inside `buildPhase21Entry` or `buildPhase2Entry` — these are pure builders; only `appendBoundExperiences` touches disk.

```typescript
export interface TieredRunSummary {
  verdicts: {
    strict_3frame: Verdict;
    relaxed_2frame: Verdict;
  };
  results_md_path: string;
  results_json_path: string;
  aggregator_entries_appended: number; // 0..3 — 2 (this phase) + 1 if Phase 2 was just seeded

function atomicWrite(filePath: string, contents: string): void {
  const tmp = `${filePath}.tmp.${process.pid}`;
  fs.mkdirSync(path.dirname(filePath), { recursive: true });
  fs.writeFileSync(tmp, contents);
  fs.renameSync(tmp, filePath);

/**
 * Per-tier verdict computation. If the tier's harness sentinel-flagged
 * pairs.total === 0 / held_out_test_n === 0 (Plan 02.1-02 corpus-too-sparse
 * sentinel), produce a BLOCKED verdict from the verdict module's existing
 * blockedVerdict() helper. Otherwise call computeVerdict normally.
 *
 * EITHER way, computeVerdict (or blockedVerdict) is called exactly once
 * per tier — total of two calls per Plan 02.1-04 runner invocation.
 */
function computePerTierVerdict(harness: HarnessRunResult, tier: string, ts_epoch: number): Verdict {
  if (harness.decision_rule_inputs.held_out_test_n === 0) {
    return blockedVerdict(`tier=${tier} pairs.total=0 (corpus-too-sparse sentinel)`, ts_epoch);
  }
  return computeVerdict(harness.decision_rule_inputs, { ts_epoch });
}

export async function runFullPhase21Measurement(
  db: Database,
  opts?: { seed?: number; ts_epoch?: number },
): Promise<TieredRunSummary> {
  const ts = opts?.ts_epoch ?? Math.floor(Date.now() / 1000);

  // --- Step 1: run both tiers ---
  const tiered: TieredHarnessResult = await runHarnessTiered(db, { seed: opts?.seed });

  // --- Step 2: compute two verdicts (exactly two calls into the verdict module) ---
  const strictVerdict = computePerTierVerdict(tiered.strict_3frame, 'strict_3frame', ts);
  const relaxedVerdict = computePerTierVerdict(tiered.relaxed_2frame, 'relaxed_2frame', ts);

  // --- Step 3: persist 02.1-results.json (atomic write, BEFORE aggregator append) ---
  const resultsJsonShape = {
    schema_version: 1 as const,
    generated_at_ts_epoch: ts,
    harness: tiered,
    verdicts: {
      strict_3frame: strictVerdict,
      relaxed_2frame: relaxedVerdict,
    },
  };
  atomicWrite(RESULTS_JSON, JSON.stringify(resultsJsonShape, null, 2));

  // --- Step 4: persist 02.1-RESULTS.md (Plan 02.1-05 owns the renderer; stub here) ---
  // For Plan 02.1-04 scope, write a minimal placeholder. Plan 02.1-05 will
  // implement renderResultsMarkdown(tiered, {strict, relaxed}, auditSummaries).
  const placeholderMd = `# Phase 2.1 Results — Placeholder\n\nThis file will be rendered fully by Plan 02.1-05.\n\nVerdicts: strict_3frame=${strictVerdict.kind}, relaxed_2frame=${relaxedVerdict.kind}\n\nSee 02.1-results.json for machine-readable structured output.\n`;
  atomicWrite(RESULTS_MD, placeholderMd);

  // --- Step 5: aggregator append (LAST write step) ---
  const today = new Date(ts * 1000).toISOString().slice(0, 10); // YYYY-MM-DD

  // Plan 02.1-03 audit precision figures join here in Plan 02.1-05; for
  // 02.1-04 scope, conditions reference audit by path; metrics are copied
  // from the per-tier HarnessRunResult.
  const phase2Json = JSON.parse(fs.readFileSync(PHASE2_RESULTS_JSON_PATH, 'utf8'));

  const newEntries: BoundExperience[] = [
    buildPhase21Entry('strict_3frame', tiered.strict_3frame, strictVerdict, null, today),
    buildPhase21Entry('relaxed_2frame', tiered.relaxed_2frame, relaxedVerdict, null, today),
  ];

  const appendedCount = appendBoundExperiences(newEntries, { phase2ResultsJson: phase2Json });

  return {
    verdicts: { strict_3frame: strictVerdict, relaxed_2frame: relaxedVerdict },
    results_md_path: RESULTS_MD,
    results_json_path: RESULTS_JSON,
    aggregator_entries_appended: appendedCount,
  };
}
```
