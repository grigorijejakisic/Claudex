4. `transcript_injection_acceptance` — L2.5 transcript-span injection landed AND next agent turn references the span (deliberation-surfacing acceptance proxy).

No verdict structure. Signal collection only.

// ── Internal helpers ─────────────────────────────────────────────────────────

function writeTelemetrySignal(
  db: Database,
  eventKind: string,
  detail: Record<string, unknown>,
): void {
  try {
    // INSERT the signal row — adapt SQL to match the real telemetry table schema
    // discovered in Task 1 (may use 'event_type'/'kind'/etc. instead of 'event_kind').
    const stmt = cachedPrepare(
      db,
      `INSERT INTO telemetry (event_kind, detail, created_at_epoch_ms)
       VALUES (?, ?, ?)`,
    );
    stmt.run(eventKind, JSON.stringify(detail), Date.now());

    // Cap rows per signal kind to avoid DB bloat
    enforceRowCap(db, eventKind);
  } catch (err) {
    // Signal write errors MUST NOT propagate to the hook pipeline
    if (process.env.CLAUDEX_DEBUG === '1') {
      console.error(`[telemetry-signals] write failed for ${eventKind}:`, err);
    }
  }
}

function enforceRowCap(db: Database, eventKind: string): void {
  try {
    // Delete oldest rows beyond the cap — idempotent, runs after every INSERT
    const countStmt = cachedPrepare(
      db,
      `SELECT COUNT(*) as cnt FROM telemetry WHERE event_kind = ?`,
    );
    const { cnt } = countStmt.get(eventKind) as { cnt: number };
    if (cnt > MAX_ROWS_PER_SIGNAL_KIND) {
      const deleteStmt = cachedPrepare(
        db,
        `DELETE FROM telemetry WHERE id IN (
           SELECT id FROM telemetry WHERE event_kind = ?
           ORDER BY created_at_epoch_ms ASC
           LIMIT ?
         )`,
      );
      deleteStmt.run(eventKind, cnt - MAX_ROWS_PER_SIGNAL_KIND);
    }
  } catch {
    // Row-cap failure is non-fatal — signal was already written
  }
}
