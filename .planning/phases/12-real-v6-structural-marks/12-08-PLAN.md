import type { Database } from 'better-sqlite3';
import { cachedPrepare } from './stmt-cache.js';
import { getPolicy } from '../intelligence/policy-registry.js';

// ── Suppression check ────────────────────────────────────────────────────────

export function areCuesEnabled(cueType: 'handoff_read' | 'decision_lock' | 'wait_for_direction'): boolean {
  const globalEnabled = getPolicy('v6.cues.enabled') ?? true;
  if (!globalEnabled) return false;
  const perTypeEnabled = getPolicy(`v6.cues.${cueType}.enabled`) ?? true;
  if (!perTypeEnabled) return false;
  // Phase-tag check: only fire when active phase has architecture/workflow/methodology tag
  return activePhaseHasCueTriggerTag();
}

function activePhaseHasCueTriggerTag(): boolean {
  // Read active phase CONTEXT.md for trigger tags
  // If we can't read the file or no tags found, suppress to avoid noise
  try {
    const { readFileSync, existsSync } = require('node:fs');
    const { join } = require('node:path');
    // Look for active phase context in the planning directory
    const planningPath = join(process.cwd(), '.planning', 'phases');
    if (!existsSync(planningPath)) return false;

    const { readdirSync } = require('node:fs');
    const phaseDirs = readdirSync(planningPath).filter((d: string) => {
      const stat = require('node:fs').statSync(join(planningPath, d));
      return stat.isDirectory();
    });

    // Find the most recently modified phase directory with a CONTEXT.md
    for (const dir of phaseDirs.slice(-3)) { // check last 3 phases
      const contextPath = join(planningPath, dir, `${dir.split('-')[0]}-CONTEXT.md`);
      const contextPaths = [
        contextPath,
        join(planningPath, dir, 'CONTEXT.md'),
        ...readdirSync(join(planningPath, dir))
          .filter((f: string) => f.endsWith('-CONTEXT.md'))
          .map((f: string) => join(planningPath, dir, f)),
      ];

      for (const cp of contextPaths) {
        if (existsSync(cp)) {
          const content = readFileSync(cp, 'utf8');
          if (/\b(architecture|workflow|methodology)\b/i.test(content)) return true;
        }
      }
    }
    return false;
  } catch {
    return false; // fail-safe: suppress cue on read error
  }
}

// ── Cue format helper ─────────────────────────────────────────────────────────

const MAX_CUE_SIZE_CHARS = 1024;
const MAX_SNIPPET_CHARS = 200;
const MAX_RESULTS = 3;

function formatCueResult(kind: string, id: string | number, title: string, snippet: string): string {
  const trimmedSnippet = snippet.length > MAX_SNIPPET_CHARS
    ? snippet.slice(0, MAX_SNIPPET_CHARS) + '…'
    : snippet;
  return `[${kind}:${id}] ${title} — ${trimmedSnippet}`;
}

function buildCueBlock(header: string, lines: string[]): string | null {
  if (lines.length === 0) return null;
  const body = lines.slice(0, MAX_RESULTS).join('\n');
  const full = `<system-reminder>\n${header}\n\n${body}\n</system-reminder>`;
  return full.length > MAX_CUE_SIZE_CHARS
    ? `<system-reminder>\n${header}\n\n${body.slice(0, MAX_CUE_SIZE_CHARS - header.length - 80)}\n…\n</system-reminder>`
    : full;
}

// ── Cue 1: Handoff-reading ────────────────────────────────────────────────────

/**
 * Build cue for when the agent is about to read a handoff file.
 * Tier 1: claudex_search by handoff topic/filename slug.
 * Tier 2: latest 5 project sessions.
 */
export async function buildHandoffReadCue(
  db: Database,
  handoffPath: string,
  sessionId: string,
): Promise<string | null> {
  const slug = extractSlugFromPath(handoffPath);
  const lines: string[] = [];

  // Tier 1: search by handoff slug
  try {
    const stmt = cachedPrepare(
      db,
      `SELECT id, kind, summary FROM artifacts
       WHERE summary LIKE ? OR summary LIKE ?
       ORDER BY importance_score DESC LIMIT ?`,
    );
    const rows = stmt.all(`%${slug}%`, `%handoff%`, MAX_RESULTS) as Array<{ id: number; kind: string; summary: string }>;
    for (const row of rows) {
      lines.push(formatCueResult(row.kind, row.id, row.summary.slice(0, 50), row.summary));
    }
  } catch { /* non-blocking */ }

  if (lines.length === 0) {
    // Tier 2: latest session events
    try {
      const stmt = cachedPrepare(
        db,
        `SELECT id, summary FROM artifacts
         WHERE kind IN ('session_event', 'observation')
         ORDER BY created_at_epoch_ms DESC LIMIT 5`,
      );
      const rows = stmt.all() as Array<{ id: number; summary: string }>;
      for (const row of rows) {
        lines.push(formatCueResult('session', row.id, 'Recent session', row.summary));
      }
    } catch { /* non-blocking */ }
  }

  return buildCueBlock(
    '## Context Pull Cue — Handoff Reading\nBefore interpreting this handoff, prior session context that may be relevant:',
    lines,
  );
}

// ── Cue 2: Decision-locking ──────────────────────────────────────────────────

/**
 * Build cue for when the agent is about to write a config or curated-context file.
 * Tier 1: search by file path keywords.
 * Tier 2: broader search by file basename.
 */
export async function buildDecisionLockCue(
  db: Database,
  filePath: string,
  sessionId: string,
): Promise<string | null> {
  const basename = filePath.split(/[\\/]/).pop()?.replace(/\.\w+$/, '') ?? '';
  const lines: string[] = [];

  // Tier 1: search by file path + diff keywords
  try {
    const stmt = cachedPrepare(
      db,
      `SELECT id, kind, summary FROM artifacts
       WHERE summary LIKE ? OR summary LIKE ?
         AND (summary LIKE '%UNVALIDATED%' OR summary LIKE '%fabricated%' OR summary LIKE '%unverified%')
       ORDER BY importance_score DESC LIMIT ?`,
    );
    const rows = stmt.all(`%${basename}%`, `%config%`, MAX_RESULTS) as Array<{ id: number; kind: string; summary: string }>;
    for (const row of rows) {
      lines.push(formatCueResult(row.kind, row.id, row.summary.slice(0, 50), row.summary));
    }
  } catch { /* non-blocking */ }

  if (lines.length === 0) {
    // Tier 2: broader search
    try {
      const stmt = cachedPrepare(
        db,
        `SELECT id, kind, summary FROM artifacts
         WHERE kind IN ('decision', 'learning')
           AND summary LIKE ?
         ORDER BY importance_score DESC LIMIT ?`,
      );
      const rows = stmt.all(`%${basename}%`, MAX_RESULTS) as Array<{ id: number; kind: string; summary: string }>;
      for (const row of rows) {
        lines.push(formatCueResult(row.kind, row.id, row.summary.slice(0, 50), row.summary));
      }
    } catch { /* non-blocking */ }
  }

  const header = '## Context Pull Cue — Decision Locking\nDid any prior session establish, contradict, or flag this value as UNVALIDATED?';
  return buildCueBlock(header, lines);
}

// ── Cue 3: Wait-for-direction ─────────────────────────────────────────────────

// Regex patterns from 12-CONTEXT.md operator-locked answer Q [12-08/Q1]
const WAIT_FOR_DIRECTION_PATTERNS = [
  /\bthe handoff is (a |the )?menu\b/i,
  /\bnot (a )?directive\b/i,
  /\bwaiting for direction\b/i,
  /\bnothing autonomous(ly)?\b/i,
  /\byour call\b.*pick the lane/i,
  /\b(neither|none).{0,30}without (your|operator) (input|direction)\b/i,
];

export function detectsWaitForDirection(assistantResponse: string): boolean {
  return WAIT_FOR_DIRECTION_PATTERNS.some((p) => p.test(assistantResponse));
}

/**
 * Build cue for when the agent drifts toward "wait for direction" stance.
 * Tier 1: parse ACTIVE.md for unresolved-task language.
 * Tier 2: latest session events for current investigation thread.
 */
export async function buildWaitForDirectionCue(
  db: Database,
  sessionId: string,
): Promise<string | null> {
  const lines: string[] = [];

  // Tier 1: check ACTIVE handoff for unresolved tasks
  try {
    const { existsSync, readFileSync } = require('node:fs');
    const { join } = require('node:path');
    const activePath = join(process.cwd(), 'context', 'handoffs', 'ACTIVE.md');
    if (existsSync(activePath)) {
      const content = readFileSync(activePath, 'utf8');
      // Extract what-next section
      const whatNextMatch = content.match(/## What.*Next\n([\s\S]{0,500})/i);
      if (whatNextMatch) {
        lines.push(formatCueResult('handoff', 'ACTIVE', 'Unresolved task in active handoff', whatNextMatch[1].slice(0, 200)));
      }
    }
  } catch { /* non-blocking */ }

  if (lines.length === 0) {
    // Tier 2: latest session context
    try {
      const stmt = cachedPrepare(
        db,
        `SELECT id, summary FROM artifacts
         WHERE kind = 'observation'
         ORDER BY created_at_epoch_ms DESC LIMIT 3`,
      );
      const rows = stmt.all() as Array<{ id: number; summary: string }>;
      for (const row of rows) {
        lines.push(formatCueResult('observation', row.id, 'Recent context', row.summary));
      }
    } catch { /* non-blocking */ }
  }

  const header = '## Context Pull Cue — Wait-For-Direction Detected\nThere may be unresolved tasks in the active investigation. Review before deferring:';
  return buildCueBlock(header, lines);
}

  return (filePath.split(/[\\/]/).pop() ?? '')
    .replace(/\.(md|yaml|yml|json|ts|js)$/, '')
    .replace(/[-_]/g, ' ');
}
