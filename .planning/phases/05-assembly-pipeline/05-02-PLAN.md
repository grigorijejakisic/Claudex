**`assembleFullContext(params: FullAssemblyParams): InjectPayload`**

Implements priority-budgeted cascade with three-tier degradation. Synchronous (all dependencies are synchronous DB reads and file reads).

```
Tier 1 (full assembly):
  try {
    budget = config.injection.budget_tokens (default 4000)
    sections: string[] = []
    sources: string[] = []
    skipped: Array<{ priority: number; formatter: () => string | null }> = []
    referenceMode = false

    // Priority 1: Identity
    identity = formatIdentitySection(params.identityDir)
    if (identity && estimateTokens(identity) <= budget) {
      sections.push(identity)
      budget -= estimateTokens(identity)
      sources.push('identity')
    }

    // Priority 2: Project context
    project = formatProjectSection(params.projectDir)
    if (project && estimateTokens(project) <= budget) {
      sections.push(project)
      budget -= estimateTokens(project)
      sources.push('project')
    }

    // Priority 3: Checkpoint
    checkpoint = loadCheckpoint(params.db, params.projectDir) // RESUME preset applied inside
    checkpointSection = formatCheckpointSection(checkpoint)
    if (checkpointSection && estimateTokens(checkpointSection) <= budget) {
      sections.push(checkpointSection)
      budget -= estimateTokens(checkpointSection)
      sources.push('checkpoint')
    }

    // Priority 4: Learnings (top 10)
    learnings = getTopLearnings(params.db, params.project, 10)
    learningsSection = formatLearningsSection(learnings)
    if (learningsSection && estimateTokens(learningsSection) <= budget) {
      sections.push(learningsSection)
      budget -= estimateTokens(learningsSection)
      sources.push('learnings')
    }

    // Priority 5: HOT files
    hotFiles = getHotFiles(params.db, params.project, 20)
    hotSection = formatHotFilesSection(hotFiles)
    if (hotSection && estimateTokens(hotSection) <= budget) {
      sections.push(hotSection)
      budget -= estimateTokens(hotSection)
      sources.push('hot_files')
    }

    // Check reference mode trigger
    if (budget < 500) referenceMode = true

    // Priority 6: GSD
    gsd = readGsdState(params.projectDir)
    gsdSection = formatGsdSection(gsd)
    if (gsdSection) {
      if (estimateTokens(gsdSection) <= budget) {
        sections.push(gsdSection)
        budget -= estimateTokens(gsdSection)
        sources.push('gsd')
      } else {
        skipped.push({ priority: 6, formatter: () => gsdSection })
      }
    }

    // Priority 7: FTS5 search
    query = params.searchQuery ?? checkpoint?.thread?.topic ?? null
    if (query && config.features.fts5_search) {
      fts5Results = searchObservations(params.db, query, params.project, { limit: 10 })
      fts5Section = formatFts5Section(fts5Results, referenceMode)
      if (fts5Section) {
        if (estimateTokens(fts5Section) <= budget) {
          sections.push(fts5Section)
          budget -= estimateTokens(fts5Section)
          sources.push('fts5')
        } else {
          skipped.push({ priority: 7, formatter: () => fts5Section })
        }
      }
    }

    // Priority 8: Recent high-quality observations
    recentObs = getObservationsByProject(params.db, params.project, { limit: 20 })
      .filter(o => o.importance >= 3)
      .filter(o => (Date.now() / 1000 - o.timestamp_epoch) < 86400) // last 24h
    recentSection = formatRecentSection(recentObs)
    if (recentSection) {
      if (estimateTokens(recentSection) <= budget) {
        sections.push(recentSection)
        budget -= estimateTokens(recentSection)
        sources.push('recent')
      } else {
        skipped.push({ priority: 8, formatter: () => recentSection })
      }
    }

    // Assemble content
    content = sections.join('\n\n')

    // Post-redaction reclaim (ASMB-05)
    preRedactionLength = content.length
    content = redactContent(content)
    postRedactionLength = content.length

    if (postRedactionLength < preRedactionLength && skipped.length > 0) {
      freedTokens = estimateTokens(content.slice(0, preRedactionLength - postRedactionLength))
      // Note: actually we reclaimed (preRedactionLength - postRedactionLength)/4 tokens
      reclaimBudget = budget + Math.floor((preRedactionLength - postRedactionLength) / 4)

      // Re-attempt skipped sections in priority order
      skipped.sort((a, b) => a.priority - b.priority)
      for (const { formatter } of skipped) {
        section = formatter()
        if (section && estimateTokens(section) <= reclaimBudget) {
          content += '\n\n' + redactContent(section)
          reclaimBudget -= estimateTokens(section)
          // Add to sources based on priority
        }
      }
    }

    return {
      content,
      tokenEstimate: estimateTokens(content),
      sources,
    }
  }

Tier 2 (checkpoint-only):
  catch {
    try {
      checkpoint = loadFromFile(params.projectDir)
      if (checkpoint) {
        checkpointMd = renderCheckpointMarkdown(checkpoint, 'RESUME')
        identity = formatIdentitySection(params.identityDir)
        content = [identity, checkpointMd].filter(Boolean).join('\n\n')
        return { content, tokenEstimate: estimateTokens(content), sources: ['identity', 'checkpoint'].filter corresponding }
      }
    }

Tier 3 (identity-only):
    catch {
      try {
        identity = formatIdentitySection(params.identityDir)
        if (identity) {
          return { content: identity, tokenEstimate: estimateTokens(identity), sources: ['identity'] }
        }
      }
      catch {}
    }
  }

  // Final fallback: empty
  return { content: '', tokenEstimate: 0, sources: [] }
```

**`assembleRegularPrompt(params: RegularPromptParams): InjectPayload`**

Implements Architecture Section 7.3: Regular Prompt Assembly.

```
Non-throwing wrapper:
try {
  // 1. Post-compaction -> full assembly
  if (params.isPostCompaction) {
    return assembleFullContext({
      db: params.db,
      project: params.project,
      projectDir: params.projectDir,
      config: params.config,
      searchQuery: params.prompt,
      identityDir: params.identityDir,
    })
  }

  // 2. Topic-shift -> micro-injection
  if (params.topicShift?.shifted) {
    pivot = assembleTopicPivot({
      shift: params.topicShift,
      db: params.db,
      project: params.project,
      config: params.config,
    })
    if (pivot.tokenEstimate > 0 && pivot.tokenEstimate <= config.injection.topic_shift_budget) {
      return pivot
    }
  }

  // 3. Gauge injection at >= threshold
  gaugeSection = formatGaugeSection(params.gauge, params.config.injection.gauge_threshold)
  if (gaugeSection) {
    return {
      content: gaugeSection,
      tokenEstimate: estimateTokens(gaugeSection),
      sources: ['gauge'],
    }
  }

  // 4. Zero injection (most turns)
  return { content: '', tokenEstimate: 0, sources: [] }
} catch {
  return { content: '', tokenEstimate: 0, sources: [] }
}
```

**`assembleTopicPivot(params: TopicPivotParams): InjectPayload`**

Implements Architecture Section 7.3.1: Topic pivot injection (max 800 tokens).

```
Non-throwing wrapper:
try {
  shift = params.shift
  budget = params.config.injection.topic_shift_budget (default 800)

  // Fetch relevant data for new topic
  relevantLearnings: LearningRow[] = []
  relevantHotFiles: PressureRow[] = []
  relevantDecisions: Array<{ content: string; source: string }> = []

  if (shift.newTopic) {
    // FTS5 search for relevant learnings (query learnings content)
    try {
      allLearnings = getTopLearnings(params.db, params.project, 10)
      relevantLearnings = allLearnings.filter(l =>
        l.content.toLowerCase().includes(shift.newTopic!.toLowerCase().split(' ')[0])
      ).slice(0, 3)
    } catch {}

    // HOT files (all, not filtered by topic — but limited)
    try {
      relevantHotFiles = getHotFiles(params.db, params.project, 5)
    } catch {}

    // Recent decisions for the session (if any match new topic keyword)
    // Note: decisions are session-scoped, not topic-filtered in DB
    // Just include top 3 most recent decisions as context
    try {
      // Use searchObservations with newTopic to find related decisions stored as observations
      // Decisions are in the decisions table, not observations — skip for now
      // The checkpoint already includes decisions; this is supplementary
    } catch {}
  }

  pivotSection = formatTopicPivotSection({
    shift,
    learnings: relevantLearnings,
    hotFiles: relevantHotFiles,
  })

  if (!pivotSection) {
    return { content: '', tokenEstimate: 0, sources: [] }
  }

  // Apply redaction
  content = redactContent(pivotSection)
  tokenEst = estimateTokens(content)

  // Enforce budget cap
  if (tokenEst > budget) {
    // Truncate: take only the transition marker (first line block)
    lines = content.split('\n')
    truncated = lines.slice(0, 3).join('\n') // ## Context Pivot + marker line + blank
    return {
      content: truncated,
      tokenEstimate: estimateTokens(truncated),
      sources: ['topic_pivot'],
    }
  }

  return {
    content,
    tokenEstimate: tokenEst,
    sources: ['topic_pivot'],
  }
} catch {
  return { content: '', tokenEstimate: 0, sources: [] }
}
```
