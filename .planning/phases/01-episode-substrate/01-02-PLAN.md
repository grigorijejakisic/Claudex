   ```typescript
   export function dualWriteAssistantMessage(
     db: Database.Database,
     sessionId: string,
     project: string,
     assistantText: string,
   ): { episodicId: number | null; updatedLegacy: boolean } {
     // 1. Inside a db.transaction:
     //    a. Try updateConversationTurnAssistant equivalent (UPDATE conversation_turns SET assistant_text WHERE session_id=? AND assistant_text IS NULL ORDER BY turn_number DESC LIMIT 1)
     //       — capture the resulting turn_number. Fallback: INSERT a fresh conversation_turns row with user_text=NULL.
     //    b. INSERT one episodic_events row: type='assistant_message', source='cc-hooks/stop', provenance='organic', content=assistantText, content_hash=sha256(assistantText), turn_number=<from step a>.
     // 2. On throw: emitTelemetry('episodic_write_failure', { hook: 'stop', ... }).
   }
   ```

   Implementation rules:
   - The transaction MUST be created via `db.transaction(() => { ... })()` (better-sqlite3 closure form). On throw, the closure auto-rollbacks.
   - Telemetry write happens OUTSIDE the transaction (so the telemetry row commits even when the dual-write fails).
   - Telemetry detail JSON shape: `{ hook: 'user-prompt-submit'|'stop', attempted_rows: number, organic_id: number|null, error_message: string, error_stack_first_5_lines: string }`. This decision is the planner's discretion per CONTEXT.md.
   - Legacy `conversation_turns` write goes FIRST inside the transaction. The episodic writes go SECOND. Both succeed or both rollback.
   - For the user prompt: `turn_number` is computed from `MAX(turn_number) + 1` for that session — same logic as today's `storeConversationTurnUserText`. Compute INSIDE the transaction so concurrent hooks don't race.
   - For dual-write of injected blocks: the SQL must use `RETURNING id` on the organic INSERT to get the parent id (better-sqlite3 supports `RETURNING` since SQLite 3.35).
   - `content` for injected rows is the wrapper's inner text (`block.content`). `source` is `'wrapper:' + block.tag` so Phase 3's retrieval can filter by wrapper type.
   - `metadata_json` for injected rows = `JSON.stringify({ tag: block.tag, attributes: block.attributes ?? null })`.
   - Use `cachedPrepare` for every INSERT (consistent with the codebase).
