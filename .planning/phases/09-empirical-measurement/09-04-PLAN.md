import * as fs from 'node:fs';
import * as path from 'node:path';
import type { Database } from 'better-sqlite3';
// Re-use plan 08-02 building blocks. Verify exact export names at coding time.
import { chunkTranscript } from './chunk-transcript.js';
import { upsertChunk } from './upsert-chunk.js';

export interface IngestReport {
  files_seen: number;
  chunks_inserted: number;
  chunks_updated: number; // upsert path
  errors: Array<{ file: string; error: string }>;
}

const SYNTHETIC_PROBES_DIR = path.resolve(
  process.cwd(),
  '.planning',
  'phases',
  '09-empirical-measurement',
  'probes',
  'synthetic-transcripts',
);

const SYNTHETIC_PROJECT_ID = 'p9-synthetic';

/**
 * Ingests synthetic deliberation transcripts authored in plan 09-01 into
 * transcript_chunk_v6 + vec_transcript_chunks_v6.
 *
 * Each *.jsonl file uses its filename basename as the deterministic
 * session_id (e.g. drift-c-05.jsonl → session_id 'synthetic-drift-c-05').
 * The probe fixtures' transcript_anchor.session_id values must match this
 * pattern exactly so B-arm retrieval can locate the anchors.
 *
 * Provenance is set to 'environmental' — the closest existing enum value
 * for non-organic, non-tool-result, non-injected content. No schema
 * extension required.
 */
export async function importSyntheticTranscripts(
  db: Database,
  dir: string = SYNTHETIC_PROBES_DIR,
): Promise<IngestReport> {
  const report: IngestReport = { files_seen: 0, chunks_inserted: 0, chunks_updated: 0, errors: [] };
  if (!fs.existsSync(dir)) {
    return report;
  }
  const files = fs.readdirSync(dir).filter(f => f.endsWith('.jsonl'));
  for (const f of files) {
    report.files_seen++;
    const fullPath = path.join(dir, f);
    const sessionId = `synthetic-${path.basename(f, '.jsonl')}`;
    try {
      const lines = fs.readFileSync(fullPath, 'utf-8').split('\n').filter(l => l.trim().length > 0);
      // Build a synthetic transcript object matching whatever shape chunkTranscript expects.
      // Inspect src/ingestion/chunk-transcript.ts for the exact input contract — likely
      // an array of {role, content, turn_index} or a JSONL-of-cc-events shape. Adapt to it.
      const turns = lines.map((line, idx) => {
        const parsed = JSON.parse(line) as { type?: string; message?: { role?: string; content?: string | unknown } };
        const role = parsed.message?.role
          ?? (parsed.type === 'user' ? 'user' : parsed.type === 'assistant' ? 'assistant' : 'system');
        const contentRaw = parsed.message?.content;
        const content = typeof contentRaw === 'string'
          ? contentRaw
          : JSON.stringify(contentRaw ?? '');
        return { role, content, turn_index: idx };
      });
      const chunks = chunkTranscript(turns, {
        sessionId,
        projectId: SYNTHETIC_PROJECT_ID,
        // Other chunkTranscript options as needed; default redaction discipline applies (P8 plan 08-02).
      });
      for (const chunk of chunks) {
        const wasUpdate = upsertChunk(db, {
          ...chunk,
          provenance: 'environmental',
          // Tag with synthetic origin for downstream auditability — the harness
          // doesn't filter on this, but tooling that reads the corpus can.
        });
        if (wasUpdate) report.chunks_updated++;
        else report.chunks_inserted++;
      }
    } catch (err) {
      report.errors.push({ file: f, error: err instanceof Error ? err.message : String(err) });
    }
  }
  return report;
}
```

#!/usr/bin/env bun
/**
 * Operator-invoked CLI to ingest the 4 synthetic deliberation transcripts
 * authored in plan 09-01 into transcript_chunk_v6. Run BEFORE
 * `bun run benchmark:deliberation-surfacing`.
 *
 * Idempotent: re-runs are a no-op for already-ingested content because
 * upsertChunk's UNIQUE(session_id, turn_index, role, sub_index) absorbs duplicates.
 *
 * Usage: bun run import:synthetic-probes
 */
import { openDatabase, closeDatabase } from '../core/storage.js';
import { getDbPath } from '../shared/paths.js';
import { importSyntheticTranscripts } from '../ingestion/synthetic-corpus-import.js';

async function main(): Promise<number> {
  const db = openDatabase(getDbPath());
  try {
    const report = await importSyntheticTranscripts(db);
    process.stdout.write(JSON.stringify(report, null, 2) + '\n');
    return report.errors.length === 0 ? 0 : 1;
  } finally {
    closeDatabase(db);
  }
}

if (import.meta.main || require.main === module) {
  main().then(c => process.exit(c)).catch(e => {
    process.stderr.write(String(e) + '\n');
    process.exit(1);
  });
}
```
