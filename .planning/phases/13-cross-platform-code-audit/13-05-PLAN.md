# Windows-specific scripts — preserve CRLF (cmd / batch / PowerShell expect it)
*.bat text eol=crlf
*.cmd text eol=crlf
*.ps1 text eol=crlf

# Binary types — preserved as-is.

*.jpg binary
*.jpeg binary
*.gif binary
*.ico binary

*.sqlite binary
