**Abstraction signature (locked):**
```typescript
export async function terminateProcess(
  pid: number,
  options?: { force?: boolean }
): Promise<void>
```

**Behavior (locked):**
- Windows (`process.platform === 'win32'`):
  - `force: true` → `execSync(\`taskkill /PID ${pid} /F\`)` (or async equivalent)
  - `force: false` or absent → `execSync(\`taskkill /PID ${pid}\`)`
- Unix (`process.platform !== 'win32'`):
  - `force: true` → `process.kill(pid, 'SIGKILL')`
  - `force: false` or absent → `process.kill(pid, 'SIGTERM')`

**Error handling:**
- If the process is already dead, `taskkill` exits non-zero (Windows) and `process.kill` throws ESRCH (Unix). Treat both as success — the goal of `terminateProcess` is "ensure this PID is no longer running"; if it's already gone, that's fine. Catch and swallow these specific cases. Re-throw other errors.
- On Windows, `taskkill` non-zero exit codes other than "process not found" should propagate. Detection: stderr containing "could not be found" / "not found" or exit code 128 → swallow; else re-throw.

**Async vs sync:**
The function is `async` and returns `Promise<void>`. Internally:
- On Unix, `process.kill` is synchronous but cheap — wrap in a resolved promise.
- On Windows, `taskkill` is a child process spawn. Use `execFile` (promisified via `util.promisify`) to keep it async without blocking the event loop. `execSync` is acceptable too if existing callsites are synchronous and the abstraction is called in a context where blocking is OK — but prefer async to keep the abstraction future-proof.

**Test mocking strategy:**
- For Windows path: mock `child_process.execFile` (or `execSync`) and verify it was called with `taskkill`, the correct PID, and the correct `/F` flag presence
- For Unix path: mock `process.kill` and verify it was called with the correct signal (`SIGKILL` vs `SIGTERM`) and PID
- Toggle platform via `Object.defineProperty(process, 'platform', { value: 'win32' })` (or `'linux'`) inside the test, restoring the original after — vitest's `beforeEach`/`afterEach` patterns work fine
- Cover the "process already dead" cases for both platforms — verify no error is thrown when the underlying call signals "no such process"

**Migration of callsites:**

1. Import: `import { terminateProcess } from '../shared/process-control'` (path adjusted to caller's depth)

4. Remove the now-unused `child_process` import if it was only used for the taskkill call

**Out of scope here:**
- Soft termination signal escalation (SIGTERM → wait → SIGKILL) — current Windows behavior is `taskkill` once and trust it; Unix mirrors that. If a callsite needed escalation behavior previously, preserve it; do not add new escalation logic.

```typescript
import { execFile } from 'node:child_process';
import { promisify } from 'node:util';

const execFileAsync = promisify(execFile);

/**
 * Terminate a process cross-platform.
 *
 * Windows: taskkill /PID <pid> [/F]
 * Unix: process.kill(pid, SIGTERM | SIGKILL)
 *
 * Treats "process already dead" as success — the goal is "ensure this PID is no
 * longer running"; if it's already gone, that's fine.
 */
export async function terminateProcess(
  pid: number,
  options?: { force?: boolean }
): Promise<void> {
  const force = options?.force === true;

  if (process.platform === 'win32') {
    const args = ['/PID', String(pid)];
    if (force) args.push('/F');
    try {
      await execFileAsync('taskkill', args);
    } catch (err: unknown) {
      // taskkill exits non-zero when the process is already gone; that is success for our purposes.
      const e = err as { stderr?: string; code?: number | string; message?: string };
      const stderr = (e.stderr ?? '').toLowerCase();
      const message = (e.message ?? '').toLowerCase();
      if (
        stderr.includes('could not be found') ||
        stderr.includes('not found') ||
        message.includes('not found') ||
        e.code === 128
      ) {
        return;
      }
      throw err;
    }
  } else {
    try {
      process.kill(pid, force ? 'SIGKILL' : 'SIGTERM');
    } catch (err: unknown) {
      // ESRCH = no such process; already gone, treat as success.
      const e = err as { code?: string };
      if (e.code === 'ESRCH') return;
      throw err;
    }
  }
}
```

```typescript
import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest';

// Mock child_process before importing the module under test
vi.mock('node:child_process', () => ({
  execFile: vi.fn((cmd, args, cb) => cb(null, { stdout: '', stderr: '' })),
}));

import * as childProcess from 'node:child_process';
import { terminateProcess } from './process-control';

describe('terminateProcess', () => {
  let originalPlatform: PropertyDescriptor | undefined;
  let originalKill: typeof process.kill;

  beforeEach(() => {
    originalPlatform = Object.getOwnPropertyDescriptor(process, 'platform');
    originalKill = process.kill;
    vi.clearAllMocks();
  });

  afterEach(() => {
    if (originalPlatform) Object.defineProperty(process, 'platform', originalPlatform);
    process.kill = originalKill;
  });

  function setPlatform(p: NodeJS.Platform) {
    Object.defineProperty(process, 'platform', { value: p, configurable: true });
  }

  describe('Windows', () => {
    beforeEach(() => setPlatform('win32'));

    it('uses taskkill /F when force=true', async () => {
      await terminateProcess(1234, { force: true });
      expect(childProcess.execFile).toHaveBeenCalledWith(
        'taskkill',
        ['/PID', '1234', '/F'],
        expect.any(Function),
      );
    });

    it('uses taskkill without /F when force is absent', async () => {
      await terminateProcess(1234);
      expect(childProcess.execFile).toHaveBeenCalledWith(
        'taskkill',
        ['/PID', '1234'],
        expect.any(Function),
      );
    });

    it('swallows "process not found" errors', async () => {
      vi.mocked(childProcess.execFile).mockImplementationOnce((_cmd, _args, cb: any) => {
        cb({ stderr: 'ERROR: The process "1234" not found.' }, { stdout: '', stderr: '' });
      });
      await expect(terminateProcess(1234, { force: true })).resolves.toBeUndefined();
    });

    it('rethrows unrelated errors', async () => {
      vi.mocked(childProcess.execFile).mockImplementationOnce((_cmd, _args, cb: any) => {
        cb({ stderr: 'ERROR: Access denied.', code: 1 }, { stdout: '', stderr: '' });
      });
      await expect(terminateProcess(1234, { force: true })).rejects.toBeDefined();
    });
  });

  describe('Unix', () => {
    beforeEach(() => setPlatform('linux'));

    it('sends SIGKILL when force=true', async () => {
      const killSpy = vi.fn();
      process.kill = killSpy as any;
      await terminateProcess(1234, { force: true });
      expect(killSpy).toHaveBeenCalledWith(1234, 'SIGKILL');
    });

    it('sends SIGTERM when force is absent', async () => {
      const killSpy = vi.fn();
      process.kill = killSpy as any;
      await terminateProcess(1234);
      expect(killSpy).toHaveBeenCalledWith(1234, 'SIGTERM');
    });

    it('swallows ESRCH errors', async () => {
      const killSpy = vi.fn(() => {
        const err: NodeJS.ErrnoException = new Error('No such process');
        err.code = 'ESRCH';
        throw err;
      });
      process.kill = killSpy as any;
      await expect(terminateProcess(1234, { force: true })).resolves.toBeUndefined();
    });

    it('rethrows non-ESRCH errors', async () => {
      const killSpy = vi.fn(() => {
        const err: NodeJS.ErrnoException = new Error('EPERM: not permitted');
        err.code = 'EPERM';
        throw err;
      });
      process.kill = killSpy as any;
      await expect(terminateProcess(1234, { force: true })).rejects.toBeDefined();
    });
  });
});
```
