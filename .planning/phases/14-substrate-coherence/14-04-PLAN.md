```typescript
/**
 * Phase 14 Plan 14-04 — P2.7 Project Knowledge surface.
 *
 * Surfaces top-K substantive same-project artifacts at session-start,
 * routed through hybrid retrieval with ACTIVE.md `summary` as the
 * implicit query. Closes the same-project starvation gap that
 * Experience Tier (cross-project only) and Recent Session Frames
 * (highlight-dependent) leave open.
 *
 * Returns null when ACTIVE.md missing OR summary empty OR no
 * substantive artifacts retrieved. Cache-stable: no clock data.
 */

import type { Database } from 'better-sqlite3';
import * as fs from 'fs';
import * as path from 'path';
import { parseHandoffHeader } from '../angel/handoff-writer.js';
import { hybridRetrieve } from '../core/hybrid-retrieval.js';  // existing API
import { getHandoffsDir } from '../shared/paths.js';
import { estimateTokens, normalizeText } from '../shared/text-utils.js';

export const P27_TOP_K = 3;
export const P27_TITLE_MAX_CHARS = 120;
export const P27_EXCERPT_LINES = 3;

export async function formatProjectKnowledgeSection(
  db: Database,
  project: string,
  projectDir: string,
  budgetTokens: number,
): Promise<string | null> {
  // 1. Read ACTIVE.md primary handoff (single file, NOT glob)
  let summary: string | null = null;
  try {
    const handoffPath = path.join(getHandoffsDir(projectDir), 'ACTIVE.md');
    if (!fs.existsSync(handoffPath)) return null;
    const raw = normalizeText(fs.readFileSync(handoffPath, 'utf-8'));
    const header = parseHandoffHeader(raw);
    if (!header) return null;
    summary = (header.summary ?? '').trim();
    if (!summary) return null;
  } catch {
    return null;
  }

  // 2. Hybrid retrieval against project artifacts, substantive-only filter
  let rows;
  try {
    rows = await hybridRetrieve(db, {
      query: summary,
      project,
      topK: P27_TOP_K,
      substantiveOnly: true,  // new flag — see Task 3
    });
  } catch {
    return null;
  }
  if (!rows || rows.length === 0) return null;

  // 3. Format. Drop lowest-ranked entries until under budget. If 1 entry
  //    over cap, truncate excerpt. If still empty, return null.
  let entries = rows.slice();
  while (entries.length > 0) {
    const rendered = render(entries);
    if (estimateTokens(rendered) <= budgetTokens) return rendered;
    entries.pop();  // drop lowest-ranked
  }
  return null;
}

function render(entries: ArtifactRow[]): string {
  const lines: string[] = ['## Project Knowledge'];
  for (const r of entries) {
    const title = (r.summary ?? r.title ?? 'Untitled').slice(0, P27_TITLE_MAX_CHARS);
    const ref = r.artifact_ref ?? `${r.artifact_type ?? r.kind}#${r.id}`;
    const excerpt = (r.content ?? r.body ?? '').split('\n').slice(0, P27_EXCERPT_LINES).join('\n');
    lines.push(`### ${title}`);
    lines.push(`*Source: ${ref}*`);
    if (excerpt.trim()) lines.push(excerpt);
    lines.push('');
  }
  return lines.join('\n').trimEnd();
}
```

`ArtifactRow` shape comes from the hybrid retriever's existing return type — cover both legacy `artifacts` shape (artifact_type, summary, content, artifact_ref) and V17 `artifact` shape (kind, title, body) since the retriever may return either.
