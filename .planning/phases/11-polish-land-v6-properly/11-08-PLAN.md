export async function runQ3(db: Database.Database, config: Q3Config): Promise<Q3Result | Q3SkippedResult> {
  // Hard preconditions
  if (!existsSync(config.q1VerdictPath) || !existsSync(config.q2VerdictPath)) {
    return skipQ3(`Q1 or Q2 verdict file missing — Plan 11-06/11-07 have not produced verdicts yet`, null, null, config);
  }
  const q1 = JSON.parse(readFileSync(config.q1VerdictPath, 'utf8'));
  const q2 = JSON.parse(readFileSync(config.q2VerdictPath, 'utf8'));
  if (q1.verdict !== 'BIND_POSITIVE' || q2.verdict !== 'BIND_POSITIVE') {
    return skipQ3(
      `Q3 only runs when Q1 + Q2 both BIND_POSITIVE per CONTEXT § Implementation Decisions § W3 (Q2). Q1=${q1.verdict}, Q2=${q2.verdict}.`,
      q1.verdict, q2.verdict, config,
    );
  }

  // Pre-flight reranker (against the cross-corpus DB)
  if (!(await preflightRerankerHealth())) {
    return inconclusiveQ3('reranker preflight failed', config, q1.verdict, q2.verdict);
  }

  // Open the cross-corpus DB
  if (!existsSync(config.bigMozzyDbPath)) {
    return skipQ3(`big-mozzy-v2 DB not found at ${config.bigMozzyDbPath}`, q1.verdict, q2.verdict, config);
  }
  const bmDb = new Database(config.bigMozzyDbPath, { readonly: false }); // routing writes telemetry; OK on a non-readonly handle

  // Load 30 cross-corpus probes; assert disjointness from P9 + Q2
  const probes: Probe[] = JSON.parse(readFileSync(config.q3ProbesFixturePath, 'utf8'));
  if (probes.length !== 30) throw new Error(`Q3 expects exactly 30 cross-corpus probes; got ${probes.length}`);

  // Run a single replication (per CONTEXT § Operational constraints — big-mozzy-v2 is stretch-goal; one replication is the budget-feasible design)
  const rep = await runReplication(bmDb, probes, config.seed, 1);

  // Fallback rate + per-judge error checks (same as Q1/Q2)
  const fallbackPct = rep.reranker_calls > 0 ? (rep.bi_encoder_fallbacks / rep.reranker_calls) * 100 : 0;
  if (fallbackPct > config.fallbackRateThresholdPct) {
    return inconclusiveQ3(`Fallback rate ${fallbackPct.toFixed(1)}% > ${config.fallbackRateThresholdPct}%`, config, q1.verdict, q2.verdict, { rep, fallbackPct });
  }
  const errorsByJudge = aggregateJudgeErrors([rep]);
  const fallbackDecision = computeRunFallback({ errorsByJudge, totalProbes: 30, thresholdPct: 10 });
  if (fallbackDecision.inconclusive) {
    return inconclusiveQ3(`> 1 judge exceeded 10% error rate`, config, q1.verdict, q2.verdict, { rep });
  }

  // Wilson on the difference (single replication n=30)
  const aPass = rep.per_probe_outcomes.filter(o => o.a_arm_pass).length;
  const bPass = rep.per_probe_outcomes.filter(o => o.b_arm_pass).length;
  const { lower, upper, point } = newcombeWilsonDelta(bPass, 30, aPass, 30);
  let verdict: Q3Result['verdict'];
  if (lower > 0) verdict = 'BIND_POSITIVE';
  else if (upper < 0) verdict = 'BIND_NEGATIVE';
  else verdict = 'INCONCLUSIVE';

  bmDb.close();

  const result: Q3Result = {
    verdict,
    wilson_lower_bound: lower,
    wilson_upper_bound: upper,
    delta_estimate: point,
    fallback_rate_pct: fallbackPct,
    per_judge_errors_pct: Object.fromEntries(Object.entries(errorsByJudge).map(([k, v]) => [k, (v / 30) * 100])),
    dropped_judge: fallbackDecision.dropped_judge,
    per_probe_outcomes: rep.per_probe_outcomes,
    q3_started_at: /* fn entry */, q3_completed_at: new Date().toISOString(),
    corpus: 'big-mozzy-v2',
  };
  writeFileSync(join(config.outDir, 'q3-verdict.json'), JSON.stringify(result, null, 2));
  appendAggregateRow({ phase: '11-polish', sub_question: 'Q3', corpus: 'big-mozzy-v2', /* per-rep summary */ });
  return result;
}

function skipQ3(reason: string, q1: string | null, q2: string | null, config: Q3Config): Q3SkippedResult {

  writeFileSync(join(config.outDir, 'q3-skipped.json'), JSON.stringify(skip, null, 2));
  return skip;
