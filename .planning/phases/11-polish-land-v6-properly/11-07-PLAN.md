export async function runQ2(db: Database.Database, config: Q2Config): Promise<Q2Result | Q2SkippedResult> {
  // (i) Hard precondition: Q1 BIND_POSITIVE
  if (!existsSync(config.q1VerdictPath)) {
    const skip: Q2SkippedResult = {
      skipped: true,
      skip_reason: 'q1-verdict.json missing — Plan 11-06 has not produced a Q1 verdict yet',
      q1_verdict: null,
      recorded_at: new Date().toISOString(),
    };
    writeFileSync(join(config.outDir, 'q2-skipped.json'), JSON.stringify(skip, null, 2));
    return skip;
  }
  const q1 = JSON.parse(readFileSync(config.q1VerdictPath, 'utf8'));
  if (q1.verdict !== 'BIND_POSITIVE') {
    const skip: Q2SkippedResult = {
      skipped: true,
      skip_reason: `Q1 verdict is ${q1.verdict}; Q2 only runs on BIND_POSITIVE per CONTEXT § Implementation Decisions § W3 (Q2). Plan 11-08 will activate substrate-only + KILL-receipt path.`,
      q1_verdict: q1.verdict,
      recorded_at: new Date().toISOString(),
    };
    writeFileSync(join(config.outDir, 'q2-skipped.json'), JSON.stringify(skip, null, 2));
    return skip;
  }

  // (ii) Pre-flight reranker health (same as Q1)
  if (!(await preflightRerankerHealth())) return inconclusiveQ2('reranker preflight failed', config);

  // (iii) Load 60 probes; assert disjointness from P9 locked set
  const probes: Probe[] = JSON.parse(readFileSync(config.q2ProbesFixturePath, 'utf8'));
  if (probes.length !== 60) {
    throw new Error(`Q2 expects exactly 60 fresh probes; got ${probes.length}`);
  }
  // Disjointness sanity check — Plan 11-07 Task 2's authoring script also checks this, but defense-in-depth
  const p9Probes = JSON.parse(readFileSync('<P9_FIXTURE_PATH>', 'utf8'));
  const p9Ids = new Set(p9Probes.map((p: any) => p.id));
  for (const p of probes) {
    if (p9Ids.has(p.id)) throw new Error(`Q2 probe ${p.id} overlaps P9 locked set — disjointness violated`);
  }

  // (iv) Run r1 (probes 0..29) + r2 (probes 30..59); these are FRESH probes per replication, not the same probes
  // measured twice. Pooling at the end is legitimate — n=60 independent samples.
  const probesR1 = probes.slice(0, 30);
  const probesR2 = probes.slice(30, 60);
  const r1 = await runReplication(db, probesR1, config.r1Seed, 1);
  const r2 = await runReplication(db, probesR2, config.r2Seed, 2);

  // (v) Aggregate fallback rate (same threshold as Q1)
  const totalCalls = r1.reranker_calls + r2.reranker_calls;
  const totalFallbacks = r1.bi_encoder_fallbacks + r2.bi_encoder_fallbacks;
  const fallbackPct = totalCalls > 0 ? (totalFallbacks / totalCalls) * 100 : 0;
  if (fallbackPct > config.fallbackRateThresholdPct) {
    return inconclusiveQ2(`Fallback rate ${fallbackPct.toFixed(1)}% > ${config.fallbackRateThresholdPct}% threshold`, config, { r1, r2 });
  }

  // (vi) Per-judge error check
  const errorsByJudge = aggregateJudgeErrors([r1, r2]);
  const totalProbes = 60;
  const fallbackDecision = computeRunFallback({ errorsByJudge, totalProbes, thresholdPct: 10 });
  if (fallbackDecision.inconclusive) {
    return inconclusiveQ2(`> 1 judge exceeded 10% error rate`, config, { r1, r2 });
  }

  // (vii) Wilson CI on pooled n=60 (legitimate pool — 60 independent probes)
  const allOutcomes = [...r1.per_probe_outcomes, ...r2.per_probe_outcomes];
  const aArmPasses = allOutcomes.filter(o => o.a_arm_pass).length;
  const bArmPasses = allOutcomes.filter(o => o.b_arm_pass).length;
  // Δ = (b-arm pass rate) − (a-arm pass rate); 95% CI via Newcombe-Wilson on the difference of two proportions
  const { lower, upper, point } = newcombeWilsonDelta(bArmPasses, 60, aArmPasses, 60); // exists in wilson.ts; if not, write it
  let verdict: Q2Result['verdict'];
  if (lower > 0) verdict = 'BIND_POSITIVE';
  else if (upper < 0) verdict = 'BIND_NEGATIVE';
  else verdict = 'INCONCLUSIVE';

  const result: Q2Result = {
    verdict,
    wilson_lower_bound: lower,
    wilson_upper_bound: upper,
    delta_estimate: point,
    fallback_rate_pct: fallbackPct,
    per_judge_errors_pct: Object.fromEntries(
      Object.entries(errorsByJudge).map(([k, v]) => [k, (v / totalProbes) * 100]),

    dropped_judge: fallbackDecision.dropped_judge,
    r1, r2,
    q2_started_at: /* captured at fn entry */,
    q2_completed_at: new Date().toISOString(),
  };

  // Persist q2-verdict.json + aggregator append
  writeFileSync(join(config.outDir, 'q2-verdict.json'), JSON.stringify(result, null, 2));
  appendAggregateRow({ phase: '11-polish', sub_question: 'Q2', /* per-rep + pooled summary */ });
  return result;
}

// scripts/build-q2-probe-pool.cjs
// Build a 60-probe disjoint pool template AND validate disjointness from P9.

const { readFileSync, writeFileSync, existsSync } = require('fs');

const Q1_VERDICT_PATH = '.planning/phases/11-polish-land-v6-properly/q1-verdict.json';
const P9_FIXTURE_PATH = process.env.P9_FIXTURE_PATH || '.planning/aggregates/p9-locked-probes.json';
const OUT_PATH = '.planning/phases/11-polish-land-v6-properly/q2-locked-probes.json';

if (!existsSync(Q1_VERDICT_PATH)) {
  console.log('[q2-author] q1-verdict.json missing — skip');
  process.exit(0);
}
const q1 = JSON.parse(readFileSync(Q1_VERDICT_PATH, 'utf8'));
if (q1.verdict !== 'BIND_POSITIVE') {
  console.log(`[q2-author] Q1 verdict is ${q1.verdict}; Q2 short-circuits — no probes authored.`);
  process.exit(0);
}

const argMode = process.argv[2]; // 'init' | 'validate'
if (argMode === 'init') {
  // Emit a template file with 60 placeholder probes
  const probes = [];
  for (let r = 1; r <= 2; r++) {
    for (let i = 1; i <= 30; i++) {
      const id = `q2-r${r}-${String(i).padStart(2, '0')}`;
      probes.push({
        id,
        kind: '__FILL__', // a/b/c/d/e drift-detection categories per ENG-02
        anchor_session_id: '__FILL__', // an actual session_id from claudex-v3 corpus
        anchor_created_at_epoch_ms: 0, // turn timestamp
        query_text: '__FILL_THE_PROBE_QUERY__',
        ground_truth_a_arm_should: '__FAIL__', // summaries should fail this drift case
        ground_truth_b_arm_should: '__PASS__', // transcripts should pass
        rubric_notes: '__authoring rules: project-internal anchor; no parametric-knowledge confound__',
      });
    }
  }
  writeFileSync(OUT_PATH, JSON.stringify(probes, null, 2));
  console.log(`[q2-author] Wrote 60-probe template to ${OUT_PATH}. Fill __FILL__ markers manually per locked authoring rules.`);
  process.exit(0);
}

if (argMode === 'validate') {
  // Validate disjointness + authoring-rule compliance
  if (!existsSync(OUT_PATH)) { console.error(`[q2-validate] ${OUT_PATH} missing`); process.exit(1); }
  const q2 = JSON.parse(readFileSync(OUT_PATH, 'utf8'));
  const errors = [];

  // Rule: exactly 60 probes
  if (q2.length !== 60) errors.push(`Expected 60 probes, got ${q2.length}`);

  // Rule: no __FILL__ markers remain
  for (const p of q2) {
    const json = JSON.stringify(p);
    if (json.includes('__FILL__') || json.includes('__FAIL__') || json.includes('__PASS__')) {
      errors.push(`Probe ${p.id} still has __FILL__ markers`);
    }
  }

  // Rule: ID format q2-r1-NN / q2-r2-NN
  for (const p of q2) {
    if (!/^q2-r[12]-\d{2}$/.test(p.id)) errors.push(`Probe ${p.id} has wrong ID format`);
  }

  // Rule: disjoint from P9
  if (existsSync(P9_FIXTURE_PATH)) {
    const p9 = JSON.parse(readFileSync(P9_FIXTURE_PATH, 'utf8'));
    const p9Ids = new Set(p9.map((p) => p.id));
    const p9Anchors = new Set(p9.map((p) => `${p.anchor_session_id}::${p.anchor_created_at_epoch_ms}`));
    const p9Queries = new Set(p9.map((p) => (p.query_text || '').trim()));
    for (const p of q2) {
      if (p9Ids.has(p.id)) errors.push(`Probe ${p.id} ID overlaps P9`);
      if (p9Anchors.has(`${p.anchor_session_id}::${p.anchor_created_at_epoch_ms}`)) errors.push(`Probe ${p.id} anchor overlaps P9`);
      if (p9Queries.has((p.query_text || '').trim())) errors.push(`Probe ${p.id} query_text overlaps P9`);
    }
  } else {
    errors.push(`Cannot validate disjointness — ${P9_FIXTURE_PATH} not found`);
  }

  // Rule: anchor session within claudex-v3 (no big-mozzy-v2 sessions — that's Q3)
  for (const p of q2) {
    if (typeof p.anchor_session_id !== 'string' || !p.anchor_session_id) errors.push(`Probe ${p.id} missing anchor_session_id`);
    // Heuristic: big-mozzy-v2 sessions have a known prefix or registry entry. If a registry exists, check; if not,
    // skip this rule and rely on manual review.
  }

  // Heuristic — flag probes likely tainted by parametric-knowledge (mirrors the audit script from Plan 11-04)
  const PARAMETRIC_MARKERS = [
    /\b(react|vue|angular|express|django|flask|spring)\b/i,
    /\b(http|https|tcp|udp|json|yaml|xml)\b/i,
    /\b(github|stackoverflow|npm|pypi)\b/i,
    /\b(big-o|sorting|hashing)\b/i,
  ];
  for (const p of q2) {
    const text = JSON.stringify(p).toLowerCase();
    const score = PARAMETRIC_MARKERS.reduce((s, r) => s + (r.test(text) ? 1 : 0), 0);
    if (score >= 2) errors.push(`Probe ${p.id} has ${score} parametric-marker hits — possible parametric-knowledge confound; review`);
  }

  if (errors.length > 0) {
    console.error(`[q2-validate] ${errors.length} validation errors:`);
    for (const e of errors) console.error(`  - ${e}`);
    process.exit(1);
  }
  console.log(`[q2-validate] OK — 60 disjoint probes, no parametric-marker flags, all authoring rules satisfied.`);
}
