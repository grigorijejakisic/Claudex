interface ReplicationResult {
  seed: number;
  per_probe_outcomes: Array<{ probe_id: string; a_arm_pass: boolean; b_arm_pass: boolean; ensemble_error: boolean; }>;
  reranker_calls: number;
  bi_encoder_fallbacks: number;

export async function runQ1(db: Database.Database, config: Q1Config): Promise<Q1Result> {
  // (i) Pre-flight reranker health check
  const rerankerOK = await preflightRerankerHealth();
  if (!rerankerOK) {
    return inconclusive(`Reranker pre-flight failed — service at port 7439 unreachable; cannot run Q1 without confirmed reranker access.`);
  }

  // (ii) Load 30 probes, byte-immutable
  const probes: Probe[] = JSON.parse(readFileSync(config.probesFixturePath, 'utf8'));
  if (probes.length !== 30) {
    throw new Error(`Q1 expects exactly 30 locked probes; got ${probes.length}. Fixture is byte-immutable per CONTEXT § Implementation Decisions § W2 (Q3).`);
  }

  // (iii) Run r1 + r2 sequentially (network calls; parallelism would saturate Ollama)
  const r1 = await runReplication(db, probes, config.r1Seed, 1);
  const r2 = await runReplication(db, probes, config.r2Seed, 2);

  // (iv) Aggregate fallback rate
  const totalCalls = r1.reranker_calls + r2.reranker_calls;
  const totalFallbacks = r1.bi_encoder_fallbacks + r2.bi_encoder_fallbacks;
  const fallbackPct = totalCalls > 0 ? (totalFallbacks / totalCalls) * 100 : 0;
  if (fallbackPct > config.fallbackRateThresholdPct) {
    return inconclusive(
      `Reranker fallback rate ${fallbackPct.toFixed(1)}% exceeded ${config.fallbackRateThresholdPct}% threshold (CONTEXT § Methodology critique #5). Measurement conflates retrieval-quality with engagement-quality.`,
      { r1, r2 },
    );
  }

  // (v) Per-judge error rate check + run fallback (Plan 11-04 Task 3 interface)
  const errorsByJudge = aggregateJudgeErrors([r1, r2]); // sum errors across replications per judge
  const totalProbes = probes.length * 2; // 30 × 2 replications
  const fallbackDecision = computeRunFallback({ errorsByJudge, totalProbes, thresholdPct: 10 });
  if (fallbackDecision.inconclusive) {
    return inconclusive(`> 1 judge exceeded 10% error rate; ensemble integrity compromised (CONTEXT § Methodology critique #6).`, { r1, r2 });
  }
  // If fallbackDecision.dropped_judge is set, the per-probe ensemble already used the fallback (Plan 11-04 Task 3
  // implements this at the per-probe level via the dropped_judge option). For Q1 we surface the dropped judge
  // as a side-fact; it doesn't prevent verdict computation.

  // (vi) Build PerProbeOutcome[] for paired-McNemar
  const r1ByProbe = new Map(r1.per_probe_outcomes.map(o => [o.probe_id, o]));
  const r2ByProbe = new Map(r2.per_probe_outcomes.map(o => [o.probe_id, o]));
  const paired: PerProbeOutcome[] = probes.map(p => ({
    probe_id: p.id,
    r1_a_arm_pass: r1ByProbe.get(p.id)?.a_arm_pass ?? false,
    r1_b_arm_pass: r1ByProbe.get(p.id)?.b_arm_pass ?? false,
    r2_a_arm_pass: r2ByProbe.get(p.id)?.a_arm_pass ?? false,
    r2_b_arm_pass: r2ByProbe.get(p.id)?.b_arm_pass ?? false,
  }));

  // (vii) Compute paired-McNemar (Plan 11-04 Task 2 interface)
  const mcnemar = pairedMcNemar(paired); // default min_discordant_threshold = 5

  return {
    verdict: mcnemar.verdict,
    paired_mcnemar: mcnemar,
    fallback_rate_pct: fallbackPct,
    per_judge_errors_pct: Object.fromEntries(
      Object.entries(errorsByJudge).map(([k, v]) => [k, (v / totalProbes) * 100]),
    ),
    dropped_judge: fallbackDecision.dropped_judge,
    r1, r2,
    q1_started_at: /* captured at fn entry */,
    q1_completed_at: new Date().toISOString(),
    angel_glm_handling: 'unmitigated', // operator overrides at runtime; default unmitigated
  };
}

async function preflightRerankerHealth(): Promise<boolean> {
  // Send a synthetic POST to http://127.0.0.1:7439/rerank with a minimal payload.
  // Return true on 200; false on any other outcome.
  try {
    const response = await fetch('http://127.0.0.1:7439/rerank', {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ query: 'preflight', candidates: ['preflight'], top_k: 1 }),
      signal: AbortSignal.timeout(5_000),

    return response.status === 200;
  } catch { return false; }
}
