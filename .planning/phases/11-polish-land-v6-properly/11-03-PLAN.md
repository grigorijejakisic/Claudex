**(c) `src/ingestion/transcript-chunker-v6.ts` — Findings #5 + #6:**

**Fix #5 — format-preserving sub-chunker:**

Locate the sentence-split logic. Plan 8 likely shape:

```ts
const sentences = body.split(/(?<=[.!?])\s+/);
```

The bug: lookbehind regex consumes the whitespace between sentences, and string-split discards format-bearing whitespace within code blocks (e.g., a body containing ```const x = 1.5;\nfoo();``` splits at `1.` because the lookbehind doesn't know about backtick fences).

**Fix — offset-tracking sub-chunker:**

```ts
function sentenceOffsets(body: string): Array<{ start: number; end: number }> {
  // Scan once, track byte offsets of sentence boundaries while respecting backtick-fenced code blocks.
  const offsets: Array<{ start: number; end: number }> = [];
  let inFence = false;
  let fenceChar = '';
  let sentenceStart = 0;
  for (let i = 0; i < body.length; i++) {
    const ch = body[i];
    // Detect ``` or ` fence open/close
    if (ch === '`') {
      if (!inFence) {
        // Look ahead for triple-backtick fence
        if (body.slice(i, i + 3) === '```') {
          inFence = true;
          fenceChar = '```';
          i += 2;
          continue;
        } else {
          inFence = true;
          fenceChar = '`';
          continue;
        }
      } else if (inFence && (fenceChar === '`' || (fenceChar === '```' && body.slice(i, i + 3) === '```'))) {
        if (fenceChar === '```') i += 2;
        inFence = false;
        fenceChar = '';
        continue;
      }
    }
    if (inFence) continue;
    // Sentence boundary: . ! ? followed by whitespace AND not at the very end
    if (/[.!?]/.test(ch) && i + 1 < body.length && /\s/.test(body[i + 1])) {
      // Walk past trailing whitespace to the start of the next sentence
      let j = i + 1;
      while (j < body.length && /\s/.test(body[j])) j++;
      offsets.push({ start: sentenceStart, end: j });
      sentenceStart = j;
    }
  }
  if (sentenceStart < body.length) {
    offsets.push({ start: sentenceStart, end: body.length });
  }
  return offsets;
}

// Then use body.slice(off.start, off.end) for each sub-chunk — preserves whitespace, capitalization,
// and intra-sentence punctuation byte-for-byte.
const subChunks = sentenceOffsets(body).map((off) => body.slice(off.start, off.end));
```

This is more code than the one-line regex but it (a) doesn't split inside fenced code blocks and (b) preserves the body exactly when reassembled. Round-trip-test below.

**Fix #6 — hard-cap force-split for over-long sentences:**

After the sub-chunker produces sub-chunks, run a second pass that force-splits any sub-chunk exceeding the embedder's token limit (snowflake-arctic-embed2 has a 512-token context — use the existing constant if one exists, else hardcode 512):

```ts
const EMBEDDER_TOKEN_LIMIT = 512; // snowflake-arctic-embed2; reuse a shared constant if one exists
const safeChunks: string[] = [];
for (const sub of subChunks) {
  const tokens = countTokens(sub);
  if (tokens <= EMBEDDER_TOKEN_LIMIT) {
    safeChunks.push(sub);
    continue;
  }
  // Force-split on a character boundary that approximates the token limit. length/4 ≈ tokens for English.
  const charsPerChunk = Math.floor(EMBEDDER_TOKEN_LIMIT * 4 * 0.95); // 5% safety margin
  for (let pos = 0; pos < sub.length; pos += charsPerChunk) {
    safeChunks.push(sub.slice(pos, pos + charsPerChunk));
  }
  // Telemetry: force-split is operator-visible per Gemini Ingestion #6.

    `INSERT INTO telemetry (event_kind, ts_epoch_ms, session_id, detail)
     VALUES ('transcript_chunk_force_split', ?, ?, ?)`,
  ).run(Date.now(), sessionId, JSON.stringify({ original_tokens: tokens, split_count: Math.ceil(sub.length / charsPerChunk) }));
}
return safeChunks;
```

// scripts/lint-test-discipline.cjs
// Flag expect(...).not.toThrow() patterns on missing-dependency tests.
// Visible, not perfect — surfaces sites that regress to the v5.0.1 silent-fail pattern.

const { readFileSync, readdirSync, statSync } = require('fs');
const { join } = require('path');

const TEST_GLOB_DIRS = ['src/tests'];
const PATTERN_FLAGS = [
  // expect(...).not.toThrow()  — most direct bug-codifying pattern
  { regex: /expect\s*\(.*?\)\s*\.\s*not\s*\.\s*toThrow\s*\(/g, label: 'expect(...).not.toThrow()' },
  // expect(result.errors).toBe(0) inside a test that mentions 'missing' or 'not exist' or 'unavailable' or 'unreachable'
  // — heuristic, not exact, but catches the common shape Gemini Ingestion #4 surfaced.
];

function walk(dir, out = []) {
  for (const entry of readdirSync(dir)) {
    const p = join(dir, entry);
    const st = statSync(p);
    if (st.isDirectory()) walk(p, out);
    else if (entry.endsWith('.test.ts')) out.push(p);
  }
  return out;
}

const flagged = [];
for (const dir of TEST_GLOB_DIRS) {
  for (const file of walk(dir)) {
    const src = readFileSync(file, 'utf8');
    for (const { regex, label } of PATTERN_FLAGS) {
      let m;
      while ((m = regex.exec(src)) !== null) {
        const lineNo = src.slice(0, m.index).split('\n').length;
        flagged.push({ file, line: lineNo, label });
      }
    }
  }
}

if (flagged.length > 0) {
  console.error(`\n[lint-test-discipline] ${flagged.length} site(s) flagged — expect(...).not.toThrow() on missing-dependency tests is the v5.0.1 silent-fail pattern. Replace with an explicit-sentinel assertion.\n`);
  for (const f of flagged) {
    console.error(`  ${f.file}:${f.line}  ${f.label}`);
  }
  process.exit(1);
}
console.log('[lint-test-discipline] OK — no flagged patterns.');

#!/usr/bin/env node
// scripts/build-production-shape-snapshot.cjs
// Produce a sanitized snapshot of the live ~/.claudex/db/claudex.db at .planning/fixtures/production-shape-v32.db.
// One-shot — re-run only when production schema or shape changes materially.
//
// Sanitization: PII/secret redaction pass.
// (a) Drop or replace user/assistant message bodies in transcript_chunk_v6 with a deterministic placeholder
//     so the structural shape (counts, joins, indexes) is preserved while content is not.
// (b) Strip all `learnings.body`, `decisions.body`, `experience_patterns.body` to placeholder text.
// (c) Drop the `cross_agent_sessions` table content entirely — it can carry session content from other projects.
// (d) Preserve schema (PRAGMA user_version, all CREATE TABLE / INDEX / VIEW / TRIGGER, vec0 virtual tables).
// (e) Preserve row counts and primary-key + foreign-key relationships — these are the production-shape signal.

const { existsSync, copyFileSync, mkdirSync } = require('fs');
const { join, dirname } = require('path');
const Database = require('better-sqlite3');
const os = require('os');

const SOURCE = join(os.homedir(), '.claudex', 'db', 'claudex.db');
const DEST = join(__dirname, '..', '.planning', 'fixtures', 'production-shape-v32.db');

if (!existsSync(SOURCE)) {
  console.error(`[snapshot] Source not found: ${SOURCE}`);
  console.error('Cannot build production-shape snapshot without a live DB. Run a session first.');
  process.exit(1);
}

mkdirSync(dirname(DEST), { recursive: true });
copyFileSync(SOURCE, DEST);

const db = new Database(DEST);
db.pragma('foreign_keys = OFF'); // sanitization is a structural rewrite

// (a) transcript_chunk_v6 body redaction — preserve length distribution by replacing with deterministic filler
db.prepare(`UPDATE transcript_chunk_v6 SET body = 'redacted-' || id WHERE body IS NOT NULL`).run();

// (b) Body-bearing artifact tables
const bodyTables = [
  { table: 'learnings', col: 'body' },
  { table: 'decisions', col: 'body' },
  { table: 'experience_patterns', col: 'body' },
  { table: 'observations', col: 'body' },
];
for (const { table, col } of bodyTables) {
  try {
    db.prepare(`UPDATE ${table} SET ${col} = 'redacted-' || id WHERE ${col} IS NOT NULL`).run();
  } catch (e) {
    // Table may not exist in this V32 — skip silently
  }
}

// (c) Cross-agent sessions — drop content
try { db.prepare(`DELETE FROM cross_agent_sessions`).run(); } catch (e) {}

// (d) Vec0 virtual tables — drop and recreate empty so embeddings don't leak content via reverse-engineering.
//    Schema is preserved by the original DDL captured during the file copy.
const vecTables = db
  .prepare(`SELECT name FROM sqlite_master WHERE type='table' AND name LIKE 'vec_%'`)
  .all()
  .map((r) => r.name);
for (const name of vecTables) {
  try {
    db.prepare(`DELETE FROM ${name}`).run();
  } catch (e) {
    // vec0 may have its own constraints; best-effort
  }
}

// (e) VACUUM to compact and remove freed pages
db.exec('VACUUM');

db.close();
console.log(`[snapshot] Built ${DEST}`);
console.log('Note: this snapshot is committed to the repo for WIR-promoted integration tests. Re-run only when production schema changes.');
