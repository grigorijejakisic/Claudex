export interface PerProbeOutcome {
  probe_id: string;
  r1_a_arm_pass: boolean;
  r1_b_arm_pass: boolean;
  r2_a_arm_pass: boolean;
  r2_b_arm_pass: boolean;

export interface McNemarResult {
  /** Number of probes where A-arm passed but B-arm failed (across both replications, paired) */
  a_only: number;
  /** Number of probes where B-arm passed but A-arm failed */
  b_only: number;
  /** Discordant pairs total (a_only + b_only) — load-bearing for power */
  discordant_pairs: number;
  /** McNemar exact two-sided p-value (binomial test on b_only out of discordant_pairs) */
  p_value: number;
  /** Pre-committed minimum-discordant-pair threshold from CONTEXT § Methodology critique #2 */
  min_discordant_threshold: number;
  /** Verdict per the pre-committed conditional outcomes table */
  verdict: 'BIND_POSITIVE' | 'BIND_NEGATIVE' | 'INCONCLUSIVE';
  /** Per-replication breakdown for transparency */
  by_replication: Array<{ replication: 1 | 2; a_pass: number; b_pass: number; n: number }>;

```ts
// verdict.ts — replaces poolReplications
export function pairedMcNemar(
  outcomes: PerProbeOutcome[],
  options: { min_discordant_threshold?: number } = {},
): McNemarResult {
  const minThreshold = options.min_discordant_threshold ?? 5; // pre-committed per CONTEXT § Methodology critique #2

  // Pair r1 + r2 per probe into a single binary pass (probe passes B-arm iff EITHER replication's B-arm passes).
  // OR-aggregation chosen so two replications increase signal; the alternative (require both to pass)
  // is too strict given known judge variance. Document choice.
  let a_only = 0;
  let b_only = 0;
  for (const o of outcomes) {
    const a_pass = o.r1_a_arm_pass || o.r2_a_arm_pass;
    const b_pass = o.r1_b_arm_pass || o.r2_b_arm_pass;
    if (a_pass && !b_pass) a_only++;
    else if (b_pass && !a_pass) b_only++;
    // concordant pairs (both pass or both fail) drop out of McNemar
  }
  const discordant_pairs = a_only + b_only;

  // McNemar exact: under H0 (no difference), b_only ~ Binomial(discordant_pairs, 0.5).
  // Two-sided p-value: 2 × P(X ≤ min(a_only, b_only) | n=discordant_pairs, p=0.5).
  let p_value = 1.0;
  if (discordant_pairs > 0) {
    const k = Math.min(a_only, b_only);
    p_value = Math.min(1, 2 * binomialCdf(k, discordant_pairs, 0.5));
  }

  let verdict: McNemarResult['verdict'];
  if (discordant_pairs < minThreshold) {
    verdict = 'INCONCLUSIVE';
  } else if (p_value < 0.05 && b_only > a_only) {
    verdict = 'BIND_POSITIVE';
  } else if (p_value < 0.05 && a_only > b_only) {
    verdict = 'BIND_NEGATIVE';
  } else {
    verdict = 'INCONCLUSIVE';
  }

  // Per-replication breakdown for transparency
  const by_replication = [1, 2].map((r) => {
    let a_pass = 0, b_pass = 0;
    for (const o of outcomes) {
      const aKey = r === 1 ? 'r1_a_arm_pass' : 'r2_a_arm_pass';
      const bKey = r === 1 ? 'r1_b_arm_pass' : 'r2_b_arm_pass';
      if (o[aKey as keyof PerProbeOutcome]) a_pass++;
      if (o[bKey as keyof PerProbeOutcome]) b_pass++;
    }
    return { replication: r as 1 | 2, a_pass, b_pass, n: outcomes.length };
  });

  return { a_only, b_only, discordant_pairs, p_value, min_discordant_threshold: minThreshold, verdict, by_replication };
}

// Binomial CDF — exact, not approximation. n is small (≤ 60), no numerical issues.
function binomialCdf(k: number, n: number, p: number): number {
  let cdf = 0;
  for (let i = 0; i <= k; i++) {
    cdf += binomialPmf(i, n, p);
  }
  return cdf;
}
function binomialPmf(k: number, n: number, p: number): number {
  return logCombination(n, k) > 700 ? 0 : Math.exp(logCombination(n, k)) * Math.pow(p, k) * Math.pow(1 - p, n - k);
}
function logCombination(n: number, k: number): number {
  let log = 0;
  for (let i = 1; i <= k; i++) log += Math.log(n - i + 1) - Math.log(i);
  return log;
}
```

export interface JudgeVerdict {

  pass: boolean | null; // null = error / unparseable response
  raw: string; // full judge response, kept for audit
  error?: string; // populated when pass=null

export interface EnsembleVerdict {
  /** Per-judge verdicts for this probe */
  per_judge: JudgeVerdict[];
  /** True if 3-of-4 (or 2-of-3 in fallback) of the non-error judges voted pass */
  pass: boolean | null;
  /** Number of judges that errored on this probe */
  error_count: number;
  /** Whether the ensemble used the 3-of-3 fallback (one judge dropped run-wide) */
  fallback_active: boolean;
  /** Run-level: which judge (if any) was dropped due to >10% error rate run-wide */
  dropped_judge?: JudgeIdentity['name'];
}

export async function adjudicateWithEnsemble(
  probe: Probe,
  rubric: ProbeOutcomeRubric,
  context: { aArmContent: string; bArmContent: string },
  options: { dropped_judge?: JudgeIdentity['name'] } = {},
): Promise<EnsembleVerdict> {
  const activeJudges = JUDGES.filter((j) => j.name !== options.dropped_judge);
  const verdicts = await Promise.all(
    activeJudges.map((j) => callSingleJudge(j, probe, rubric, context)),
  );
  const validVerdicts = verdicts.filter((v) => v.pass !== null);
  const passCount = validVerdicts.filter((v) => v.pass === true).length;
  const errorCount = verdicts.length - validVerdicts.length;

  // Majority threshold: ceil(activeJudges.length / 2) + adjustment for ties
  // 4 judges → 3 of 4. 3 judges (fallback) → 2 of 3.
  const majorityThreshold = Math.floor(activeJudges.length / 2) + 1;

  let pass: boolean | null;
  if (validVerdicts.length < majorityThreshold) {
    // Too many errors on this probe — can't reach majority; mark probe-level error
    pass = null;
  } else if (passCount >= majorityThreshold) {
    pass = true;
  } else {
    pass = false;
  }

  return {
    per_judge: verdicts,
    pass,
    error_count: errorCount,
    fallback_active: !!options.dropped_judge,
    dropped_judge: options.dropped_judge,
  };
}

// Run-level orchestration: detect >10% per-judge error rate and drop highest-error judge for the rest of the run.
export interface RunErrorBudget {
  /** Map judge → error count over the run so far */
  errorsByJudge: Record<string, number>;
  totalProbes: number;
  thresholdPct: number; // 10 per CONTEXT
}

export function computeRunFallback(budget: RunErrorBudget): { dropped_judge?: JudgeIdentity['name']; inconclusive: boolean } {
  const overThreshold = Object.entries(budget.errorsByJudge)
    .filter(([_, n]) => n / budget.totalProbes > budget.thresholdPct / 100)
    .map(([name]) => name as JudgeIdentity['name']);
  if (overThreshold.length === 0) return {};
  if (overThreshold.length === 1) return { dropped_judge: overThreshold[0] };
  // > 1 judge over threshold → INCONCLUSIVE per CONTEXT § Methodology critique #6
  return { inconclusive: true };
}
