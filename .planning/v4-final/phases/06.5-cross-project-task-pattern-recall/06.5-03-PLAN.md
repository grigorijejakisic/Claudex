    **Step 2 — `src/shared/claude-md-flags.ts` (NEW).**

    ```typescript
    import * as fs from 'fs';
    import * as path from 'path';

    /**
     * Phase 6.5 RETR-07: per-project opt-out via CLAUDE.md flag
     *   claudex.cross_project_search: false
     *
     * Default: true. Returns false ONLY when an explicit `claudex.cross_project_search: false`
     * line exists in the project's CLAUDE.md.
     *
     * Resilient to missing CLAUDE.md (returns true).
     */
    export function readCrossProjectSearchFlag(projectRoot: string): boolean {
      try {
        const claudeMdPath = path.join(projectRoot, 'CLAUDE.md');
        if (!fs.existsSync(claudeMdPath)) return true;
        const content = fs.readFileSync(claudeMdPath, 'utf8');
        const match = content.match(/^\s*claudex\.cross_project_search\s*:\s*(true|false)\s*$/im);
        if (match && match[1].toLowerCase() === 'false') return false;
        return true;
      } catch {
        return true;
      }
    }
    ```

    Single-call, no caching at the parser level (cheap; called once per `claudex_search` invocation; assembler-level caching can be added later if needed).
