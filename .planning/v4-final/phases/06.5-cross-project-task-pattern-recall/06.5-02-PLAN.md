    ```typescript
    import type { Database } from 'better-sqlite3';
    import { recordTelemetry } from './telemetry-counters.js';

    export interface HandleSet {
      tools_used: string[];
      files_touched: string[];
      user_framing_tokens: string[];
      errors_encountered: string[];
    }

    export interface EquivalenceResult {
      match: boolean;
      stage1Shared: number;
      stage2Cosine: number | null;
      band: 'match' | 'ambiguous' | 'reject' | 'stage1-fail';
    }

    export const STAGE_1_THRESHOLD = 3;
    export const STAGE_2_MATCH_THRESHOLD = 0.85;
    export const STAGE_2_AMBIGUOUS_FLOOR = 0.70;

    export function stageOneHandleOverlap(a: HandleSet, b: HandleSet): number {
      // Sum of |intersect| across all four handle dimensions
      // files_touched normalized: take basename + immediate parent dir as the unit
      // (so 'src/scraper.ts' shares with 'src/scraper-v2.ts' via the basename token,
      //  and shares with 'src/main.ts' via the 'src/' parent)
      // tools_used + framing_tokens + errors_encountered: plain set intersect
    }

    export async function stageTwoCosine(
      salienceA: string,
      salienceB: string,
    ): Promise<number | null>; // null on embedder failure (don't crash; reject equivalence)

    export function logAmbiguous(
      db: Database,
      sessionId: string,
      aId: number,
      bId: number,
      cosine: number,
      projectA: string,
      projectB: string,
    ): void;

    export async function isCrossProjectEquivalent(
      a: HandleSet & { id: number; project: string; salience: string },
      b: HandleSet & { id: number; project: string; salience: string },
      db: Database,
      sessionId: string,
    ): Promise<EquivalenceResult>;
    ```

    **Stage 1 logic:**
    - Normalize `files_touched`: each path becomes a 2-element token list `[basename, parentDir]`. Aggregate across the array. Set intersect.
    - For `tools_used`, `user_framing_tokens`, `errors_encountered`: lowercase-trim then plain Set intersect.
    - Total shared = sum of |intersect| across all four dimensions. Threshold ≥ 3.

    **Stage 2 logic:**
    - Reuse the existing `/api/embed` Ollama HTTP client used in `src/core/hybrid-retrieval.ts` (find the helper; if it's not exported, copy the call shape exactly — same model: `snowflake-arctic-embed2`, same dimension: 1024, same 1-second timeout).
    - Compute cosine: `dot(a,b) / (norm(a) * norm(b))` inline (1024 dims; trivial).
    - On any embedder failure (timeout, non-2xx, empty response): return `null` and log a `reranker_fallback` telemetry row with reason=`embedder_unreachable_for_equivalence` (reuses Phase 6 RETR-08 telemetry surface; do NOT add a new event_kind for this case).

    **`isCrossProjectEquivalent` orchestration:**
    1. Stage 1 → if shared < 3 → return `{ match: false, band: 'stage1-fail' }`
    2. Stage 2 cosine → if null (embedder failed) → return `{ match: false, band: 'reject', stage2Cosine: null }`
    3. cosine ≥ 0.85 → `{ match: true, band: 'match' }`
    4. 0.70 ≤ cosine < 0.85 → call `logAmbiguous(...)` (writes `event_kind='cross_project_ambiguous'` row from V21 enum) → return `{ match: false, band: 'ambiguous' }`
    5. cosine < 0.70 → `{ match: false, band: 'reject' }`

    **Candidate query (planner-tuned SQL; see RESEARCH.md):**
    ```sql
    SELECT a.id, a.project, a.summary, a.content,
           atp.task_pattern, atp.classifier_confidence,
           a.timestamp_epoch,
           CASE WHEN a.timestamp_epoch >= unixepoch('now', '-14 days') THEN 1 ELSE 0 END AS last_used_within_14d,
           NULL AS helpful_yn,                                    -- joined later if pointer matches
           0 AS already_injected                                  -- filled by post-query lookup
    FROM artifacts a
    INNER JOIN artifact_task_pattern atp ON atp.artifact_id = a.id
    WHERE atp.task_pattern != '__abstain__'
      AND a.project != ?            -- different project = cross-project signal
      AND a.artifact_type IN ('learning', 'observation', 'memory_file')
    ORDER BY a.timestamp_epoch DESC
    LIMIT 200                        -- bounded candidate pool
    ```
