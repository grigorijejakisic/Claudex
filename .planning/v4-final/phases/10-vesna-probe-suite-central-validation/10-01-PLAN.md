```ts
export type ProbeCategory =
  | "entity-recall"
  | "constraint-recall"
  | "handoff-pickup"
  | "cross-project"
  | "lesson-application"
  | "self-instrumented"
  | "buffer";

export interface ExpectedRecall {
  artifact_id_or_pattern: string;        // e.g., "project_mozzart_cloudflare_429.md" or regex pattern
  must_surface_within_turns: number;     // 1 or 2 typically
  must_contain_phrase_pattern: string[]; // regex strings, OR-matched
}

export interface Probe {
  id: string;                            // convention: <category>-<3-digit>, e.g., "entity-001"
  category: ProbeCategory;
  source_session_id: string;             // provenance — real session this was mined from
  source_project: string;                // claudex-v3, lacuna-betting, oracle, big-mozzy-v2, desktop-01dcc792, nexus
  source_turn_idx?: number;              // optional — turn within source session
  scenario: string;                      // human-readable description of the retrieval moment
  user_prompt: string;                   // exact text shown to the agent
  expected_recall: ExpectedRecall;
  lexical_exclusions: string[];          // tokens that MUST NOT appear in user_prompt
  evaluation: "auto" | "semi-auto";
  setup_steps?: SetupStep[];             // optional — if omitted, no DB setup
  buffer_placeholder?: boolean;          // true for unallocated buffer slots — skipped, not run
}

export type SetupStep =
  | { kind: "artifact"; payload: { kind: "decision" | "learning" | "observation"; summary: string; project: string; tags?: string[] } }
  | { kind: "handoff"; payload: { status: "active" | "paused" | "archived"; phase: string; summary: string; topic: string; body_what_next?: string } }
  | { kind: "critical_rule"; payload: { rule: string; project?: string } }
  | { kind: "narration_directive"; payload: { silent: boolean } };

export interface ProbeTrialResult { passed: boolean; diagnostic: string; agent_output: string; turns_taken: number; }
export interface ProbeResult { probe_id: string; category: ProbeCategory; trials: ProbeTrialResult[]; verdict: "pass" | "fail" | "flaky"; }
export interface SuiteReport {
  aggregate_pass_rate: number;
  per_category: Record<ProbeCategory, { pass_rate: number; total: number; passed: number; flaky: number }>;
  flaky_probes: string[];
  failed_probes: { id: string; category: ProbeCategory; diagnostics: string[] }[];
  gated: boolean; // true iff aggregate ≥0.8 AND every category ≥0.8 (excluding buffer placeholders)
}
```

Create `setup.ts` — the deterministic, idempotent DB-population DSL:
- `applySetup(db: Database, steps: SetupStep[]): Promise<void>` — applies each step in order against an isolated test DB.
- `resetTestDb(db: Database): Promise<void>` — wipes probe-scoped state between probes (claudex_artifacts rows tagged with `probe_test=1`, ACTIVE.md temp file, narration_directive flag) WITHOUT touching production tables.
- Step handlers:
  - `kind: "artifact"` → INSERT INTO claudex_artifacts (kind, summary, project, tags, probe_test) VALUES (...). Tag `probe_test=1` for cleanup.
  - `kind: "handoff"` → write to a probe-scoped ACTIVE.md fixture path (e.g., `tests/.tmp/probe-active.md`); harness uses an env var or override to point lifecycle at it.
  - `kind: "critical_rule"` → INSERT INTO critical_rules with `probe_test=1` tag.
  - `kind: "narration_directive"` → set narration_silent flag in test config; reset on `resetTestDb`.
- Use a separate test DB path (e.g., `~/.claudex/db/claudex-vesna-test.db`) — NEVER touches the production `claudex.db`. Migrate the test DB on first run using existing migration runner.
- Idempotent: applySetup → applySetup with same steps → identical state.

Create `runner.ts`:
- `runProbe(probe: Probe, opts: { trials: number; modelConfig: { model: string; temperature: number; seed?: number } }): Promise<ProbeResult>`
  - For trials in 1..opts.trials (default 3):
    - resetTestDb()
    - applySetup(probe.setup_steps ?? [])
    - Invoke an agent with `probe.user_prompt`. The agent invocation MUST use the actual session-start + user-prompt-submit pipeline (hybrid retrieval, context injection) so probes test the real production surface, not a mocked one. Use a thin wrapper around `src/adapters/cc-hooks/lifecycle.ts` exports — capture the assembled context payload + a single agent turn output.
    - For v4 simplification: model output is collected via a deterministic test agent (use `~/.claudex/db/claudex-vesna-test.db` + Ollama bi-encoder fallback path is acceptable when reranker unavailable; otherwise use the reranker on port 7439). Record `agent_text`, `turns_taken`, `tool_calls[]`.
    - evaluate(output, probe.expected_recall) → trial result.
  - Verdict: `pass` if all 3 trials pass; `fail` if all 3 fail; `flaky` if 1/3 or 2/3.
  - Per CONTEXT line 134: flaky probes do NOT gate the suite (they're tagged for investigation).

Create `index.ts` — top-level orchestrator:
- `runVesnaSuite(opts: { probesDir?: string; trials?: number; modelConfig?: ... }): Promise<SuiteReport>`
- Flow: loadProbes → for each non-buffer probe, runProbe → aggregate per-category counts → compute aggregate_pass_rate (passes / (passes + fails), excluding flaky from denominator) → compute per_category pass rates → set `gated = aggregate ≥ 0.8 && every non-empty category ≥ 0.8` → return SuiteReport.
- Buffer-placeholder probes are excluded from totals (they're slots, not probes).
- Empty categories (no real probes) are excluded from per-category gate (so phase rollout doesn't fail before all categories populated).

- applySetup with `kind: "handoff"` writes the fixture file.
- resetTestDb deletes probe_test=1 rows but preserves untagged rows (production-data-safety check).
- Idempotency: applySetup × 2 → same state.
- Use a separate vitest setup that points at the test DB path.
