 118:   it('filters by project (including __global__)', () => {

 188:   let db: TestDatabase;

 209:     createSession(db, { session_id: 's1', project: 'proj1' });
 210:     createSession(db, { session_id: 's2', project: 'proj1' });
 211:     insertObservation(db, {
 212:       session_id: 's1', project: 'proj1', tool_name: 'Read',
 213:       category: 'code', title: 'Read file', content: 'read src/a.ts',
 214:       importance: 3, files_modified: ['src/a.ts'],
 215:     });
 216:     insertObservation(db, {
 217:       session_id: 's1', project: 'proj1', tool_name: 'Edit',
 218:       category: 'architecture', title: 'Edit file', content: 'edited',
 219:       importance: 4, files_modified: ['src/b.ts'],
 220:     });
 221:     insertDecision(db, { session_id: 's1', project: 'proj1', content: 'dec1', source: 'confirmation', fingerprint: 'fp1' });
 222:     upsertLearning(db, { project: 'proj1', fingerprint: 'fp1', content: 'learn1' });
 223: 
 224:     const result = queryStats(db);
 225:     expect(result.total_observations).toBe(2);
 226:     expect(result.total_decisions).toBe(1);

 228:     expect(result.total_sessions).toBe(2);
 229:   });
 230: 
 231:   it('returns top categories ordered by count', () => {
 232:     createSession(db, { session_id: 's1', project: 'proj1' });
 233:     for (let i = 0; i < 5; i++) {
 234:       insertObservation(db, {
 235:         session_id: 's1', project: 'proj1', tool_name: 'Read',
 236:         category: 'code', title: `code ${i}`, content: `code content ${i}`,
 237:         importance: 3, files_modified: [],
 238:       });
 239:     }
 240:     for (let i = 0; i < 3; i++) {
 241:       insertObservation(db, {
 242:         session_id: 's1', project: 'proj1', tool_name: 'Bash',
 243:         category: 'error', title: `err ${i}`, content: `error content ${i}`,
 244:         importance: 4, files_modified: [],
 245:       });
 246:     }
 247: 
 248:     const result = queryStats(db);
 249:     expect(result.top_categories).toHaveLength(2);
 250:     expect(result.top_categories[0].category).toBe('code');
 251:     expect(result.top_categories[0].count).toBe(5);
 252:     expect(result.top_categories[1].category).toBe('error');

 254:   });
 255:

 257:     createSession(db, { session_id: 's1', project: 'proj1' });
 258:     createSession(db, { session_id: 's2', project: 'proj2' });
 259:     insertObservation(db, {
 260:       session_id: 's1', project: 'proj1', tool_name: 'Read',
 261:       category: 'code', title: 'p1 obs', content: 'content',
 262:       importance: 3, files_modified: [],
 263:     });
 264:     insertObservation(db, {
 265:       session_id: 's2', project: 'proj2', tool_name: 'Read',
 266:       category: 'code', title: 'p2 obs', content: 'content',
 267:       importance: 3, files_modified: [],

 405: // ── runDashboard integration ─────────────────────────────────────

 407: describe('runDashboard', () => {
 408:   let db: InstanceType<typeof Database>;
 409:   let dbPath: string;
 410: 
 411:   beforeEach(() => {
 412:     // Create an on-disk temp DB for runDashboard (which opens by path)
 413:     const tmpDir = require('fs').mkdtempSync(require('path').join(require('os').tmpdir(), 'claudex-dashboard-'));
 414:     dbPath = require('path').join(tmpDir, 'test.db');
 415:     db = new Database(dbPath);
 416:     initializeSchema(db);
 417:   });
 418: 
 419:   afterEach(() => {
 420:     db.close();
 421:     try {
 422:       require('fs').rmSync(require('path').dirname(dbPath), { recursive: true, force: true });
 423:     } catch {
 424:       // best effort
 425:     }
 426:   });
 427: 
 428:   it('returns help text for "help" subcommand', () => {
 429:     const output = runDashboard({ subcommand: 'help' }, dbPath);
 430:     expect(output).toContain('Usage:');
 431:     expect(output).toContain('learnings');
 432:     expect(output).toContain('decisions');
 433:     expect(output).toContain('stats');
 434:     expect(output).toContain('topics');
 435:   });
 436: 
 437:   it('returns learnings output', () => {
 438:     upsertLearning(db, { project: 'proj1', fingerprint: 'fp1', content: 'Test learning' });
 439: 
 440:     const output = runDashboard({ subcommand: 'learnings' }, dbPath);
 441:     expect(output).toContain('Test learning');
 442:     expect(output).toContain('=== Learnings ===');
 443:   });
 444: 
 445:   it('returns decisions output with project filter', () => {
 446:     insertDecision(db, { session_id: 's1', project: 'proj1', content: 'Dec1', source: 'confirmation', fingerprint: 'fp1' });
 447:     insertDecision(db, { session_id: 's2', project: 'proj2', content: 'Dec2', source: 'direction', fingerprint: 'fp2' });
 448: 
 449:     const output = runDashboard({ subcommand: 'decisions', project: 'proj1' }, dbPath);
 450:     expect(output).toContain('Dec1');
 451:     expect(output).not.toContain('Dec2');
 452:   });
 453: 
 454:   it('returns stats output', () => {
 455:     createSession(db, { session_id: 's1', project: 'proj1' });

 457:       session_id: 's1', project: 'proj1', tool_name: 'Read',

 459:       importance: 3, files_modified: [],

 465:     expect(output).toContain('Observations: 1');
 466:   });
 467: 
 468:   it('returns topics output', () => {
 469:     upsertThreadState(db, { session_id: 's1', topic: 'My Topic', summary: 'My Summary' });
 470: 
 471:     const output = runDashboard({ subcommand: 'topics' }, dbPath);
 472:     expect(output).toContain('=== Topic History ===');
