  54: describe('patchSettingsJson', () => {

  57:   beforeEach(() => {
  58:     tmpDir = createTmpDir();
  59:   });
  60: 
  61:   afterEach(() => {
  62:     fs.rmSync(tmpDir, { recursive: true, force: true });
  63:   });
  64: 
  65:   const hookPaths: Record<string, string> = {
  66:     SessionStart: '/opt/CLAUDEXv3/dist/session-start.js',
  67:     UserPromptSubmit: '/opt/CLAUDEXv3/dist/user-prompt-submit.js',
  68:     PostToolUse: '/opt/CLAUDEXv3/dist/post-tool-use.js',
  69:     Stop: '/opt/CLAUDEXv3/dist/stop.js',
  70:     PreCompact: '/opt/CLAUDEXv3/dist/pre-compact.js',
  71:     SessionEnd: '/opt/CLAUDEXv3/dist/session-end.js',
  72:   };
  73: 
  74:   it('creates new settings.json with hooks when file does not exist', () => {
  75:     const settingsPath = path.join(tmpDir, 'settings.json');
  76:     const result = patchSettingsJson(settingsPath, hookPaths);
  77: 
  78:     expect(result.patched).toBe(true);
  79:     expect(result.created).toBe(true);
  80: 
  81:     const settings = JSON.parse(fs.readFileSync(settingsPath, 'utf-8'));
  82:     expect(Object.keys(settings.hooks)).toHaveLength(6);
  83:     expect(settings.hooks.SessionStart).toHaveLength(1);
  84:   });
  85: 
  86:   it('merges hooks into existing settings.json preserving other keys', () => {
  87:     const settingsPath = path.join(tmpDir, 'settings.json');
  88:     fs.writeFileSync(settingsPath, JSON.stringify({ someOtherKey: true, hooks: {} }));
  89: 
  90:     patchSettingsJson(settingsPath, hookPaths);
  91: 
  92:     const settings = JSON.parse(fs.readFileSync(settingsPath, 'utf-8'));
  93:     expect(settings.someOtherKey).toBe(true);
  94:     expect(Object.keys(settings.hooks)).toHaveLength(6);
  95:   });
  96: 
  97:   it('preserves existing non-Claudex hooks', () => {
  98:     const settingsPath = path.join(tmpDir, 'settings.json');
  99:     fs.writeFileSync(
 100:       settingsPath,
 101:       JSON.stringify({
 102:         hooks: {
 103:           PreToolUse: [
 104:             { matcher: '', hooks: [{ type: 'command', command: 'echo pre-tool' }] },
 105:           ],
 106:         },
 107:       })
 108:     );
 109: 
 110:     patchSettingsJson(settingsPath, hookPaths);
 111: 
 112:     const settings = JSON.parse(fs.readFileSync(settingsPath, 'utf-8'));
 113:     // PreToolUse preserved + 6 Claudex hooks
 114:     expect(settings.hooks.PreToolUse).toHaveLength(1);
 115:     expect(settings.hooks.SessionStart).toHaveLength(1);
 116:   });
 117: 
 118:   it('updates existing Claudex hooks in-place (no duplicates)', () => {
 119:     const settingsPath = path.join(tmpDir, 'settings.json');
 120: 
 121:     // First patch
 122:     patchSettingsJson(settingsPath, hookPaths);
 123: 
 124:     // Second patch with different install dir

 148:   it('generated command uses node with quoted absolute path', () => {
 149:     const settingsPath = path.join(tmpDir, 'settings.json');
 150:     patchSettingsJson(settingsPath, hookPaths);
 151: 
 152:     const settings = JSON.parse(fs.readFileSync(settingsPath, 'utf-8'));

 235:     const settingsPath = path.join(tmpDir, 'settings.json');

 237: 
 238:     const settings = JSON.parse(fs.readFileSync(settingsPath, 'utf-8'));
